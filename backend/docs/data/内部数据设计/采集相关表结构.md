collect_fetchers        元数据，只读，启动时校验代码一致性
  task_type (PK)        'cap_top_list'          ← 与代码里 registry 的 key 一致
  name, label, facet, sub_facet, description
  source                'Tushare' / 'Pytdx' / 'Adata-THS'
  protocol              'TopListFetcher'
  method                'fetch_top_list'
  default_params        {"days": 1}
  target_table          'cap_top_lists'
  status                ready / planned

collect_plans           触发配置
  id (PK)
  name                   '盘前同步'
  schedule_type          time / interval
  times                  ["09:30", "15:00"]
  interval_seconds
  enabled
  stop_on_fail           ← 吸收自 groups
  last_run_at, last_task_id

collect_plan_items      plan ↔ fetcher 关联
  id (PK)
  plan_id (FK)
  task_type (FK → collect_fetchers.task_type)
  params                 {"days": 1}
  enabled                ← 单项临时停用
  sort_order             ← 执行顺序
