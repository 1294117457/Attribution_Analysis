## 3. 关键交互流程

### 3.1 账户数据 Tab

```
初始化 │
   ↓
  loadUsers()  → GET /api/v1/users/?page=1&page_size=20
   │
   ↓
  表格渲染
   │
用户搜索 │ 用户点"启用/禁用"
   ↓   ↓
  loadUsers()  toggleStatus(row)
   │           │
   │           ├─ 弹 ElMessageBox.confirm
   │           ├─ 调 PATCH /users/{id}/status
   │           └─ 重新 loadUsers
   │
用户点"分配角色"   用户点"重置密码"
   ↓               ↓
  openAssignRole   openResetPassword
  Dialog 弹窗       Dialog 弹窗
```

### 3.2 角色 Tab

```
切到"角色" Tab → loadRoles() → GET /api/v1/roles/
  │
  └─ 表格：ID | Code | Name
```

### 3.3 权限 Tab

```
切到"权限" Tab → loadPermissions() → GET /api/v1/permissions/
  │
  └─ 表格：ID | Code | Resource | Action | Name
```

## 4. 关键代码骨架

```vue
<!-- frontend/src/views/auth/AccountPage.vue -->
<template>
  <PageWrapper>
    <template #title>
      <el-icon><Setting /></el-icon>
      <span>账户管理</span>
    </template>

    <el-tabs v-model="activeTab" class="account-tabs">
      <!-- ── Tab 1: 账户数据 ── -->
      <el-tab-pane label="账户数据" name="users">
        <div class="toolbar">
          <el-input
            v-model="searchKeyword"
            placeholder="搜索邮箱/用户名/昵称"
            style="width: 240px"
            clearable
            @clear="loadUsers"
            @keyup.enter="loadUsers"
          />
          <el-select
            v-model="filterActive"
            placeholder="状态"
            clearable
            style="width: 120px; margin-left: 12px"
            @change="loadUsers"
          >
            <el-option label="启用" :value="true" />
            <el-option label="禁用" :value="false" />
          </el-select>
          <el-button
            type="primary"
            @click="loadUsers"
            style="margin-left: 12px"
          >
            搜索
          </el-button>
        </div>

        <el-table
          :data="userList"
          v-loading="loading"
          stripe
          style="width: 100%"
        >
          <el-table-column prop="id" label="ID" width="80" />
          <el-table-column prop="email" label="邮箱" min-width="180" />
          <el-table-column prop="username" label="用户名" width="120" />
          <el-table-column prop="nickname" label="昵称" width="120" />
          <el-table-column label="角色" width="200">
            <template #default="{ row }">
              <el-tag
                v-for="r in row.roles"
                :key="r.id"
                size="small"
                style="margin-right: 4px"
              >
                {{ r.name }}
              </el-tag>
            </template>
          </el-table-column>
          <el-table-column label="状态" width="100">
            <template #default="{ row }">
              <el-tag
                :type="row.is_active ? 'success' : 'danger'"
                size="small"
              >
                {{ row.is_active ? '启用' : '禁用' }}
              </el-tag>
            </template>
          </el-table-column>
          <el-table-column prop="last_login_at" label="最后登录" width="170">
            <template #default="{ row }">
              {{ row.last_login_at ? new Date(row.last_login_at).toLocaleString() : '-' }}
            </template>
          </el-table-column>
          <el-table-column label="操作" width="240" fixed="right">
            <template #default="{ row }">
              <el-button size="small" @click="openAssignRoleDialog(row)">
                分配角色
              </el-button>
              <el-button
                size="small"
                :type="row.is_active ? 'warning' : 'success'"
                @click="toggleStatus(row)"
              >
                {{ row.is_active ? '禁用' : '启用' }}
              </el-button>
              <el-button
                size="small"
                type="danger"
                @click="openResetPasswordDialog(row)"
              >
                重置密码

            </template>
          </el-table-column>
        </el-table>

          v-model:page-size="pageSize"
          :page-sizes="[10, 20, 50, 100]"
          :total="total"
          layout="total, sizes, prev, pager, next, jumper"
          @current-change="loadUsers"
          @size-change="loadUsers"

      <el-tab-pane label="角色" name="roles">

          <el-table-column prop="id" label="ID" width="80" />

        </el-table>

    <!-- 分配角色弹窗 -->
    <el-dialog v-model="roleDialogVisible" title="分配角色" width="420">

        </el-checkbox>
      </el-checkbox-group>

        <el-button @click="roleDialogVisible = false">取消</el-button>
        <el-button type="primary" @click="confirmAssignRole">保存</el-button>
      </template>
    </el-dialog>

    <!-- 重置密码弹窗 -->
    <el-dialog v-model="pwdDialogVisible" title="重置密码" width="420">
      <p>为 <strong>{{ currentUser?.email }}</strong> 重置密码：</p>
      <el-input
        v-model="newPassword"
        type="password"
        placeholder="新密码（至少 8 位）"
        show-password
      />
      <template #footer>
        <el-button @click="pwdDialogVisible = false">取消</el-button>
        <el-button type="primary" @click="confirmResetPassword">重置</el-button>
      </template>
    </el-dialog>
  </PageWrapper>
</template>

<script setup lang="ts">
import { ref, onMounted, watch } from 'vue'
import { ElMessage, ElMessageBox } from 'element-plus'
import { Setting } from '@element-plus/icons-vue'
import PageWrapper from '@/components/PageWrapper.vue'
import { userApi, roleApi, permissionApi } from '@/api/auth'

const activeTab = ref('users')

// 账户数据
const userList = ref<any[]>([])
const loading = ref(false)
const page = ref(1)
const pageSize = ref(20)
const total = ref(0)
const searchKeyword = ref('')
const filterActive = ref<boolean | undefined>()

// 角色和权限
const roleList = ref<any[]>([])
const permissionList = ref<any[]>([])

// 弹窗
const roleDialogVisible = ref(false)
const pwdDialogVisible = ref(false)
const currentUser = ref<any>(null)
const selectedRoleIds = ref<number[]>([])
const newPassword = ref('')

async function loadUsers() {
  loading.value = true
  try {
    const result = await userApi.listUsers({
      page: page.value,
      page_size: pageSize.value,
      keyword: searchKeyword.value || undefined,
      is_active: filterActive.value,
    })
    userList.value = result.items
    total.value = result.total
  } finally {
    loading.value = false
  }
}

async function loadRoles() {
  roleList.value = await roleApi.listRoles()
}

async function loadPermissions() {
  permissionList.value = await permissionApi.listPermissions()
}

function openAssignRoleDialog(user: any) {
  currentUser.value = user
  selectedRoleIds.value = user.roles.map((r: any) => r.id)
  roleDialogVisible.value = true
}

async function confirmAssignRole() {
  const newIds = selectedRoleIds.value
  const oldIds = currentUser.value.roles.map((r: any) => r.id)

  // 计算差集
  const toAdd = newIds.filter((id) => !oldIds.includes(id))
  const toRemove = oldIds.filter((id: number) => !newIds.includes(id))

  try {
    for (const rid of toAdd) {
      await userApi.assignRole(currentUser.value.id, rid)
    }
    for (const rid of toRemove) {
      await userApi.removeRole(currentUser.value.id, rid)
    }
    ElMessage.success('角色已更新')
    roleDialogVisible.value = false
    loadUsers()
  } catch (e: any) {
    ElMessage.error(e.message || '更新失败')
  }
}

async function toggleStatus(user: any) {
  const action = user.is_active ? '禁用' : '启用'
  try {
    await ElMessageBox.confirm(
      `确认${action}用户 ${user.email}?`,
      '提示',
      { type: 'warning' },
    )
    await userApi.updateStatus(user.id, !user.is_active)
    ElMessage.success(`已${action}`)
    loadUsers()
  } catch (e: any) {
    if (e !== 'cancel') ElMessage.error(e.message || '操作失败')
  }
}

function openResetPasswordDialog(user: any) {
  currentUser.value = user
  newPassword.value = ''
  pwdDialogVisible.value = true
}

async function confirmResetPassword() {
  if (newPassword.value.length < 8) {
    ElMessage.error('密码至少 8 位')
    return
  }
  try {
    await userApi.resetPassword(currentUser.value.id, newPassword.value)
    ElMessage.success('密码已重置')
    pwdDialogVisible.value = false
  } catch (e: any) {
    ElMessage.error(e.message || '重置失败')
  }
}

// 切换 Tab 懒加载
watch(activeTab, (tab) => {
  if (tab === 'roles' && roleList.value.length === 0) loadRoles()
  if (tab === 'permissions' && permissionList.value.length === 0) loadPermissions()
})

onMounted(() => {
  loadUsers()
})
</script>

<style scoped>

  padding: 0 16px;
}
.toolbar {
  display: flex;
  align-items: center;
  margin-bottom: 16px;
}
</style>
```

## 5. 关键设计

| 项 | 选型 | Why |
|---|---|---|
| 用 `PageWrapper` 组件 | 复用 | 与现有页面风格一致 |
| 3 个 Tab 用 `<el-tabs>` | element-plus | 用户需求"账户数据 + 权限 Tab" |
| 启用/禁用弹 `ElMessageBox.confirm` | 防止误操作 | 安全考虑 |
| 重置密码强制 ≥ 8 位 | 前端先校验 | 减少无效请求 |
| 角色 Tab / 权限 Tab **只读** | 简化 | 角色/权限目前不需要增删 |
| `watch` Tab 懒加载 | 用户可能不看其他 tab | 性能 |
| 分配角色用 checkbox-group | 直观 | 用户可一次改多个角色 |

### 5.1 分配角色的差集算法

```typescript
const newIds = selectedRoleIds.value                                    // 用户勾选的新 ID
const oldIds = currentUser.value.roles.map((r) => r.id)                  // 当前已绑定的 ID

const toAdd = newIds.filter((id) => !oldIds.includes(id))                // 需要新增
const toRemove = oldIds.filter((id) => !newIds.includes(id))            // 需要删除
```

举例：
- 用户原本绑定 [member, viewer]
- 新勾选 [admin, member]
- toAdd = [admin]
- toRemove = [viewer]

避免 N+1 调用（一个弹窗 2 个请求足够）。
