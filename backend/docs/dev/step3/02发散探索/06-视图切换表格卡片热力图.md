### 3.2 `ViewModeSwitcher.vue`

```vue
<template>
  <el-button-group>
    <el-button
      v-for="opt in options"
      :key="opt.value"
      :type="modelValue === opt.value ? 'primary' : 'default'"
      :title="opt.label"
      @click="onChange(opt.value)"
    >
      <el-icon><component :is="opt.icon" /></el-icon>
    </el-button>
  </el-button-group>
</template>

<script setup lang="ts">
import { List, Grid, PictureFilled } from '@element-plus/icons-vue'

defineProps<{ modelValue: 'table' | 'card' | 'heatmap' }>()
const emit = defineEmits<{ (e: 'update:modelValue', v: 'table' | 'card' | 'heatmap'): void }>()

const options = [
  { value: 'table'   as const, label: '表格',  icon: List },
  { value: 'card'    as const, label: '卡片',  icon: Grid },
  { value: 'heatmap' as const, label: '热力图', icon: PictureFilled },
]

function onChange(v: 'table' | 'card' | 'heatmap') {
  emit('update:modelValue', v)
}
</script>
```

### 3.4 `StockHeatmap.vue`（热力图 — 用 ECharts treemap）

```vue
<template>
  <div ref="chartRef" class="sh-chart" v-loading="loading"></div>
</template>

<script setup lang="ts">
import { ref, watch, onMounted, onUnmounted } from 'vue'
import * as echarts from 'echarts/core'
import { TreemapChart } from 'echarts/charts'
import { TitleComponent, TooltipComponent, VisualMapComponent } from 'echarts/components'
import { CanvasRenderer } from 'echarts/renderers'
import { use } from 'echarts/core'
import type { StockRealtimeItem } from '../api'

use([TreemapChart, TitleComponent, TooltipComponent, VisualMapComponent, CanvasRenderer])

const props = defineProps<{ items: StockRealtimeItem[]; loading: boolean }>()
const emit = defineEmits<{ (e: 'select', item: StockRealtimeItem): void }>()

const chartRef = ref<HTMLDivElement>()
let chart: echarts.ECharts | null = null
let resizeObserver: ResizeObserver | null = null

function render() {
  if (!chartRef.value) return
  if (!chart) chart = echarts.init(chartRef.value)
  const data = props.items
    .filter(i => i.total_mv != null)
    .map(i => ({
      name: `${i.symbol} ${i.name || ''}`,
      value: i.total_mv,
      symbol: i.symbol,
      pct_change: i.pct_change,
      itemStyle: { color: pctColor(i.pct_change) },
    }))
  chart.setOption({
    tooltip: {
      formatter: (info: any) => {
        const d = info.data
        return `<b>${d.symbol}</b><br/>涨跌幅：${d.pct_change?.toFixed(2) ?? '—'}%<br/>市值：${(d.value / 10000).toFixed(1)} 亿`
      },
    },
    series: [{
      type: 'treemap',
      data,
      width: '100%',
      height: '100%',
      roam: false,
      nodeClick: false,
      breadcrumb: { show: false },
      label: {
        show: true,
        formatter: (info: any) => `{name|${info.name}}\n{pct|${info.data.pct_change?.toFixed(1) ?? '—'}%}`,
        rich: {
          name: { color: '#fff', fontSize: 12, fontWeight: 600, lineHeight: 18 },
          pct:  { color: '#fff', fontSize: 14, fontWeight: 700, lineHeight: 20 },
        },
      },
      upperLabel: { show: false },
      itemStyle: { borderColor: '#0f172a', borderWidth: 1, gapWidth: 1 },
    }],
  })
  chart.on('click', (params: any) => {
    const sym = params.data?.symbol
    if (sym) {
      const item = props.items.find(i => i.symbol === sym)
      if (item) emit('select', item)
    }
  })
}

function pctColor(p: number | null | undefined) {
  if (p == null) return '#475569'
  if (p > 7)   return '#dc2626'
  if (p > 3)   return '#f87171'
  if (p > 0)   return '#fca5a5'
  if (p > -3)  return '#86efac'
  if (p > -7)  return '#4ade80'
  return '#16a34a'
}

onMounted(() => {
  render()
  resizeObserver = new ResizeObserver(() => chart?.resize())
  if (chartRef.value) resizeObserver.observe(chartRef.value)
})
watch(() => props.items, render, { deep: true })
onUnmounted(() => {
  resizeObserver?.disconnect()
  chart?.dispose()
})
</script>

<style scoped>
.sh-chart { width: 100%; height: 100%; min-height: 500px; }
</style>
```

## 4. 性能与边界

| 维度 | 数值 | 备注 |
|---|---|---|
| 视图切换响应 | < 50ms | 纯 v-if 切换，无数据请求 |
| 热力图渲染 | ~200ms | 100~500 个节点 treemap |
| 卡片网格 | ~50ms | 纯 CSS grid |
| 配色 | 涨红跌绿（与现有约定一致） | 7 档渐变（>7/3~7/0~3/-3~0/-7~-3/<-7） |

## 5. 关键决策

| 问 | 答 |
|---|---|
| 为什么不自己画 SVG 热力图？ | ECharts Treemap 是项目已有依赖，且支持自动布局 / 点击 / 缩放 |
| 热度按市值还是按涨跌幅？ | **面积按市值（更直观看到大票）**，**颜色按涨跌幅（色块）** |
| 卡片网格每行几个？ | `repeat(auto-fill, minmax(180px, 1fr))` 自动响应，密度可调 |
| 视图切换是否触发请求？ | 不。数据 `items` 已是同一份引用，切换 0 延迟。 |
