class TestSchedulerJobs:
    # ① 定时模式一个时间点一个 job
    def test_sync_time_mode_registers_one_job_per_time(self):
        s = CollectScheduler("Asia/Shanghai")
        s.sync_plan(1, True, "time", ["09:30", "15:00"], None)
        assert s._scheduler.get_job("plan:1@09:30") is not None
        assert s._scheduler.get_job("plan:1@15:00") is not None
        assert s._scheduler.get_job("plan:1") is None       # 定时模式不注册裸 id

    # ② 定频模式单 job
    def test_sync_interval_mode(self):
        s = CollectScheduler("Asia/Shanghai")
        s.sync_plan(1, True, "interval", [], 300)
        assert s._scheduler.get_job("plan:1") is not None
        assert all(j.id != "plan:1@x" for j in s._scheduler.get_jobs())

    # ③ 前缀清理不误删（⚠️ 边界测试，必写）
    def test_remove_prefix_does_not_touch_similar_ids(self):
        s = CollectScheduler("Asia/Shanghai")
        s.sync_plan(1,  True, "time", ["09:30"], None)   # plan:1@09:30
        s.sync_plan(11, True, "time", ["09:30"], None)   # plan:11@09:30
        s.sync_plan(1,  False, None, [], None)            # 停用 plan 1
        assert s._scheduler.get_job("plan:1@09:30") is None
        assert s._scheduler.get_job("plan:11@09:30") is not None   # ← 关键

    # ④ next_run 取最早
    def test_next_run_returns_earliest(self):
        s = CollectScheduler("Asia/Shanghai")
        s.sync_plan(1, True, "time", ["15:00", "09:30"], None)
        nxt = s.next_run("plan:1")
        assert nxt is not None and nxt.hour == 9 and nxt.minute == 30

    # ⑤ 改配置后旧 job 被清掉
    def test_resync_clears_old_time_jobs(self):
        s = CollectScheduler("Asia/Shanghai")
        s.sync_plan(1, True, "time", ["09:30", "15:00"], None)
        s.sync_plan(1, True, "time", ["09:30"], None)     # 删掉一个时间点
        assert s._scheduler.get_job("plan:1@15:00") is None
        assert s._scheduler.get_job("plan:1@09:30") is not None

    # ⑥ 停用后不注册
    def test_disabled_registers_nothing(self):
        s = CollectScheduler("Asia/Shanghai")
        s.sync_plan(1, False, "time", ["09:30"], None)
        assert not [j for j in s._scheduler.get_jobs() if j.id.startswith("plan:1")]

    # ⑦ 旧数据兼容（灰度路径）
    def test_null_schedule_type_is_safe(self):
        s = CollectScheduler("Asia/Shanghai")
        s.sync_plan(1, True, None, [], None)
        assert not [j for j in s._scheduler.get_jobs() if j.id.startswith("plan:1")]

    # ⑧ 定时模式停用后 interval job 也要清掉（跨模式切换）
    def test_switch_from_time_to_interval_clears_time_jobs(self):
        s = CollectScheduler("Asia/Shanghai")
        s.sync_plan(1, True, "time", ["09:30"], None)
        s.sync_plan(1, True, "interval", [], 300)
        assert s._scheduler.get_job("plan:1@09:30") is None
        assert s._scheduler.get_job("plan:1") is not None
