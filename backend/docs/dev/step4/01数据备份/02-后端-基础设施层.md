## 3. path_resolver.py

```python
class PathResolver:
    """路径白名单 + 安全校验

    读 / 写路径必须落在 settings.BACKUP_ALLOWED_ROOTS 任一根目录下。
    """
    def __init__(self, allowed_roots: list[str]):
        self._allowed_roots = [Path(r).resolve() for r in allowed_roots]

    def resolve_write(self, raw: str | None) -> Path:
        """解析写入路径；raw 为空 → 用 settings.BACKUP_DIR 默认"""
        if not raw:
            return Path(settings.BACKUP_DIR).resolve()
        p = Path(raw).resolve()
        self._assert_within(p)
        return p

    def resolve_read(self, raw: str) -> Path:
        """解析读取路径"""
        p = Path(raw).resolve()
        self._assert_within(p)
        if not p.is_file():
            raise InvalidBackupFileError(f"文件不存在: {p}")
        return p

    def _assert_within(self, p: Path) -> None:
        for root in self._allowed_roots:
            try:
                p.relative_to(root)
                return
            except ValueError:
                continue
        raise PathSecurityError(f"路径不在白名单: {p}")
```

## 6. engine.py（合并 FileNamer 进来）

```python
import secrets
import re
from datetime import datetime, timezone

SAFE_NAME_RE = re.compile(r"[^A-Za-z0-9_\-]")

class FileNamer:
    """文件名生成（嵌入 engine）"""
    def make(self, backup_type: str, tables_count: int, user_name: str = "") -> str:
        ts = datetime.now().strftime("%Y%m%d_%H%M%S")
        suffix = secrets.token_hex(4)  # 8 字符防冲突
        prefix = (
            self._sanitize(user_name) if user_name
            else f"{backup_type}_{ts}_{tables_count}t"
        )
        return f"{prefix}_{suffix}.sql"

    def make_default_name(self, backup_type: str) -> str:
        return f"{backup_type}_{datetime.now().strftime('%Y%m%d_%H%M%S')}"

    @staticmethod
    def _sanitize(name: str) -> str:
        return SAFE_NAME_RE.sub("_", name)[:80]


class BackupEngine:
    """备份引擎（编排 schema + data + 写文件 + 进度上报）"""
    def __init__(
        self,
        *,
        schema_dumper: SchemaDumper,
        data_dumper: DataDumper,
        path_resolver: PathResolver,
        app_service: "BackupAppService",  # 进度回调走 service
    ):
        self.schema_dumper = schema_dumper
        self.data_dumper = data_dumper
        self.path_resolver = path_resolver
        self.app_service = app_service
        self.file_namer = FileNamer()

    async def run(self, record_id: int, req: "CreateBackupRequest") -> None:
        """主入口（被 AppService 用 asyncio.create_task 启动）"""
        # 1. 解析输出路径
        out_dir = self.path_resolver.resolve_write(req.output_dir)
        out_dir.mkdir(parents=True, exist_ok=True)

        # 2. 决定表
        all_tables = list(Base.metadata.tables.keys())
        tables = self._resolve_tables(req, all_tables)
        tables = [t for t in tables if t not in BACKUP_EXCLUDE_TABLES]

        # 3. 文件名
        file_name = self.file_namer.make(req.backup_type, len(tables))
        file_path = out_dir / file_name

        # 4. 写文件（流式）
        await self.app_service.update_progress(record_id, progress=5, msg="开始导出")

        async with aiofiles.open(file_path, "w", encoding="utf-8") as f:
            # 4.1 写头部（schema 头 + BEGIN）
            await f.write(self._render_header(req, tables))
            await self.app_service.update_progress(record_id, progress=10, msg="schema")

            # 4.2 写 schema
            if req.backup_type in ("full", "schema"):
                await self.schema_dumper.dump_tables(tables, f)

            # 4.3 写数据
            if req.backup_type in ("full", "data"):
                for i, table in enumerate(tables, 1):
                    await self.data_dumper.dump_table(table, f)
                    pct = 10 + int(80 * i / len(tables))
                    await self.app_service.update_progress(
                        record_id, progress=pct, msg=f"已导出 {table} ({i}/{len(tables)})"
                    )

            # 4.4 写 COMMIT
            await f.write("\nCOMMIT;\n")

        # 5. 完成
        await self.app_service.mark_success(
            record_id, file_path=file_path, file_size=file_path.stat().st_size
        )

    def _render_header(self, req, tables):
        """写头部注释 + SET 语句"""
        return f"""-- backup_name: {req.name or self.file_namer.make_default_name(req.backup_type)}
-- generator: attribution-analysis
-- generator_version: 1.0
-- backup_type: {req.backup_type}
-- scope: {req.scope}
-- tables_count: {len(tables)}
-- tables: {','.join(tables)}
-- created_at: {datetime.now(timezone.utc).isoformat()}
-- schema_version: {settings.BACKUP_SCHEMA_VERSION}
SET client_encoding = 'UTF8';
SET standard_conforming_strings = on;
SET session_replication_role = 'origin';
BEGIN;
"""

    def _resolve_tables(self, req, all_tables) -> list[str]:
        """根据 req.scope / req.tables 决定要备份的表"""
        if req.scope == "all":
            return all_tables
        return [t for t in all_tables if t in req.tables]
```

> **简化说明**：
> - 时钟直接 `datetime.now()`（**v1 不引入 ClockPort**）
> - 进度通过 `self.app_service.update_progress()` 直调（**v1 不引入 progress_cb 回调**）
> - FileNamer 内嵌到 engine（**v1 不独立**）

## 9. 关键决策小结

| 决策 | 选型 | 不选 |
|---|---|---|
| SQL 渲染 | SQLAlchemy Inspector 反射 | 手写 DDL |
| 数据导出 | cursor.copy_to() 流式 COPY | `INSERT VALUES` |
| 文件格式 | 纯 SQL 文本 | JSON / 二进制 |
| 时钟 | 直接 `datetime.now()` | ClockPort（v1 业务用不到）|
| 路径策略 | PathResolver 单类 | PathSecurityPort Protocol（v1 单实现）|
| 进度回调 | app_service.update_progress() | Callable[[Progress], Awaitable[None]] |
| 任务管理 | asyncio.create_task + dict 状态 | TaskManager 独立类 |
| 状态机 | 3 态 + is_failed 布尔 | 8 态枚举 |
| 异步驱动 | FastAPI lifespan + asyncio.create_task | 后台 worker 进程 |
| 任务恢复 | 启动时查"running 状态"→ 标记 failed | 真正恢复（v1 简化）|
