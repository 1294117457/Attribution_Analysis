## 3. api/backup.ts

```typescript
import request from "@/utils/request"

const BASE = "/api/v1/backups"

// 备份
export const createBackup = (data: CreateBackupRequest) =>
  request.post(BASE, data)
export const listBackups = (params) =>
  request.get(BASE, { params })
export const getBackup = (id: number) =>
  request.get(`${BASE}/${id}`)
export const deleteBackup = (id: number) =>
  request.delete(`${BASE}/${id}`)
export const downloadBackupUrl = (id: number) =>
  `${import.meta.env.VITE_API_BASE}${BASE}/${id}/download`

// 表元信息
export const listTables = () =>
  request.get(`${BASE}/_tables`)

// 配置
export const getBackupConfig = () =>
  request.get(`${BASE}/config`)
export const updateBackupConfig = (data: UpdateConfigRequest) =>
  request.put(`${BASE}/config`, data)

// 恢复
export const restore = (data: RestoreRequest) =>
  request.post(`${BASE}/restore`, data)
export const listRestores = (params) =>
  request.get(`${BASE}/restore`, { params })
export const getRestore = (id: number) =>
  request.get(`${BASE}/restore/${id}`)

// 上传
export const uploadBackupFile = (file: File) => {
  const fd = new FormData()
  fd.append("file", file)
  return request.post(`${BASE}/upload`, fd, {
    headers: { "Content-Type": "multipart/form-data" },
  })
}
```

## 4. stores/backup.ts

```typescript
import { defineStore } from "pinia"
import { ref } from "vue"

const PATH_KEY = "bkup-paths"

export const useBackupStore = defineStore("backup", () => {
  // ── 路径持久化（localStorage） ──
  const pathConfig = ref(loadPath())
  function loadPath() {
    try {
      return JSON.parse(localStorage.getItem(PATH_KEY) || "{}")
    } catch { return {} }
  }
  function setOutputDir(d: string) {
    pathConfig.value.outputDir = d
    localStorage.setItem(PATH_KEY, JSON.stringify(pathConfig.value))
  }
  function setInputDir(d: string) {
    pathConfig.value.inputDir = d
    localStorage.setItem(PATH_KEY, JSON.stringify(pathConfig.value))
  }

  // ── 轮询（按 record_id 缓存 Task） ──
  const pollers = new Map<number, number>()  // record_id → intervalId
  function startPolling(record_id: number, cb: (r: any) => void) {
    if (pollers.has(record_id)) return
    const tick = async () => {
      try {
        const r = await getBackup(record_id)
        cb(r)
        if (r.status === "success" || r.status === "failed") {
          stopPolling(record_id)
        }
      } catch {}
    }
    pollers.set(record_id, window.setInterval(tick, 2000))
  }
  function stopPolling(record_id: number) {
    const id = pollers.get(record_id)
    if (id) {
      clearInterval(id)
      pollers.delete(record_id)
    }
  }

  return { pathConfig, setOutputDir, setInputDir, startPolling, stopPolling }
})
```

## 5. composables/useBackupDensity.ts

> **直接照抄 page-development-guide §四** 的 JS 模板，改 DENSITY_KEY 和前缀 `bkup-`

```typescript
import { ref, computed, watch } from "vue"

const DENSITY_KEY = "bkup-density"
function clamp(v: number) {
  const n = Math.max(0.6, Math.min(1.5, Number.isFinite(v) ? v : 1))
  return Math.round(n * 100) / 100
}
function load() {
  try {
    const raw = localStorage.getItem(DENSITY_KEY)
    if (!raw) return { page: 1, top: 1, middle: 1, bottom: 1 }
    const p = JSON.parse(raw)
    return {
      page: clamp(Number(p.page ?? 1)),
      top: clamp(Number(p.top ?? 1)),
      middle: clamp(Number(p.middle ?? 1)),
      bottom: clamp(Number(p.bottom ?? 1)),
    }
  } catch { return { page: 1, top: 1, middle: 1, bottom: 1 } }
}

export function useBackupDensity() {
  const init = load()
  const pageDensity = ref(init.page)
  const topDensity = ref<number | null>(init.top === 1 ? null : init.top)
  const middleDensity = ref<number | null>(init.middle === 1 ? null : init.middle)
  const bottomDensity = ref<number | null>(init.bottom === 1 ? null : init.bottom)

  const densityStyle = computed(() => {
    const s: Record<string, string> = { "--bkup-page-density": String(pageDensity.value) }
    if (topDensity.value != null) s["--bkup-top-density"] = String(topDensity.value)
    if (middleDensity.value != null) s["--bkup-middle-density"] = String(middleDensity.value)
    if (bottomDensity.value != null) s["--bkup-bottom-density"] = String(bottomDensity.value)
    return s
  })

  // Proxy 滑块（拖动断开跟随）+ colW 函数
  const topProxy = computed({ get: () => topDensity.value ?? pageDensity.value, set: v => topDensity.value = v })
  const midProxy = computed({ get: () => middleDensity.value ?? pageDensity.value, set: v => middleDensity.value = v })
  const botProxy = computed({ get: () => bottomDensity.value ?? pageDensity.value, set: v => bottomDensity.value = v })
  const colW = (px: number) => Math.round(px * midProxy.value)

  const allSynced = computed(() =>
    topDensity.value === null && middleDensity.value === null && bottomDensity.value === null
  )
  function reset() { pageDensity.value = 1; topDensity.value = null; middleDensity.value = null; bottomDensity.value = null }
  function syncToPage() { topDensity.value = null; middleDensity.value = null; bottomDensity.value = null }

  watch([pageDensity, topDensity, middleDensity, bottomDensity], ([p, t, m, b]) => {
    try { localStorage.setItem(DENSITY_KEY, JSON.stringify({ page: p, top: t ?? 1, middle: m ?? 1, bottom: b ?? 1 })) } catch {}
  })

  return { pageDensity, topProxy, midProxy, botProxy, densityStyle, colW, allSynced, reset, syncToPage }
}
```

## 9. components/backup/BackupPathConfig.vue（路径配置弹窗）

```vue
<template>
  <el-dialog v-model="visible" title="路径配置" width="500">
    <el-form label-width="120px">
      <el-form-item label="备份输出根">
        <el-input v-model="outputDir" placeholder="默认 ./backups" />
      </el-form-item>
      <el-form-item label="恢复读取根">
        <el-input v-model="inputDir" placeholder="默认 ./backups" />
      </el-form-item>
      <el-form-item label="服务端白名单">
        <div class="bkup-roots-tip">
          来自服务端 config：<code>{{ allowedRoots.join(" | ") }}</code>
        </div>
      </el-form-item>
    </el-form>
    <template #footer>
      <el-button @click="visible = false">取消</el-button>
      <el-button type="primary" @click="onSave">保存</el-button>
    </template>
  </el-dialog>
</template>

<script setup lang="ts">
import { ref, watch } from "vue"
import { useBackupStore } from "@/stores/backup"
import { getBackupConfig, updateBackupConfig } from "@/api/backup"

const props = defineProps<{ modelValue: boolean }>()
const emit = defineEmits<{ "update:modelValue": [v: boolean] }>()
const visible = ref(props.modelValue)
watch(() => props.modelValue, v => visible.value = v)
watch(visible, v => emit("update:modelValue", v))

const store = useBackupStore()
const outputDir = ref(store.pathConfig.outputDir || "")
const inputDir = ref(store.pathConfig.inputDir || "")
const allowedRoots = ref<string[]>([])

watch(visible, async v => {
  if (v) {
    const c = await getBackupConfig()
    allowedRoots.value = c.allowed_roots
  }
})

async function onSave() {
  store.setOutputDir(outputDir.value)
  store.setInputDir(inputDir.value)
  await updateBackupConfig({ default_output_dir: outputDir.value })
  visible.value = false
}
</script>
```

## 11. 关键决策小结

| 决策 | 选型 | 不选 |
|---|---|---|
| 路由 | 真子路由 + 3 个真组件 | tab meta 区分同一 component |
| 密度复用 | composable + panel 组件 | 每页面复制粘贴 |
| 持久化 | localStorage + Pinia | 单独后端 API 存 |
| 轮询 | setInterval 2s | WebSocket |
| 上传 | 一次性 POST FormData | 流式分片 |
| 文件下载 | window.open URL | 后端 Blob 流 |
| 二次确认 | ElMessageBox | 自定义对话框 |
| 路径配置 | 弹窗 + localStorage | 独立设置页 |
| 进度 | el-progress | 自定义 |
| API 错误 | ElMessage.error（全局拦截器已有）| 自定义错误组件 |
