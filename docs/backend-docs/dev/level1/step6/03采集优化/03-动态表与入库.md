## 2. `resolve_table_schema`：采集时反查

```python
class ColumnDef(NamedTuple):
    name:        str
    data_type:   str   # INFORMATION_SCHEMA 的 DATA_TYPE
    is_nullable: bool

class TableSchema(NamedTuple):
    columns:     list[ColumnDef]
    primary_key: list[str]
    table_name:  str


async def resolve_table_schema(
    engine: AsyncEngine, table_name: str
) -> TableSchema:
    """采集时查 INFORMATION_SCHEMA，返回表结构（PostgreSQL）"""
    async with engine.begin() as conn:
        # 列信息
        cols = await conn.execute(text("""
            SELECT COLUMN_NAME, DATA_TYPE, IS_NULLABLE
            FROM INFORMATION_SCHEMA.COLUMNS
            WHERE TABLE_SCHEMA = current_schema()
              AND TABLE_NAME = :t
            ORDER BY ORDINAL_POSITION
        """), {"t": table_name})
        col_list = [
            ColumnDef(
                name=r.COLUMN_NAME,
                data_type=r.DATA_TYPE,
                is_nullable=(r.IS_NULLABLE == "YES"),
            )
            for r in cols
        ]
        if not col_list:
            raise PortSchemaMismatchError(
                port="?", table=table_name,
                reason=f"表 `{table_name}` 不存在或没有列",
            )

        # 主键
        pks = await conn.execute(text("""
            SELECT COLUMN_NAME
            FROM INFORMATION_SCHEMA.KEY_COLUMN_USAGE
            WHERE TABLE_SCHEMA = current_schema()
              AND TABLE_NAME = :t
              AND CONSTRAINT_NAME = 'PRIMARY'
            ORDER BY ORDINAL_POSITION
        """), {"t": table_name})
        pk_list = [r.COLUMN_NAME for r in pks]

    return TableSchema(columns=col_list, primary_key=pk_list, table_name=table_name)
```

**每次采集都会调用**，保证拿到最新的表结构。

> **双库下的 engine 选择**：`resolve_table_schema` 接收的 `engine` 必须是 **raw 库的 engine**（`raw_engine`），因为采集表都在 `raw_data_db` 里。
> 调用方：`DynamicRawDataStore` 注入的就是 `raw_engine`。

## 3. 字段映射校验

第三方（Tushare / Adata）返回的 dict 键名必须与表列名一致：

```python
def validate_row_fields(rows: list[dict], schema: TableSchema) -> None:
    """rows 的每个键必须在表列里；多余列忽略（SELECT 时不选），缺列报错"""
    table_cols = {c.name for c in schema.columns}
    for i, row in enumerate(rows):
        missing = set()
        for col in table_cols:
            if col not in row:
                missing.add(col)
        if missing:
            raise PortSchemaMismatchError(
                port="?", table=schema.table_name,
                reason=f"第 {i+1} 行缺少表里有的必填列: {sorted(missing)} "
                       f"（如果列确实不需要入库，在 SQL 中用视图过滤）",
            )
```

> **注意**：只校验"表有但数据没有"的列（可能漏传）。表没有但数据有的列在入库时 SQLAlchemy 会忽略（bindparam 只绑定 row 实际有的键）。

## 4. `upsert` 动态拼装（PostgreSQL 语法）

```python
def build_upsert_sql(
    table_name: str,
    schema: TableSchema,
    upsert_mode: str,
    row_keys: list[str],
) -> str:
    """根据表的主键动态拼 upsert SQL（PostgreSQL）"""
    col_list = ", ".join(f'"{c}"' for c in row_keys)
    placeholders = ", ".join(f":{c}" for c in row_keys)

    if upsert_mode == "insert_only":
        return f'INSERT INTO "{table_name}" ({col_list}) VALUES ({placeholders})'

    pk = schema.primary_key
    if not pk:
        raise PortSchemaMismatchError(
            port="?", table=table_name,
            reason=f"upsert_mode=by_unique 但表没有主键",
        )

    # PostgreSQL 用 ON CONFLICT (pk_cols) DO UPDATE SET col=EXCLUDED.col
    pk_cols = ", ".join(f'"{c}"' for c in pk)
    non_pk = [c for c in row_keys if c not in pk]

    if upsert_mode == "on_conflict_do_nothing":
        return (
            f'INSERT INTO "{table_name}" ({col_list}) VALUES ({placeholders}) '
            f'ON CONFLICT ({pk_cols}) DO NOTHING'
        )
    else:  # by_unique
        update_clause = ", ".join(f'"{c}" = EXCLUDED."{c}"' for c in non_pk)
        return (
            f'INSERT INTO "{table_name}" ({col_list}) VALUES ({placeholders}) '
            f'ON CONFLICT ({pk_cols}) DO UPDATE SET {update_clause}'
        )
```

> **MySQL → PostgreSQL 语法对照**：
>
> | 操作 | MySQL | PostgreSQL |
> |:--|:--|:--|
> | UPSERT | `INSERT ... ON DUPLICATE KEY UPDATE col=VALUES(col)` | `INSERT ... ON CONFLICT (pk) DO UPDATE SET col=EXCLUDED.col` |
> | IGNORE | `INSERT IGNORE INTO ...` | `INSERT ... ON CONFLICT (pk) DO NOTHING` |
> | 当前 schema | `DATABASE()` | `current_schema()` |
> | 标识符引号 | `` ` `` | `"` |

## 5. `DynamicRawDataStore` 完整实现

```python
class SaveResult(NamedTuple):
    written:  int   # 实际入库行数
    skipped:  int   # 跳过的行数（如 ON CONFLICT DO NOTHING 撞键）


class DynamicRawDataStore:
    """采集时选表：反查表结构 → 字段校验 → 试运行 → 正式入库"""

    def __init__(self, raw_engine: AsyncEngine, adapter: DataSourceAdapter):
        self._raw_engine = raw_engine   # ★ 双库：必须是 raw 库的 engine
        self._adapter    = adapter

    async def save(
        self,
        port_name: str,
        table_name: str,
        params: dict,
    ) -> SaveResult:
        # 1. 采集数据
        rows = await self._adapter.fetch(port_name, params)
        if not rows:
            return SaveResult(written=0, skipped=0)

        # 2. 反查表结构（用 raw_engine）
        schema = await resolve_table_schema(self._raw_engine, table_name)

        # 3. 字段校验
        validate_row_fields(rows, schema)

        # 4. 拼 upsert SQL
        upsert_mode = self._catalog[port_name].upsert_mode
        row_keys = list(rows[0].keys())
        sql = build_upsert_sql(table_name, schema, upsert_mode, row_keys)

        # 5. 试运行（只写 1 行）
        await self._execute(self._raw_engine, sql, [rows[0]])

        # 6. 正式入库（批量）
        written = await self._execute(self._raw_engine, sql, rows)
        return SaveResult(written=written, skipped=0)

    async def _execute(
        self, engine: AsyncEngine, sql: str, rows: list[dict]
    ) -> int:
        stmt = text(sql)
        async with engine.begin() as conn:
            await conn.execute(stmt, rows)
        return len(rows)

    async def existing(
        self,
        table_name: str,
        incremental_column: str,
        incremental_unit: str,
        units: list[str] | None,
    ) -> set[str]:
        """通用 existing：根据表结构查已入库的单元"""
        if not incremental_column:
            return set()
        async with self._raw_engine.begin() as conn:
            if units is None:
                rows = await conn.execute(text(
                    f'SELECT DISTINCT "{incremental_column}" FROM "{table_name}"'
                ))
            else:
                values = [
                    _parse_date(u) if incremental_unit == "trade_date" else u
                    for u in units
                ]
                rows = await conn.execute(text(
                    f'SELECT DISTINCT "{incremental_column}" FROM "{table_name}" '
                    f'WHERE "{incremental_column}" IN :values'
                ).bindparams(bindparam("values", expanding=True)),
                    {"values": values},
                )
        raw = [r[0] for r in rows]
        if incremental_unit == "trade_date":
            return {v.strftime("%Y%m%d") for v in raw}
        return set(raw)
```

## 6. 错误处理

| 场景 | 行为 |
| :-- | :-- |
| 表不存在（`resolve_table_schema` 抛错） | 采集任务 fail |
| 第三方返回列 ⊄ 表列 | `validate_row_fields` 抛错，采集任务 fail，**不写任何行** |
| 试运行失败（类型不匹配） | 抛错，采集任务 fail（已入库的 1 行保留；可接受，因为是同数据） |
| 正式入库失败 | 抛错，任务 fail；试运行行已在表里 |
| `upsert_mode=by_unique` 但表无主键 | 抛错，采集任务 fail |
| PostgreSQL 类型错误（如字符串写进 INT 列） | SQLAlchemy 报错 → 任务 fail |
| 跨库误用（如 `existing` 用了业务库 engine） | 查不到数据，返回空集，**任务 fail（无数据）** |

## 7. 读侧迁移

```python
# route/api/v1/data_board.py

async def query_raw_table(
    table_name: str,
    filters: dict[str, Any] | None = None,
    limit: int | None = None,
) -> list[dict]:
    """读 raw 库任意表（PostgreSQL）"""
    sql = f'SELECT * FROM "{table_name}"'
    params = {}
    if filters:
        wheres = []
        for i, (k, v) in enumerate(filters.items()):
            ph = f"p{i}"
            wheres.append(f'"{k}" = :{ph}')
            params[ph] = v
        sql += " WHERE " + " AND ".join(wheres)
    if limit:
        sql += f" LIMIT {int(limit)}"

    # ★ 用 raw_session，不是 AsyncSessionLocal
    async with RawSessionLocal() as session:
        rows = await session.execute(text(sql), params)
    return [dict(r._mapping) for r in rows]
```

表名规范（**5 个面短前缀**）：`tech_*` / `capital_*` / `fundamental_*` / `news_*` / `sentiment_*`。
**无 `raw_` 前缀**——库名 `raw_data_db` 已表达"这是 raw 库"。

跨库 join 详见 [`10-分库与表结构.md §7`](10-分库与表结构.md)。
