### 3.4 仓储实现（**只在 panel_compose_repository.py 末尾追加**）

```python
# backend/src/infrastructure/persistence/repositories/panel_compose_repository.py 末尾追加

async def list_realtime_rows(
    self,
    q=None, industry=None, market=None, exchange=None,
    is_hs=None, list_status="L", exclude_st=None,
    min_total_mv=None, with_pools=True, page=1, page_size=50,
) -> tuple[list, int]:
    """实时看板行：stock_infos ⨝ fin_daily_basics(最新一行)"""
    # 1) fin_daily_basics 最新一行子查询
    latest_fdb = (
        select(
            FinDailyBasic.symbol,
            func.max(FinDailyBasic.trade_date).label("max_date"),
        )
        .group_by(FinDailyBasic.symbol)
        .subquery()
    )

    # 2) 主查询
    stmt = (
        select(
            StockInfo.symbol, StockInfo.name, StockInfo.industry,
            StockInfo.market, StockInfo.exchange, StockInfo.is_hs,
            StockInfo.list_status,
            FinDailyBasic.close.label("latest_close"),
            FinDailyBasic.total_mv, FinDailyBasic.circ_mv,
            FinDailyBasic.pe_ttm, FinDailyBasic.pb,
            FinDailyBasic.turnover_rate_f,
        )
        .select_from(StockInfo)
        .outerjoin(latest_fdb, latest_fdb.c.symbol == StockInfo.symbol)
        .outerjoin(
            FinDailyBasic,
            and_(
                FinDailyBasic.symbol == latest_fdb.c.symbol,
                FinDailyBasic.trade_date == latest_fdb.c.max_date,
            ),
        )
        .where(StockInfo.list_status == (list_status or "L"))
    )
    if q:
        like = f"%{q}%"
        stmt = stmt.where(or_(
            StockInfo.symbol.ilike(like),
            StockInfo.name.ilike(like),
            StockInfo.cnspell.ilike(like),
        ))
    if industry:    stmt = stmt.where(StockInfo.industry == industry)
    if market:      stmt = stmt.where(StockInfo.market == market)
    if exchange:    stmt = stmt.where(StockInfo.exchange == exchange)
    if is_hs:       stmt = stmt.where(StockInfo.is_hs == is_hs)
    if exclude_st:  stmt = stmt.where(~StockInfo.name.ilike("%ST%"))
    if min_total_mv: stmt = stmt.where(FinDailyBasic.total_mv >= min_total_mv * 10000)

    # 3) 分页
    total = await self._session.scalar(
        select(func.count()).select_from(stmt.subquery())
    )
    stmt = stmt.offset((page - 1) * page_size).limit(page_size)
    result = await self._session.execute(stmt)
    rows = result.mappings().all()

    # 4) 转 dataclass
    out = [
        StockRealtimeRow(
            symbol=r["symbol"], name=r["name"], industry=r["industry"],
            market=r["market"], exchange=r["exchange"], is_hs=r["is_hs"],
            list_status=r["list_status"],
            latest_close=r["latest_close"], total_mv=r["total_mv"],
            circ_mv=r["circ_mv"], pe_ttm=r["pe_ttm"], pb=r["pb"],
            turnover_rate_f=r["turnover_rate_f"],
        )
        for r in rows
    ]
    return out, (total or 0)
```

### 3.5 应用服务扩展（**只在 panel_app_service.py 末尾追加**）

```python
# backend/src/application/service/panel_app_service.py 末尾追加

class StockPanelAppService:
    # ... 现有方法不动 ...

    async def query_realtime(
        self, req: StockRealtimeQueryRequest
    ) -> StockRealtimeListVO:
        """实时看板：分页 + 池反查 + 实时行情补齐
        性能：
        - 1 DB（分页 + 估值）
        - 1 DB（池反查，with_pools=True）
        - 1 实时接口 query_many（page_size≤500 时一次性）
        """
        rows, total = await self._repo.list_realtime_rows(
            q=req.q, industry=req.industry, market=req.market,
            exchange=req.exchange, is_hs=req.is_hs, list_status=req.list_status,
            exclude_st=req.exclude_st, min_total_mv=req.min_total_mv,
            with_pools=req.with_pools, page=req.page, page_size=req.page_size,
        )

        # 池信息（避免 N+1）
        symbols = [r.symbol for r in rows]
        pool_map: dict[str, list[PoolMembershipVO]] = {}
        if req.with_pools and symbols:
            pool_map = await self._repo.list_membership_by_symbols(symbols)

        # 实时行情批量补齐（使用 stock_minute_kline 实时接口）
        # 注：每只调一次实时接口，最多 500 只，命中 Redis 15 秒缓存
        realtime_map: dict[str, dict] = {}
        if self._realtime and symbols:
            results = await self._realtime.query_many(
                "stock_minute_kline",
                [{"symbol": s, "interval": "1d"} for s in symbols],
            )
            for s, res in zip(symbols, results):
                if res and res.data and not res.stale:
                    realtime_map[s] = _parse_realtime(res.data, prev_close_lookup.get(s))

        # 组装 VO
        items = [
            _to_realtime_vo(
                r, pool_map.get(r.symbol, []), realtime_map.get(r.symbol, {})
            )
            for r in rows
        ]
        return StockRealtimeListVO.from_list(items, total, req.page, req.page_size)


def _parse_realtime(data: dict, prev_close: Optional[float]) -> dict:
    """解析 stock_minute_kline 实时接口响应 → 看板字段"""
    price = data.get("close") or data.get("price")
    open_ = data.get("open")
    high = data.get("high")
    low = data.get("low")
    vol = data.get("volume")
    amt = data.get("amount")
    pct = data.get("pct_change")
    change = data.get("change")

    amp = None
    if high and low and prev_close and prev_close > 0:
        amp = round((high - low) / prev_close * 100, 2)

    return {
        "price": price,
        "open": open_, "high": high, "low": low,
        "volume": vol, "amount": amt,
        "change": change, "pct_change": pct,
        "amplitude": amp,
        "color": "up" if (pct or 0) > 0 else "down" if (pct or 0) < 0 else "flat",
        "trade_time": data.get("trade_time"),
        "captured_at": data.get("captured_at") or data.get("fetched_at"),
    }


def _to_realtime_vo(r, pools, rt) -> StockRealtimeItemVO:
    return StockRealtimeItemVO(
        symbol=r.symbol, name=r.name, industry=r.industry,
        market=r.market, exchange=r.exchange, is_hs=r.is_hs,
        list_status=r.list_status,
        latest_close=r.latest_close, total_mv=r.total_mv,
        circ_mv=r.circ_mv, pe_ttm=r.pe_ttm, pb=r.pb,
        turnover_rate_f=r.turnover_rate_f,
        pools=pools, **rt,
    )
```

### 4.4 `frontend/src/composables/useRealtimeBoard.ts`

```ts
import { ref, onUnmounted } from 'vue'
import { useRealtimePoll } from './useRealtimePoll'
import { getStockRealtime, type StockRealtimeItem } from '@/views/realtime-board/api'

interface Opts {
  pageSize?: number
  intervalMs?: number
  // ... 透传筛选条件
}

/** 实时看板数据 composable：分页 + 池 + 实时行情，每 15s 轮询 */
export function useRealtimeBoard(opts: Opts = {}) {
  const items = ref<StockRealtimeItem[]>([])
  const total = ref(0)
  const loading = ref(false)
  const page = ref(1)
  const pageSize = ref(opts.pageSize ?? 100)
  const filters = ref<Record<string, any>>({})

  async function load() {
    loading.value = true
    try {
      const res = await getStockRealtime({
        ...filters.value,
        with_pools: true,
        page: page.value,
        page_size: pageSize.value,
      })
      items.value = res.items
      total.value = res.total
    } finally {
      loading.value = false
    }
  }

  // 复用 useRealtimePoll（自动判断交易时段 + 页面可见）
  const { start, stop, isTrading } = useRealtimePoll(load, { intervalMs: opts.intervalMs ?? 15000 })

  function setFilter(key: string, value: any) {
    filters.value = { ...filters.value, [key]: value }
    page.value = 1
    load()
  }
  function resetFilter() {
    filters.value = {}
    page.value = 1
    load()
  }

  onUnmounted(() => stop())

  return {
    items, total, loading, page, pageSize, filters,
    load, setFilter, resetFilter,
    start, stop, isTrading,
  }
}
```

## 5. 性能与边界

| 维度 | 数值 | 备注 |
|---|---|---|
| 首次加载 | ~800ms | 1 DB + 1 池反查 + N 实时（命中 Redis 缓存 < 200ms） |
| 15s 轮询 | ~300ms | 大部分命中缓存；缓存失效时穿透到 pytdx |
| 单页最大 | 500 只 | 与 `page_size=500` 限制一致 |
| 实时源 | pytdx（通达信） | 北交所代码不支持（与现有 `stock_minute_kline` 一致） |
| 弱网降级 | `stale=true` | 显示"行情源暂不可用" |
