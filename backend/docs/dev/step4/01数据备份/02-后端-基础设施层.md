### 关于"时钟"的 DDD 边界

> **DDD §2.2 明确要求：ApplicationService 管事务/权限/日志/缓存/幂等/时钟等"基础设施"；基础设施层不应直接调 `datetime.now()`**。

本模块在以下场景需要时间戳：

| 场景 | 当前方案 | DDD 合规修正 |
|---|---|---|
| 备份文件头部 `created_at` | `datetime.now(timezone.utc).isoformat()` | ❌ 直接调时钟 |
| 备份文件名时间戳 | `datetime.now().strftime(...)` | ❌ 直接调时钟 |
| 任务状态回写 `started_at / finished_at` | `NOW()` (SQL `NOW()`) | ✅ SQL 端，不算应用层时钟 |

**修正方案**（v1 落地 + v2 严格化）：

```python
# application/port/clock_port.py
from typing import Protocol
from datetime import datetime

class ClockPort(Protocol):
    """时钟端口 — 基础设施层提供实现"""
    def now(self) -> datetime: ...
    def now_iso(self) -> str: ...

# infrastructure/adapter/clock/system_clock.py
class SystemClockAdapter:
    def now(self) -> datetime:
        return datetime.now(timezone.utc)
    def now_iso(self) -> str:
        return self.now().isoformat()

# engine.py 中通过构造函数注入（不直接 import）
class BackupEngine:
    def __init__(self, ..., clock: ClockPort):
        self.clock = clock
    def _render_header(self, ...):
        return f"...\n-- created_at: {self.clock.now_iso()}\n..."
```

**v1 简化**：在 BackupAppService 里统一调 `clock.now()`，再传给引擎；**避免引擎自己 import `datetime`**。

## 2. 核心数据契约

### 2.1 BackupRequest

```python
from dataclasses import dataclass
from typing import Literal

@dataclass(frozen=True)
class BackupRequest:
    backup_type: Literal["full", "schema", "data"]
    scope: Literal["all", "partial"]
    tables: list[str]                       # scope=partial 时必填；scope=all 时填空
    output_dir: str | None                  # None = 用 ENV / 默认
    name: str                               # 用户给的备份名（可空，引擎补默认）

    def validate(self) -> None:
        if self.backup_type not in ("full", "schema", "data"):
            raise ValueError("backup_type 必须是 full/schema/data")
        if self.scope == "partial" and not self.tables:
            raise ValueError("partial 模式必须指定 tables")
        if self.scope == "all":
            object.__setattr__(self, "tables", [])  # 实际表名在引擎里查
```

## 3. `engine.py` — 备份引擎

### 3.1 主流程

```python
class BackupEngine:
    def __init__(
        self,
        *,
        schema_dumper: SchemaDumper,
        data_dumper: DataDumper,
        path_resolver: PathResolver,
        file_naming: FileNamer,
        clock: ClockPort,            # ← DDD §2.2：时钟通过 Port 注入
    ):
        self.clock = clock

    async def run(
        self,
        record_id: int,
        req: BackupRequest,
        progress_cb: Callable[[BackupProgress], Awaitable[None]],
    ) -> None:
        """主入口；被 task_manager 包成 asyncio.create_task"""
        # 1. 解析输出目录（安全校验）
        out_dir = self.path_resolver.resolve_write(req.output_dir)
        out_dir.mkdir(parents=True, exist_ok=True)

        # 2. 决定要备份的表
        all_tables = list(Base.metadata.tables.keys())
        backup_tables = self._resolve_tables(req, all_tables)
        # 过滤掉 BACKUP_EXCLUDE_TABLES
        backup_tables = [t for t in backup_tables if t not in BACKUP_EXCLUDE_TABLES]

        # 3. 生成文件名（时钟走 ClockPort，DDD §2.2）
        file_name = self.file_naming.make(req.backup_type, len(backup_tables))
        file_path = out_dir / file_name

        # 4. 渲染并写文件（流式）
        file_size = 0
        row_count = 0
        async with aiofiles.open(file_path, "w", encoding="utf-8") as f:
            # 4.1 写头部
            await f.write(self._render_header(req, backup_tables))

            # 4.2 写 schema（如果是 full / schema）
            if req.backup_type in ("full", "schema"):
                for i, table_name in enumerate(backup_tables, 1):
                    await progress_cb(BackupProgress(phase="schema", current_table=table_name, total_tables=len(backup_tables), done_tables=i, pct=int(i / len(backup_tables) * 30)))  # schema 阶段占 0-30%
                    table_sql = self.schema_dumper.dump(table_name)
                    await f.write(f"\n-- Table: {table_name}\n{table_sql}\n")

            # 4.3 写 data（如果是 full / data）
            if req.backup_type in ("full", "data"):
                for i, table_name in enumerate(backup_tables, 1):
                    await progress_cb(BackupProgress(phase="data", current_table=table_name, total_tables=len(backup_tables), done_tables=i, pct=30 + int(i / len(backup_tables) * 70)))  # data 阶段占 30-100%
                    rows, sql_chunk = await self.data_dumper.dump(table_name, mode="cover" if req.backup_type == "full" else "data")
                    await f.write(sql_chunk)
                    row_count += rows

            await f.write("\nCOMMIT;\n")

        file_size = file_path.stat().st_size

        # 5. 写完成态（由 task_manager 负责回写 DB；这里只返回）
        return BackupResult(file_path=str(file_path), file_size=file_size, row_count=row_count)
```

```python
BEGIN;
"""
```

## 4. `schema_dumper.py` — 结构渲染

### 4.1 实现思路

用 SQLAlchemy 的 `CreateTable` / `DropTable` / `CreateIndex` + PostgreSQL dialect 编译成 SQL。

```python
from sqlalchemy.schema import CreateTable, DropTable, CreateIndex
from sqlalchemy.dialects import postgresql

class SchemaDumper:
    def __init__(self):
        self.dialect = postgresql.dialect()

    def dump(self, table_name: str) -> str:
        table = Base.metadata.tables[table_name]
        parts: list[str] = []

        # 1. DROP
        drop_sql = str(DropTable(table, if_exists=True).compile(
            dialect=self.dialect,
            statement_cache={},
        ))
        parts.append(f"{drop_sql};")

        # 2. CREATE TABLE
        create_sql = str(CreateTable(table).compile(
            dialect=self.dialect,
            statement_cache={},
        ))
        parts.append(f"{create_sql};")

        # 3. CREATE INDEX（关键！SQLAlchemy 的 CreateTable 不会输出 Index）
        for idx in table.indexes:
            idx_sql = str(CreateIndex(idx).compile(dialect=self.dialect, statement_cache={}))
            parts.append(f"{idx_sql};")

        return "\n".join(parts)
```

### 4.2 已知问题与对策

| 问题 | 影响 | 对策 |
|---|---|---|
| `JSONB` / `ARRAY` 编译 | 实际可用 | 测过 OK |
| `BigSerial` 自增序列 | SQLAlchemy 会输出 `BIGSERIAL` | OK |
| `Sequence` 对象（显式序列）| 部分场景不输出 | 项目内 BigSerial 够用；显式 Sequence 不在 v1 表里 |
| `CheckConstraint` | 编译有 bug | 测过 OK；遇到再补 |
| `ForeignKey` 跨 schema | 单 schema 不影响 | OK |
| `Index` 名字带 schema | v1 不跨 schema | OK |
| `Enum` 类型 | 编译可能输出 `CREATE TYPE` | v1 表里没用 Enum；遇到再加 |

### 4.3 调试辅助

```python
def dump_to_file(self, table_name: str, out: Path) -> None:
    """调试用：把单表 dump 写到文件"""
    out.write_text(self.dump(table_name), encoding="utf-8")
```

## 5. `data_dumper.py` — 数据导出

### 5.1 设计：COPY 协议（核心性能关键）

PostgreSQL `COPY FROM stdin` 协议：客户端把 tab 分隔的数据流发给 PG server，**比 `INSERT` 快 10-20 倍**。

```python
import io
import csv
from sqlalchemy import text, select, Table

class DataDumper:
    """对单表导出数据为 COPY 块"""

    CHUNK_SIZE = 5000  # 每批 5000 行

    def __init__(self, engine: AsyncEngine):
        self.engine = engine

    async def dump(self, table_name: str, mode: Literal["cover", "data"] = "data") -> tuple[int, str]:
        """返回 (行数, COPY 块 SQL 字符串)"""
        table = Base.metadata.tables[table_name]
        columns = [c.name for c in table.columns]

        total_rows = 0
        chunks: list[str] = []

        async with self.engine.connect() as conn:
            # 设置快照隔离（保证一致性视图）
            await conn.execute(text("SET TRANSACTION ISOLATION LEVEL REPEATABLE READ"))

            # 流式查询
            result = await conn.stream(select(table))

            buf = io.StringIO()
            writer = csv.writer(buf, delimiter="\t", lineterminator="\n", quoting=csv.QUOTE_NONE, escapechar="\\")
            for row in result:
                writer.writerow([_to_copy_value(v) for v in row.values()])
                total_rows += 1
                if total_rows % self.CHUNK_SIZE == 0:
                    chunks.append(self._build_copy_block(table_name, columns, buf.getvalue()))
                    buf = io.StringIO()
                    writer = csv.writer(buf, delimiter="\t", lineterminator="\n", quoting=csv.QUOTE_NONE, escapechar="\\")
            if buf.tell() > 0:
                chunks.append(self._build_copy_block(table_name, columns, buf.getvalue()))

        return total_rows, "\n".join(chunks)

    def _build_copy_block(self, table_name: str, columns: list[str], tsv: str) -> str:
        cols_csv = ", ".join(columns)
        return f"\n-- Table data: {table_name}\nCOPY {table_name} ({cols_csv}) FROM stdin;\n{tsv}\n\\.\n"

def _to_copy_value(v: Any) -> str:
    """转换值：None → \N; datetime → ISO; bytes → hex; 其他 str()"""
    if v is None:
        return "\\N"
    if isinstance(v, (datetime, date)):
        return v.isoformat()
    if isinstance(v, bool):
        return "t" if v else "f"
    if isinstance(v, (dict, list)):
        return json.dumps(v, ensure_ascii=False)
    return str(v).replace("\\", "\\\\").replace("\t", "\\t").replace("\n", "\\n").replace("\r", "\\r")
```

### 5.2 特殊值处理

| 值类型 | COPY 协议 | 实现 |
|---|---|---|
| `NULL` | `\N` | `v is None → "\\N"` |
| `True/False` | `t/f` | `isinstance(v, bool)` |
| `\t` / `\n` | `\\t` / `\\n` | `str.replace` |
| `\` | `\\` | `str.replace` |
| `dict/list` | JSON 字符串 | `json.dumps(ensure_ascii=False)` |
| `datetime` | ISO 8601 | `v.isoformat()` |
| `bytes` | hex `\\x...` | 简化：v1 转 base64（业务里少）|

### 5.3 大表分块

## 6. `restorer.py` — 恢复引擎

> **DDD 合规说明**：本类在 `infrastructure/adapter/backup/`，属于"外部 SQL 适配"（绕开 ORM 直接执行备份文件里的 SQL）。这与领域层无关——是**外部命令的忠实执行**，不是业务规则。**不放在 domain/，不放在 application/**，放 adapter 合理。
>
> 任何"恢复模式的业务规则"（cover/upsert 选择、自动 pre-backup、确认）**都在 application 层做**；Restorer 只接受 `confirm=True` 后的纯执行。

### 6.1 核心原则

1. **单事务**：DROP + CREATE + COPY 全部在一个事务内，失败整体回滚。
2. **强校验头部**：必须匹配 `generator=attribution-analysis`，否则拒。
3. **cover vs upsert 区别**：
   - `cover` → 走 `engine.py` 生成的 SQL（已含 `DROP TABLE IF EXISTS ... CASCADE`），直接 exec 整段 SQL
   - `upsert` → 自己解析，**跳过 DROP 和 CREATE**（保留现有 schema），只把 `INSERT VALUES` 改成 `INSERT ... ON CONFLICT DO NOTHING`

### 6.2 主流程（cover 模式）

```python
class Restorer:
    def __init__(self, engine: AsyncEngine, path_resolver: PathResolver, backup_engine: BackupEngine):
        self.engine = engine
        self.path_resolver = path_resolver
        self.backup_engine = backup_engine

    async def run(
        self,
        record_id: int,
        req: RestoreRequest,
        progress_cb: Callable[[RestoreProgress], Awaitable[None]],
    ) -> RestoreResult:
        if not req.confirm:
            raise ValueError("恢复必须 confirm=True")

        # 1. 解析 + 校验源文件
        abs_path = self.path_resolver.resolve_read(req.source_path)
        if not abs_path.exists():
            raise FileNotFoundError(f"备份文件不存在: {abs_path}")
        sql_text = abs_path.read_text(encoding="utf-8")
        header = parse_header(sql_text)
        _validate_header(header)  # 强校验

        # 2. cover 模式：自动做 pre-restore 备份（v1 同步等它完成；v2 可后台）
        pre_backup_id: int | None = None
        if req.restore_mode == "cover" and req.auto_pre_backup:
            pre_backup_id = await self._auto_pre_backup(record_id, abs_path)

        # 3. 执行恢复（单事务）
        rows_inserted = 0
        try:
            if req.restore_mode == "cover":
                rows_inserted = await self._restore_cover(sql_text, progress_cb)
            else:
                rows_inserted, rows_skipped = await self._restore_upsert(sql_text, header, progress_cb)
        except Exception as e:
            # 事务已回滚；状态在 service 层标 failed
            raise

        return RestoreResult(rows_inserted=rows_inserted, pre_backup_id=pre_backup_id, header=header)

    async def _restore_cover(self, sql_text: str, progress_cb) -> int:
        """cover 模式：直接 exec 整段 SQL（事务内）"""
        # 拆成单个语句，逐条 exec（避免 PG 拒超大单语句）
        statements = _split_sql_statements(sql_text)
        total = len(statements)
        rows_inserted = 0
        async with self.engine.begin() as conn:
            for i, stmt in enumerate(statements, 1):
                stmt = stmt.strip()
                if not stmt or stmt.startswith("--"):
                    continue
                try:
                    result = await conn.execute(text(stmt))
                    if result.rowcount and result.rowcount > 0:
                        rows_inserted += result.rowcount
                except Exception as e:
                    # 关键：把语句和错误一起抛，便于排查
                    raise RestoreError(f"第 {i}/{total} 条语句失败:\n{stmt[:200]}\n{type(e).__name__}: {e}") from e
                await progress_cb(RestoreProgress(pct=int(i / total * 100), current=f"exec {i}/{total}"))
        return rows_inserted
```

### 6.4 `_split_sql_statements`

```python
def _split_sql_statements(sql_text: str) -> list[str]:
    """按分号切，但保留 COPY ... \\. 整块（不被分号切碎）"""
    statements = []
    buf = []
    in_copy = False
    for line in sql_text.splitlines():
        if line.strip().upper().startswith("COPY ") and "FROM STDIN" in line.upper():
            in_copy = True
        buf.append(line)
        if in_copy and line.rstrip().endswith("\\."):
            statements.append("\n".join(buf))
            buf = []
            in_copy = False
        elif not in_copy and line.rstrip().endswith(";"):
            statements.append("\n".join(buf))
            buf = []
    if buf:
        statements.append("\n".join(buf))
    return [s for s in statements if s.strip() and not s.strip().startswith("--")]
```

## 7. `path_resolver.py` — 路径安全

### 7.1 白名单

```python
DEFAULT_ALLOWED_ROOTS = [
    Path("/data/backups"),                                  # Linux 服务器
    Path(__file__).resolve().parent.parent.parent / "backups",  # backend/backups/
]

class PathResolver:
    """单例：从 settings 读白名单 + ENV 追加"""

    def __init__(self, settings):
        self.allowed_roots: list[Path] = list(DEFAULT_ALLOWED_ROOTS)
        extra = settings.BACKUP_ALLOWED_ROOTS  # ENV: "D:/extra1,D:/extra2"
        if extra:
            self.allowed_roots.extend(Path(p.strip()) for p in extra.split(",") if p.strip())

    def resolve_read(self, raw: str | None) -> Path:
        """解析读路径（用户给历史备份的绝对路径）"""
        if not raw:
            raise PathSecurityError("路径不能为空")
        p = Path(raw).resolve()
        self._check(p, mode="read")
        return p

    def resolve_write(self, raw: str | None) -> Path:
        """解析写路径（用户给输出目录）"""
        if not raw:
            # 用 ENV 兜底
            default = os.getenv("BACKUP_DIR", str(DEFAULT_ALLOWED_ROOTS[-1]))
            raw = default
        p = Path(raw).resolve()
        self._check(p, mode="write")
        return p

    def _check(self, p: Path, mode: str) -> None:
        # 1. 必须在白名单根下
        ok = any(self._is_under(p, root) for root in self.allowed_roots)
        if not ok:
            raise PathSecurityError(f"路径不在白名单: {p}；允许: {[str(r) for r in self.allowed_roots]}")
        # 2. 写模式额外检查
        if mode == "write":
            forbidden = ("/src/", "/migrations/", "/.git/", "/.keys/", "/.env")
            path_str = str(p).replace("\\", "/")
            for f in forbidden:
                if f in path_str:
                    raise PathSecurityError(f"禁止写入: {f}")
        # 3. 禁止符号链接穿透
        if p.is_symlink():
            raise PathSecurityError(f"符号链接禁止: {p}")

    @staticmethod
    def _is_under(p: Path, root: Path) -> bool:
        try:
            p.relative_to(root)
            return True
        except ValueError:
            return False
```

### 7.2 配置入口

```python
# infrastructure/config/settings.py（追加）
```

## 8. `task_manager.py` — 异步任务

### 8.1 设计

```python
class BackupTaskManager:
    """单例；管理 in-memory 任务表 + 持久化到 DB"""

    def __init__(self, engine: AsyncEngine, backup_engine: BackupEngine, restore_engine: Restorer):
        self.engine = engine
        self.backup_engine = backup_engine
        self.restore_engine = restore_engine
        self._tasks: dict[int, asyncio.Task] = {}  # record_id → Task

    async def submit_backup(self, record_id: int, req: BackupRequest) -> None:
        """提交备份任务；非阻塞"""
        if record_id in self._tasks:
            raise TaskAlreadyRunningError(f"备份 #{record_id} 已在执行")
        task = asyncio.create_task(self._run_backup(record_id, req))
        self._tasks[record_id] = task
        task.add_done_callback(lambda t: self._tasks.pop(record_id, None))

    async def submit_restore(self, record_id: int, req: RestoreRequest) -> None:
        """提交恢复任务"""
        ...

    async def _run_backup(self, record_id: int, req: BackupRequest) -> None:
        """实际跑备份；更新 DB 状态"""
        async with self.engine.begin() as conn:
            await conn.execute(text(
                "UPDATE sys_backup_records SET status='running', started_at=NOW(), "
                "progress=0, progress_msg='初始化' WHERE id=:id"
            ), {"id": record_id})

        async def cb(progress: BackupProgress):
            async with self.engine.begin() as conn:
                await conn.execute(text(
                    "UPDATE sys_backup_records SET progress=:pct, progress_msg=:msg "
                    "WHERE id=:id"
                ), {
                    "id": record_id,
                    "pct": progress.pct,
                    "msg": f"{progress.phase}: {progress.current_table} ({progress.done_tables}/{progress.total_tables})",
                })

        try:
            result = await self.backup_engine.run(record_id, req, cb)
            async with self.engine.begin() as conn:
                await conn.execute(text(
                    "UPDATE sys_backup_records SET status='success', progress=100, "
                    "progress_msg='完成', file_path=:fp, file_size=:fs, row_count=:rc, "
                    "finished_at=NOW() WHERE id=:id"
                ), {
                    "id": record_id, "fp": result.file_path,
                    "fs": result.file_size, "rc": result.row_count,
                })
        except Exception as e:
            logging.exception("备份任务 #%d 失败", record_id)
            async with self.engine.begin() as conn:
                await conn.execute(text(
                    "UPDATE sys_backup_records SET status='failed', "
                    "finished_at=NOW(), error_message=:err WHERE id=:id"
                ), {"id": record_id, "err": f"{type(e).__name__}: {e}"[:1000]})
```

## 9. `file_naming.py`

```python
import re
import secrets

SAFE_NAME_RE = re.compile(r"[^A-Za-z0-9_\-]")

class FileNamer:
    """生成唯一文件名 + 校验用户提供的名字

    时钟通过 ClockPort 注入（DDD §2.2 边界）"""
    def __init__(self, clock: ClockPort):
        self.clock = clock

    def make(self, backup_type: str, tables_count: int, user_name: str = "") -> str:
        ts = self.clock.now().strftime("%Y%m%d_%H%M%S")
        suffix = secrets.token_hex(4)  # 8 字符随机
        prefix = self._sanitize(user_name) if user_name else f"{backup_type}_{ts}_{tables_count}t"
        return f"{prefix}_{suffix}.sql"

    def make_default_name(self, backup_type: str) -> str:
        ts = self.clock.now().strftime("%Y%m%d_%H%M%S")
        return f"{backup_type}_{ts}"

    @staticmethod
    def _sanitize(name: str) -> str:
        return SAFE_NAME_RE.sub("_", name)[:80]
```

## 12. 边界与不做

| 不做 | 原因 |
|---|---|
| 不调 `pg_dump.exe` | Windows 路径坑 + 跨版本兼容 |
| 不实现压缩 | 交给 OS（gzip / 7z）|
| 不实现加密 | 交给 OS（EFS / BitLocker）|
| 不实现物理备份 | 规模不需要 |
| 不实现增量/差异备份 | 复杂度高，业务不需要 |
| 不实现远程备份（OSS / S3）| v2 再做 |
| 不实现定时备份 | v2 再做 |
| 不实现 `pg_dump --schema=public --schema=sys` 分 schema | 业务不跨 schema |
