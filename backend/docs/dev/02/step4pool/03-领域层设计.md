# 操作池 - 领域层设计

> 本文档定义"操作池"模块的领域层设计：聚合根、值对象、仓储接口、领域事件。
>
> **设计原则**：领域层**不依赖**任何外部框架（SQLAlchemy、FastAPI、Pydantic），仅使用 Python 标准库。

```python
from dataclasses import dataclass
from domain.base import ValueObject


@dataclass(frozen=True)
class PoolType(ValueObject):
    """池类型值对象（不可变）"""

    code: str
    label: str

    def _get_values(self) -> tuple:
        return (self.code,)

    # ── 预设类型 ─────────────────────────────────────────

    WATCHLIST = None  # 懒加载（类属性）
    INDUSTRY  = None
    STRATEGY  = None
    CUSTOM    = None

    @classmethod
    def _init_constants(cls):
        cls.WATCHLIST = cls("watchlist", "自选股")
        cls.INDUSTRY  = cls("industry",  "行业")
        cls.STRATEGY  = cls("strategy",  "策略")
        cls.CUSTOM    = cls("custom",    "自定义")

    @classmethod
    def from_code(cls, code: str) -> "PoolType":
        """从代码创建值对象（未识别代码返回 CUSTOM）"""
        mapping = {
            "watchlist": cls.WATCHLIST,
            "industry":  cls.INDUSTRY,
            "strategy":  cls.STRATEGY,
            "custom":    cls.CUSTOM,
        }
        return mapping.get(code.lower(), cls.CUSTOM)

    @classmethod
    def all(cls) -> tuple["PoolType", ...]:
        return (cls.WATCHLIST, cls.INDUSTRY, cls.STRATEGY, cls.CUSTOM)

    @property
    def is_system(self) -> bool:
        """系统级类型（不可删除其语义）"""
        return self.code in ("watchlist", "industry", "strategy")


# 初始化常量（模块加载时）
PoolType._init_constants()
```

```python
class PoolName(ValueObject):
    """池名称值对象（不可变，验证合法性）"""

    name: str

    MIN_LENGTH = 1
    MAX_LENGTH = 64

    def _get_values(self) -> tuple:
        return (self.name,)

    def __post_init__(self):
        if not self.name or not self.name.strip():
            raise ValueError("池名称不能为空")
        if len(self.name) > self.MAX_LENGTH:
            raise ValueError(f"池名称不能超过 {self.MAX_LENGTH} 字符")
        # 去除首尾空白
        object.__setattr__(self, 'name', self.name.strip())

    @classmethod
    def from_raw(cls, name: str) -> "PoolName":
        return cls(name)

    @property
    def display(self) -> str:
        return self.name
```

```python
import re


@dataclass(frozen=True)
class PoolColor(ValueObject):
    """池展示颜色值对象（不可变）"""

    hex_code: str  # 如 #FF5722

    HEX_PATTERN = re.compile(r"^#[0-9A-Fa-f]{6}$")

    def _get_values(self) -> tuple:
        return (self.hex_code,)

    def __post_init__(self):
        if self.hex_code and not self.HEX_PATTERN.match(self.hex_code):
            raise ValueError(f"无效的 HEX 颜色码: {self.hex_code}")

    @classmethod
    def from_hex(cls, hex_code: str | None) -> "PoolColor | None":
        if not hex_code:
            return None
        return cls(hex_code.upper())

    @property
    def css_value(self) -> str:
        return self.hex_code or "#1890ff"  # Element Plus 默认蓝
```

## 3. 聚合根（Aggregate Root）

### 3.1 `StockPool` — 操作池聚合根

```python
# domain/stock_pool/entity.py

from __future__ import annotations

from dataclasses import dataclass, field
from datetime import datetime
from typing import Optional
import uuid

from domain.base import AggregateRoot
from domain.stock_pool.value_objects import (
    PoolType,
    PoolName,
    PoolColor,
    PoolMember,
    OperationStatus,
    OperationType,
)


@dataclass
class StockPool(AggregateRoot):
    """操作池聚合根

    业务规则：
    1. 默认池不可删除
    2. 成员不能重复（同一 symbol 不能在同一池中出现两次）
    3. 池内成员数量无上限（但 UI 可提示）
    """

    # ── 标识与元信息 ─────────────────────────────────────
    id: int
    name: str
    pool_type: PoolType
    description: Optional[str] = None
    color: Optional[str] = None
    icon: Optional[str] = None
    sort_order: int = 0
    is_default: bool = False
    is_archived: bool = False

    # ── 成员 ─────────────────────────────────────────────
    members: list[PoolMember] = field(default_factory=list)

    # ── 时间 ─────────────────────────────────────────────
    created_at: Optional[datetime] = None
    updated_at: Optional[datetime] = None

    # ── 内部状态（不暴露给外部）──────────────────────────
    _next_sort_order: int = field(default=0, repr=False)

    def __post_init__(self):
        AggregateRoot.__init__(self)
        # 如果 members 非空但 _next_sort_order 为 0，重新计算
        if self._next_sort_order == 0 and self.members:
            self._next_sort_order = max(m.sort_order for m in self.members) + 1

    # ════════════════════════════════════════════════════════
    # 工厂方法
    # ════════════════════════════════════════════════════════

    @classmethod
    def create(
        cls,
        name: str,
        pool_type: str = "custom",
        description: Optional[str] = None,
        color: Optional[str] = None,
        icon: Optional[str] = None,
        is_default: bool = False,
        id: int = 0,
    ) -> StockPool:
        """工厂方法：创建新池"""
        now = datetime.now()
        pool = cls(
            id=id,
            name=name,
            pool_type=PoolType.from_code(pool_type),
            description=description,
            color=color,
            icon=icon,
            sort_order=0,
            is_default=is_default,
            is_archived=False,
            members=[],
            created_at=now,
            updated_at=now,
        )
        pool.add_event(PoolCreated(pool))
        return pool

    @classmethod
    def reconstitute(
        cls,
        id: int,
        name: str,
        pool_type: str,
        description: Optional[str],
        color: Optional[str],
        icon: Optional[str],
        sort_order: int,
        is_default: bool,
        is_archived: bool,
        members: list[PoolMember],
        created_at: datetime,
        updated_at: datetime,
    ) -> StockPool:
        """工厂方法：从数据库重建聚合根"""
        pool = cls(
            id=id,
            name=name,
            pool_type=PoolType.from_code(pool_type),
            description=description,
            color=color,
            icon=icon,
            sort_order=sort_order,
            is_default=is_default,
            is_archived=is_archived,
            members=members,
            created_at=created_at,
            updated_at=updated_at,
        )
        return pool

    # ════════════════════════════════════════════════════════
    # 业务行为
    # ════════════════════════════════════════════════════════

    def add_member(self, symbol: str, memo: str = "") -> PoolMember:
        """添加池成员

        规则：
        - 不允许重复（同一 symbol 已在池中）
        - symbol 必须为 6 位数字
        """
        symbol = symbol.strip()
        if any(m.symbol == symbol for m in self.members):
            raise DuplicateMemberError(symbol, self.id)

        member = PoolMember(
            symbol=symbol,
            memo=memo,
            sort_order=self._next_sort_order,
            added_at=datetime.now(),
        )
        self.members.append(member)
        self._next_sort_order += 1
        self.touch()
        self.add_event(MemberAdded(self.id, symbol))
        return member

    def remove_member(self, symbol: str) -> PoolMember:
        """移除池成员"""
        symbol = symbol.strip()
        for i, m in enumerate(self.members):
            if m.symbol == symbol:
                removed = self.members.pop(i)
                self.touch()
                self.add_event(MemberRemoved(self.id, symbol))
                return removed
        raise MemberNotFoundError(symbol, self.id)

    def update_member_memo(self, symbol: str, memo: str) -> None:
        """更新成员备注"""
        for m in self.members:
            if m.symbol == symbol:
                # 值对象不可变，需要重建
                idx = self.members.index(m)
                new_member = PoolMember(
                    symbol=m.symbol,
                    memo=memo,
                    sort_order=m.sort_order,
                    added_at=m.added_at,
                )
                self.members[idx] = new_member
                self.touch()
                return
        raise MemberNotFoundError(symbol, self.id)

    def clear_members(self) -> list[PoolMember]:
        """清空所有成员"""
        removed = self.members.copy()
        self.members.clear()
        self.touch()
        for m in removed:
            self.add_event(MemberRemoved(self.id, m.symbol))
        return removed

    def rename(self, new_name: str) -> None:
        """重命名池"""
        PoolName.from_raw(new_name)  # 验证合法性
        self.name = new_name
        self.touch()
        self.add_event(PoolRenamed(self.id, self.name))

    def update_description(self, description: Optional[str]) -> None:
        """更新描述"""
        self.description = description
        self.touch()

    def update_color(self, color: Optional[str]) -> None:
        """更新颜色"""
        if color:
            PoolColor.from_hex(color)  # 验证合法性
        self.color = color
        self.touch()

    def update_icon(self, icon: Optional[str]) -> None:
        """更新图标"""
        self.icon = icon
        self.touch()

    def archive(self) -> None:
        """归档池（软删除）"""
        self.is_archived = True
        self.touch()
        self.add_event(PoolArchived(self.id))

    def unarchive(self) -> None:
        """取消归档"""
        self.is_archived = False
        self.touch()

    def mark_default(self) -> None:
        """设为默认池"""
        self.is_default = True
        self.touch()

    # ════════════════════════════════════════════════════════
    # 查询方法
    # ════════════════════════════════════════════════════════

    def has_member(self, symbol: str) -> bool:
        """是否包含某成员"""
        return any(m.symbol == symbol for m in self.members)

    def get_member(self, symbol: str) -> Optional[PoolMember]:
        """获取指定成员"""
        for m in self.members:
            if m.symbol == symbol:
                return m
        return None

    @property
    def member_count(self) -> int:
        return len(self.members)

    @property
    def symbols(self) -> list[str]:
        """获取所有成员代码"""
        return [m.symbol for m in self.members]

    # ════════════════════════════════════════════════════════
    # 私有方法
    # ════════════════════════════════════════════════════════

    def _can_delete(self) -> bool:
        """是否可以删除（默认池不可删）"""
        return not self.is_default

    def touch(self) -> None:
        """更新时间戳"""
        self.updated_at = datetime.now()

    def _validate_deletable(self) -> None:
        """验证是否可以删除，失败则抛异常"""
        if not self._can_delete():
            raise CannotDeleteDefaultPoolError(self.id)

    # ════════════════════════════════════════════════════════
    # 序列化（供仓储层 / 应用层使用）
    # ════════════════════════════════════════════════════════

    def to_dict(self) -> dict:
        """转为字典（供 Pydantic / JSON 序列化）"""
        return {
            "id": self.id,
            "name": self.name,
            "pool_type": self.pool_type.code,
            "description": self.description,
            "color": self.color,
            "icon": self.icon,
            "sort_order": self.sort_order,
            "is_default": self.is_default,
            "is_archived": self.is_archived,
            "member_count": self.member_count,
            "created_at": self.created_at,
            "updated_at": self.updated_at,
        }

    def member_list_dict(self) -> list[dict]:
        """成员列表转为字典"""
        return [
            {
                "symbol": m.symbol,
                "memo": m.memo,
                "sort_order": m.sort_order,
                "added_at": m.added_at,
            }
            for m in self.members
        ]
```

```python
    """池操作记录视图对象"""

    pool_id: int
    operation_type: str
    status: str
    params: dict = Field(default_factory=dict)
    result_summary: Optional[dict] = None
    progress: dict = Field(default_factory=dict)
    error_message: Optional[str] = None
    started_at: Optional[datetime] = None
    finished_at: Optional[datetime] = None
    created_at: Optional[datetime] = None

    @property
    def progress_text(self) -> str:
        """进度文本描述"""
        total = self.progress.get("total", 0)
        done = self.progress.get("done", 0)
        if total == 0:
            return "等待中..."
        return f"{done}/{total}"

    @property
    def progress_percent(self) -> int:
        """进度百分比"""
        total = self.progress.get("total", 0)
        done = self.progress.get("done", 0)
        if total == 0:
            return 0
        return int(done / total * 100)
```

## 7. 关键设计决策

### 7.1 为什么 `PoolMember` 是值对象而不是实体？

| 分析 | 结论 |
|------|------|
| `PoolMember` 无全局唯一 ID | 只有 `(pool_id, symbol)` 联合主键 |
| 相等性基于 `(pool_id, symbol)` | 无唯一标识，不需要身份生命周期 |
| 成员修改（如改备注）需要重建 | 值对象不可变特性天然适合 |

### 7.2 聚合根边界

`StockPool` 聚合根包含：
- ✅ 池元信息（name / type / color）
- ✅ 成员列表（`members: list[PoolMember]`）
- ❌ **不包含**：操作记录（`pool_operations` 是独立的聚合，通过 `pool_id` 引用）

### 7.3 为什么不把 `pool_operations` 纳入聚合根？

| 原因 | 说明 |
|------|------|
| 数据量差异 | `stock_pools` 最多 1000 行，`pool_operations` 可达 10 万行 |
| 生命周期不同 | 操作记录永久保留，池可删除 |
| 事务边界 | 操作记录需要独立的事务（异步任务中写入） |

### 7.4 成员 `symbol` 不建外键的原因

已在 [数据库设计文档](./02-数据库设计.md) 中说明：
1. 允许包含退市股票
2. 退市股票可被删除，但池成员保留
3. 应用层 JOIN 校验存在性

### 7.5 为什么用 Protocol 而非 ABC？

| 对比 | Protocol | ABC |
|------|---------|-----|
| 结构检查 | 运行时 duck typing | 运行时 + IDE |
| 侵入性 | 无需继承 | 需要继承 |
| 兼容性 | 与其他框架耦合低 | SQLAlchemy 会冲突 |

项目已有 `StockInfoRepository` 用 `@runtime_checkable Protocol`，保持一致。
