# 03 — 列自定义与看板预设模板（全链路方案）

> 状态：方案，**未改代码**
> 范围：纯前端，**0 后端改动**（localStorage 持久化）
> 主路由：`/home/my-views`（独立路由，左侧菜单加「🎨 我的视图」一级入口）
> 嵌入点：`/home/realtime-board` 的「列设置」按钮 + 顶部「视图切换」下拉

---

## 1. 业务需求

| 维度 | 说明 |
|---|---|
| 场景 | 不同用户关注不同列（短线看换手 / 量比；价值看 PE / PB），且希望保存个人偏好 |
| 核心功能 | ① 单页列自定义（可见性 / 顺序 / 列宽） ② 整套列预设保存为「视图模板」 ③ 视图模板管理（重命名 / 删除 / 设默认） ④ 跨页面共享（预设可应用到任意用 `useColumnPresets` 的页面） |
| 持久化 | **localStorage**（不存后端，避免新增表 / 接口 / 权限） |
| 预设模板 | 内置 3 套（短线王 / 价值投资 / 默认全量），用户可自建任意多套 |
| 导出 | JSON 复制到剪贴板（暂不实现导入 / 跨设备同步） |

### 4.2 `frontend/src/composables/useColumnPresets.ts`（核心 composable）

```ts
import { ref, computed, watch, onMounted } from 'vue'
import type { ColumnDef } from '@/views/realtime-board/columns'

/** 单个列预设模板 */
export interface PresetTemplate {
  id: string                  // uuid
  name: string                // "短线王"
  description?: string
  icon?: string               // emoji
  isBuiltin?: boolean         // 内置不可删
  isDefault?: boolean         // 默认应用
  createdAt: number           // Date.now()
  /** 列定义：[{ key, width, visible }] — 只存用户改过的字段，其他从 COLUMNS 查 */
  columns: Array<{
    key: string
    width?: number            // 用户拖过的宽度
    visible: boolean
  }>
  /** 视图模式偏好 */
  viewMode?: 'table' | 'card' | 'heatmap'
  /** 排序默认 */
  defaultSort?: { prop: string; order: 'ascending' | 'descending' }
}

const STORAGE_KEY = 'rb-presets-v1'
const ACTIVE_KEY = 'rb-preset-active'
const VISIBLE_KEY = 'rb-visible-cols-v1'

/** 内置模板 */
const BUILTIN_PRESETS: PresetTemplate[] = [
  {
    id: 'builtin-short-term',
    name: '短线王',
    description: '换手 / 量比 / 涨速为主',
    icon: '⚡',
    isBuiltin: true,
    columns: [
      { key: 'symbol', visible: true, width: 90 },
      { key: 'name', visible: true, width: 100 },
      { key: 'price', visible: true, width: 90 },
      { key: 'pct_change', visible: true, width: 100 },
      { key: 'change', visible: true, width: 90 },
      { key: 'turnover_rate', visible: true, width: 80 },
      { key: 'volume_ratio', visible: true, width: 70 },
      { key: 'speed_5m', visible: true, width: 100 },
      { key: 'amount', visible: true, width: 110 },
      { key: 'pools', visible: true, width: 200 },
    ],
    viewMode: 'table',
    defaultSort: { prop: 'pct_change', order: 'descending' },
  },
  {
    id: 'builtin-value',
    name: '价值投资',
    description: 'PE / PB / 总市值为主',
    icon: '💎',
    isBuiltin: true,
    columns: [
      { key: 'symbol', visible: true, width: 90 },
      { key: 'name', visible: true, width: 100 },
      { key: 'industry', visible: true, width: 110 },
      { key: 'total_mv', visible: true, width: 100 },
      { key: 'pe_ttm', visible: true, width: 90 },
      { key: 'pb', visible: true, width: 80 },
      { key: 'latest_close', visible: true, width: 80 },
      { key: 'pct_change', visible: true, width: 100 },
      { key: 'pools', visible: true, width: 200 },
    ],
    viewMode: 'table',
    defaultSort: { prop: 'pe_ttm', order: 'ascending' },
  },
  {
    id: 'builtin-full',
    name: '默认全量',
    description: '所有列（按 columns.ts 默认）',
    icon: '📊',
    isBuiltin: true,
    isDefault: true,
    columns: [],   // 空 = 用 COLUMNS 的 defaultVisible
    viewMode: 'table',
  },
]

function loadAll(): PresetTemplate[] {
  try {
    const raw = localStorage.getItem(STORAGE_KEY)
    const userPresets: PresetTemplate[] = raw ? JSON.parse(raw) : []
    return [...BUILTIN_PRESETS, ...userPresets]
  } catch {
    return [...BUILTIN_PRESETS]
  }
}

function saveAll(presets: PresetTemplate[]) {
  try {
    const userOnes = presets.filter(p => !p.isBuiltin)
    localStorage.setItem(STORAGE_KEY, JSON.stringify(userOnes))
  } catch { /* quota */ }
}

function getActiveId(): string {
  return localStorage.getItem(ACTIVE_KEY) || 'builtin-full'
}

/** useColumnPresets — 主 composable
 *
 * 用法（在 RealtimeBoard.vue 中）：
 *   const { visibleColumns, applyPreset, presets, activePreset, saveAsPreset } = useColumnPresets(COLUMNS)
 *   const v = visibleColumns.value  // 喂给 <el-table-column v-for>
 */
export function useColumnPresets(allColumns: ColumnDef[]) {
  const presets = ref<PresetTemplate[]>(loadAll())
  const activeId = ref<string>(getActiveId())
  const customizerVisible = ref(false)

  const activePreset = computed(() =>
    presets.value.find(p => p.id === activeId.value) || presets.value[0]
  )

  /** 当前显示列：合并 allColumns + active preset */
  const visibleColumns = computed<ColumnDef[]>(() => {
    const preset = activePreset.value
    return allColumns
      .map(col => {
        const override = preset.columns.find(c => c.key === col.key)
        if (!override) return col.defaultVisible ? col : null
        return {
          ...col,
          visible: override.visible,
          width: override.width ?? col.defaultWidth,
        }
      })
      .filter((c): c is ColumnDef => c !== null && c.visible)
  })

  function applyPreset(id: string) {
    activeId.value = id
    try { localStorage.setItem(ACTIVE_KEY, id) } catch {}
  }

  function saveAsPreset(name: string, description?: string): PresetTemplate {
    const newPreset: PresetTemplate = {
      id: 'user-' + Date.now().toString(36),
      name,
      description,
      icon: '🎨',
      createdAt: Date.now(),
      columns: visibleColumns.value.map(c => ({
        key: String(c.key), width: c.defaultWidth, visible: true,
      })),
    }
    presets.value = [...presets.value, newPreset]
    saveAll(presets.value)
    activeId.value = newPreset.id
    try { localStorage.setItem(ACTIVE_KEY, newPreset.id) } catch {}
    return newPreset
  }

  function updatePreset(id: string, patch: Partial<PresetTemplate>) {
    const idx = presets.value.findIndex(p => p.id === id)
    if (idx === -1) return
    if (presets.value[idx].isBuiltin) return  // 内置不可改
    presets.value[idx] = { ...presets.value[idx], ...patch }
    saveAll(presets.value)
  }

  function deletePreset(id: string) {
    const p = presets.value.find(x => x.id === id)
    if (!p || p.isBuiltin) return
    presets.value = presets.value.filter(x => x.id !== id)
    saveAll(presets.value)
    if (activeId.value === id) applyPreset('builtin-full')
  }

  function setDefault(id: string) {
    presets.value = presets.value.map(p => ({ ...p, isDefault: p.id === id }))
    saveAll(presets.value)
  }

  /** 实时更新当前预设的某一列宽度（用户在表格上拖列宽时调用） */
  function updateColumnWidth(key: string, width: number) {
    const preset = activePreset.value
    if (!preset) return
    const col = preset.columns.find(c => c.key === key)
    if (col) col.width = width
    if (!preset.isBuiltin) saveAll(presets.value)
  }

  /** 切换列可见性 */
  function toggleColumn(key: string, visible: boolean) {
    const preset = activePreset.value
    if (!preset) return
    const col = preset.columns.find(c => c.key === key)
    if (col) {
      col.visible = visible
    } else {
      preset.columns.push({ key, visible })
    }
    if (!preset.isBuiltin) saveAll(presets.value)
    // 内置：也保存到 user 层（覆盖）— 通过下一次 saveAsPreset 落盘
  }

  /** 导出当前预设为 JSON（剪贴板） */
  function exportPreset(id: string): string {
    const p = presets.value.find(x => x.id === id)
    if (!p) return ''
    return JSON.stringify(p, null, 2)
  }

  return {
    presets, activeId, activePreset, visibleColumns,
    customizerVisible,
    applyPreset, saveAsPreset, updatePreset, deletePreset, setDefault,
    updateColumnWidth, toggleColumn, exportPreset,
  }
}
```

### 4.3 `ColumnCustomizer.vue`（列自定义抽屉）

```vue
<template>
  <el-drawer v-model="visible" title="列设置" size="420px" direction="rtl">
    <div class="cc-wrap">
      <!-- 顶部：预设切换 + 另存为 -->
      <div class="cc-toolbar">
        <el-select v-model="selectedPresetId" @change="onSwitchPreset">
          <el-option v-for="p in presets" :key="p.id" :label="p.icon + ' ' + p.name" :value="p.id" />
        </el-select>
        <el-button @click="onSaveAs" :disabled="!currentName">另存为</el-button>
        <el-button v-if="!isBuiltin" type="danger" link @click="onDelete">删除</el-button>
        <el-button v-if="!isBuiltin && !isDefault" @click="onSetDefault">设为默认</el-button>
      </div>

      <!-- 列拖拽列表（用 Sortable.js 或原生 HTML5 drag） -->
      <div class="cc-list">
        <div
          v-for="col in orderedColumns"
          :key="col.key"
          class="cc-item"
          draggable="true"
          @dragstart="onDragStart($event, col.key)"
          @dragover.prevent
          @drop="onDrop($event, col.key)"
        >
          <el-checkbox v-model="col.visible" @change="onToggle(col)">
            {{ col.label }}
          </el-checkbox>
          <el-input-number
            v-model="col.width"
            :min="40" :max="400" :step="10"
            size="small"
            class="cc-width"
            @change="onWidthChange(col)"
          />
        </div>
      </div>
    </div>
  </el-drawer>
</template>

<script setup lang="ts">
import { ref, computed, watch } from 'vue'
import { ElMessage, ElMessageBox } from 'element-plus'
import type { ColumnDef } from '../columns'

const props = defineProps<{
  visible: boolean
  columns: ColumnDef[]      // 当前可见列（从 useColumnPresets 传过来）
  allColumns: ColumnDef[]   // 全部列定义
  presets: any[]
  activePresetId: string
  applyPreset: (id: string) => void
  saveAsPreset: (name: string, desc?: string) => any
  deletePreset: (id: string) => void
  setDefault: (id: string) => void
  toggleColumn: (key: string, visible: boolean) => void
  updateColumnWidth: (key: string, width: number) => void
}>()
const emit = defineEmits<{ (e: 'update:visible', v: boolean): void }>()

const visible = computed({
  get: () => props.visible,
  set: v => emit('update:visible', v),
})
const selectedPresetId = ref(props.activePresetId)
watch(() => props.activePresetId, v => { selectedPresetId.value = v })

const isBuiltin = computed(() => {
  const p = props.presets.find((x: any) => x.id === selectedPresetId.value)
  return p?.isBuiltin
})
const isDefault = computed(() => {
  const p = props.presets.find((x: any) => x.id === selectedPresetId.value)
  return p?.isDefault
})

// 列出所有列（从 allColumns 拿定义，叠加用户当前可见性）
const orderedColumns = computed(() => {
  return props.allColumns.map(c => {
    const cur = props.columns.find(x => x.key === c.key)
    return {
      ...c,
      visible: cur ? true : false,   // 当前 visible
      width: cur?.defaultWidth ?? c.defaultWidth,
    }
  })
})

function onSwitchPreset(id: string) {
  props.applyPreset(id)
  ElMessage.success('已切换预设')
}

async function onSaveAs() {
  const { value: name } = await ElMessageBox.prompt('请输入新预设名称', '另存为预设', { inputPattern: /.+/, inputErrorMessage: '名称不能为空' })
  props.saveAsPreset(name, '')
  ElMessage.success('已保存')
}
function onDelete() {
  ElMessageBox.confirm('确认删除此预设？', '提示').then(() => {
    props.deletePreset(selectedPresetId.value)
    selectedPresetId.value = 'builtin-full'
    ElMessage.success('已删除')
  }).catch(() => {})
}
function onSetDefault() {
  props.setDefault(selectedPresetId.value)
  ElMessage.success('已设为默认')
}
function onToggle(col: any) {
  props.toggleColumn(col.key, col.visible)
}
function onWidthChange(col: any) {
  props.updateColumnWidth(col.key, col.width)
}

// HTML5 拖拽
let dragKey: string | null = null
function onDragStart(e: DragEvent, key: string) { dragKey = key }
function onDrop(e: DragEvent, targetKey: string) {
  if (!dragKey || dragKey === targetKey) return
  // 简单实现：把 dragKey 移到 targetKey 位置
  // ... 详细实现见 PR
  dragKey = null
}
</script>

.cc-item {
  display: flex; justify-content: space-between; align-items: center;
  padding: 8px 12px; background: #f8fafc; border-radius: 6px; cursor: move;
}
.cc-width { width: 110px; }
</style>
```

```vue
<template>
  <PageWrapper>
    <template #title>🎨 我的视图</template>
    <template #toolbar>
      <div class="toolbar-row">
        <div class="toolbar-search">
          <el-input v-model="search" placeholder="搜索视图..." clearable :prefix-icon="Search" />
        </div>

            新建视图
          </el-button>
        </div>
      </div>
    </template>

    <div class="mv-grid">
      <PresetCard
        v-for="p in filteredPresets"
        :key="p.id"
        :preset="p"
        :is-active="p.id === activeId"
        @apply="onApply"
        @edit="onEdit"
        @delete="onDelete"
        @set-default="onSetDefault"
        @export="onExport"
      />
    </div>

    <PresetEditorDialog
      v-model:visible="createDialogVisible"
      :preset="editingPreset"
      @saved="onSaved"
    />
  </PageWrapper>

<script setup lang="ts">
import { ref, computed } from 'vue'
import { ElMessage, ElMessageBox } from 'element-plus'
import { Search, Plus } from '@element-plus/icons-vue'
import PageWrapper from '@/components/PageWrapper.vue'
import PresetCard from './components/PresetCard.vue'
import PresetEditorDialog from './components/PresetEditorDialog.vue'
import { useColumnPresets } from '@/composables/useColumnPresets'

const search = ref('')
const { presets, activeId, applyPreset, deletePreset, setDefault } = useColumnPresets([] as any)

const filteredPresets = computed(() => {
  const q = search.value.toLowerCase().trim()
  if (!q) return presets.value
  return presets.value.filter(p => p.name.toLowerCase().includes(q) || (p.description ?? '').toLowerCase().includes(q))
})

function onApply(id: string) { applyPreset(id); ElMessage.success('已应用') }
function onEdit(p: any) { editingPreset.value = p; createDialogVisible.value = true }
async function onDelete(p: any) {
  await ElMessageBox.confirm(`确认删除「${p.name}」？`, '提示')
  deletePreset(p.id)
  ElMessage.success('已删除')
}
function onSetDefault(p: any) { setDefault(p.id); ElMessage.success('已设为默认') }
async function onExport(p: any) {
  const json = JSON.stringify(p, null, 2)
  await navigator.clipboard.writeText(json)
  ElMessage.success('已复制到剪贴板')
}

const createDialogVisible = ref(false)
const editingPreset = ref<any>(null)
function onSaved() { createDialogVisible.value = false; editingPreset.value = null }
```

## 5. 嵌入点：realtime-board 与 my-views 的联动

| 位置 | 触发 | 行为 |
|---|---|---|
| `/home/realtime-board` 顶部「列设置」按钮 | 点击 | 打开 `ColumnCustomizer` 抽屉 |
| `ColumnCustomizer` 顶部下拉 | 切换 | 调 `applyPreset(id)` → 表格列立即变化 |
| `ColumnCustomizer` 「另存为」 | 点击 | 弹 prompt 输名字 → 调 `saveAsPreset` |
| `ColumnCustomizer` 「删除」 | 点击 | 调 `deletePreset` → 自动切回默认 |
| `/home/my-views` 卡片「应用」 | 点击 | 调 `applyPreset(id)` → 跳 `/home/realtime-board`（带 query 提示已切换） |
| `/home/my-views` 卡片「设为默认」 | 点击 | 调 `setDefault(id)` |

## 6. 关键决策

| 问 | 答 |
|---|---|
| 为什么不存后端？ | 简化：0 表 / 0 接口 / 0 鉴权。如未来要云端同步，加 `sys_user_preferences` 表 + 接口即可。 |
| 跨页面共享怎么实现？ | `useColumnPresets` 内部用 `localStorage` 共享 key `rb-presets-v1`，所有 `useColumnPresets` 实例读同一份数据。其他模块（如 Dashboard 卡片布局）未来要共享，新建独立 key。 |
| 预设导出格式？ | JSON（包含 id / name / description / columns / viewMode / defaultSort），用户可粘贴到 IM 分享；不实现导入（避免恶意 JSON 注入） |
| 列宽调整怎么持久化？ | 表格列宽由 Element Plus 的 `@header-dragend` 事件触发，实时调 `updateColumnWidth(key, width)` → 写 localStorage。 |
| 拖拽排序用什么？ | 原生 HTML5 drag API（避免引入 sortable.js 增加 ~10KB）。如未来要更复杂（如跨分组拖），再换 vuedraggable。 |
