## 1. application/service/backup_app_service.py

> **注意**：v1 合并了原"TaskManager"功能到此文件——AppService 自己用 `asyncio.create_task` 管理后台任务。

```python
class BackupAppService:
    def __init__(
        self,
        *,
        engine,                    # SQLAlchemy Engine
        backup_engine: BackupEngine,    # 来自 infrastructure/adapter/backup
        restorer: Restorer,
        path_resolver: PathResolver,
    ):
        self._engine = engine
        self._backup_engine = backup_engine
        self._restorer = restorer
        self._path_resolver = path_resolver
        # 内存任务表：{ record_id: asyncio.Task }
        self._running_tasks: dict[int, asyncio.Task] = {}

    # ── 备份 ──────────────────────────────────────────
    async def create_backup(self, req: "CreateBackupRequest", user_id: int) -> int:
        """1. 校验并发（只允许 1 个 running）
        2. INSERT sys_backup_records (status=pending)
        3. asyncio.create_task(self._backup_engine.run(record_id, req))
        4. 返回 record_id
        """
        await self._assert_no_running_backup()
        async with self._engine.begin() as conn:
            record_id = (await conn.execute(text("""
                INSERT INTO sys_backup_records (
                    name, backup_type, scope, tables, tables_count,
                    output_dir, file_name, created_by, status
                ) VALUES (
                    :name, :bt, :scope, CAST(:tables AS jsonb), :tc,
                    :od, :fn, :uid, 'pending'
                ) RETURNING id
            """), {...})).scalar()
        task = asyncio.create_task(self._safe_run_backup(record_id, req))
        self._running_tasks[record_id] = task
        return record_id

    async def _safe_run_backup(self, record_id, req):
        try:
            await self._backup_engine.run(record_id, req)
        except Exception as e:
            await self.mark_failed(record_id, error=str(e))
        finally:
            self._running_tasks.pop(record_id, None)

    # ── 备份进度 / 状态 ───────────────────────────────
    async def update_progress(self, record_id: int, progress: int, msg: str) -> None:
        """由 BackupEngine 调用，写 status='running' + progress"""
        async with self._engine.begin() as conn:
            await conn.execute(text("""
                UPDATE sys_backup_records
                SET status = 'running', progress = :p, progress_msg = :m
                WHERE id = :id
            """), {"p": progress, "m": msg, "id": record_id})

    async def mark_success(self, record_id: int, file_path: Path, file_size: int) -> None:
        async with self._engine.begin() as conn:
            await conn.execute(text("""
                UPDATE sys_backup_records
                SET status = 'success', progress = 100, file_path = :fp,
                    file_size = :fs, finished_at = NOW()
                WHERE id = :id
            """), {"fp": str(file_path), "fs": file_size, "id": record_id})

    async def mark_failed(self, record_id: int, error: str) -> None:
        async with self._engine.begin() as conn:
            await conn.execute(text("""
                UPDATE sys_backup_records
                SET status = 'failed', error_message = :err, finished_at = NOW()
                WHERE id = :id
            """), {"err": error, "id": record_id})

    # ── 备份列表 / 单条 / 下载 / 删除 ─────────────────
    async def list_backups(self, page, page_size, status=None) -> tuple[list, int]:
        """返回 (items, total)"""
        ...

    async def get_backup(self, record_id: int) -> dict:
        ...

    async def delete_backup(self, record_id: int) -> None:
        """1. 校验 status in (success, failed) → 不能删 running
        2. 删 .sql 文件
        3. DELETE row
        """
        ...

    async def get_download_path(self, record_id: int) -> Path:
        """返回 .sql 路径（前端用 a 标签下载）"""
        ...

    # ── 表元信息 ──────────────────────────────────────
    async def list_tables(self) -> list[dict]:
        """返回 [{name, row_count, size_mb, category}, ...]"""
        # row_count / size_mb 用 SELECT reltuples / pg_total_relation_size
        # category 硬编码：业务表 = 白名单，采集表 = 含 _daily 后缀，系统表 = sys_*
        ...

    # ── 路径配置 ──────────────────────────────────────
    async def get_backup_config(self) -> dict:
        """返回 {default_output_dir, allowed_roots, schema_version}"""
        return {
            "default_output_dir": settings.BACKUP_DIR,
            "allowed_roots": settings.BACKUP_ALLOWED_ROOTS,
            "schema_version": settings.BACKUP_SCHEMA_VERSION,
            "max_file_size": settings.BACKUP_MAX_FILE_SIZE,
        }

    async def update_backup_config(self, req: "UpdateConfigRequest", user_id: int) -> None:
        """更新默认路径（写到 settings 或 DB？）—— v1 写内存 dict"""
        ...

    # ── 恢复 ──────────────────────────────────────────
    async def restore(self, req: "RestoreRequest", user_id: int) -> int:
        """1. 校验 confirm=True
        2. INSERT sys_restore_records (status=pending)
        3. asyncio.create_task(self._safe_run_restore(record_id, req))
        """
        if not req.confirm:
            raise BackupError("恢复操作必须 confirm=True")
        ...

    async def list_restores(self, page, page_size) -> tuple[list, int]:
        ...

    async def get_restore(self, record_id: int) -> dict:
        ...

    # ── 工具 ──────────────────────────────────────────
    async def _assert_no_running_backup(self) -> None:
        async with self._engine.connect() as conn:
            count = (await conn.execute(text(
                "SELECT COUNT(*) FROM sys_backup_records WHERE status = 'running'"
            ))).scalar()
            if count > 0:
                raise BackupTaskRunningError("已有备份任务在跑")

    async def recover_interrupted_tasks(self) -> None:
        """启动时调用：把上次进程异常退出遗留的 running 标记为 failed"""
        async with self._engine.begin() as conn:
            await conn.execute(text("""
                UPDATE sys_backup_records
                SET status = 'failed', error_message = COALESCE(error_message,'') || ' [服务重启，任务中断]',
                    finished_at = NOW()
                WHERE status = 'running'
            """))
            await conn.execute(text("""
                UPDATE sys_restore_records
                SET status = 'failed', error_message = COALESCE(error_message,'') || ' [服务重启，任务中断]',
                    finished_at = NOW()
                WHERE status = 'running'
            """))
```

### DI 工厂

```python
# infrastructure/config/di.py 追加
def get_backup_app_service(
    session: AsyncSession = Depends(get_db),
) -> "BackupAppService":
    from application.service.backup_app_service import BackupAppService
    from infrastructure.adapter.backup.engine import BackupEngine
    from infrastructure.adapter.backup.restorer import Restorer

    engine = sync_engine  # 用 sync engine 给 adapter（更简单）
    return BackupAppService(
        engine=engine,
        backup_engine=BackupEngine(
            schema_dumper=SchemaDumper(engine),
            data_dumper=DataDumper(engine),
            path_resolver=PathResolver(settings.BACKUP_ALLOWED_ROOTS),
            app_service=...  # ← 注意循环依赖：先用 lambda
        ),
        restorer=Restorer(engine=engine, path_resolver=...),
        path_resolver=PathResolver(settings.BACKUP_ALLOWED_ROOTS),
    )
```

> **循环依赖**：BackupEngine 需要 app_service 调 update_progress，但 AppService 又持有 BackupEngine。
> **解法**：AppService 持有 BackupEngine 用 None 初始化，构造完 AppService 后再 `backup_engine.app_service = self`。
> 或用 closure：
> ```python
> class BackupAppService:
>     def __init__(self, ...):
>         self._backup_engine = BackupEngine(..., app_service=self)
> ```

## 5. 鉴权：追加 `system:backup` 权限

```sql
-- 在 auth 系统初始化后，admin 角色默认拿到全部权限
-- 但 system:backup 是新权限，需要 INSERT：
INSERT INTO sys_permissions (code, name, module) VALUES
  ('system:backup', '数据备份与恢复', 'system')
ON CONFLICT (code) DO NOTHING;

INSERT INTO sys_role_permissions (role_id, permission_id)
SELECT r.id, p.id
FROM sys_roles r, sys_permissions p
WHERE r.code = 'admin' AND p.code = 'system:backup'
ON CONFLICT DO NOTHING;
```

> 路由层依赖：和现有 `require_permission("system:backup")` 一致。

## 6. 关键决策小结

| 决策 | 选型 | 不选 |
|---|---|---|
| AppService 内部管理任务 | `asyncio.create_task` + dict | 独立 TaskManager 类 |
| 进度回调 | `app_service.update_progress()` 直调 | Callable[[P], Awaitable[None]] |
| 状态字段 | 3 态枚举 + `progress` 0-100 | 8 态状态机 |
| 异常失败处理 | `try/except/finally` | 装饰器 |
| DTO 组织 | 6 文件分 `request/response/` | 单文件 |
| 循环依赖解法 | AppService 构造后绑定 backup_engine.app_service | 引入 setter |
| 上传 | 落盘 + 解析头部 | 直接读 stream |
| 启动期恢复 | SQL UPDATE 标记 failed | 真正恢复 |
