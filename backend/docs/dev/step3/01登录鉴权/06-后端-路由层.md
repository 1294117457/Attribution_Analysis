## 2. `route/dto/auth.py` — 11 个 Pydantic DTO

### 2.1 风格对齐

参考 `route/dto/request/stock.py`：
- 用 `BaseModel` + `Field(..., description=...)`
- 用 Pydantic v2 注解（`Optional[str] = None`）
- `EmailStr` 校验邮箱格式

### 2.2 请求 DTO（6 个）

| 类 | 字段 | 用途 |
|---|---|---|
| `RegisterRequest` | email (`EmailStr`), username (3-32), password (8-64), nickname? (max 64) | 注册 |
| `LoginRequest` | email (`EmailStr`), password | 登录 |
| `RefreshTokenRequest` | refresh_token | 刷新 |
| `ChangePasswordRequest` | old_password, new_password (8-64) | 自己改密码（**预留**，当前不实现）|
| `UpdateUserStatusRequest` | is_active | 启用/禁用 |
| `ResetPasswordRequest` | new_password (8-64) | 重置密码 |
| `AssignRoleRequest` | role_id | 分配角色 |

> 上表 7 个但合并到「6 个请求 DTO」是因为 `RegisterRequest` 含 4 字段。**实际有 7 个类**。

### 2.3 响应 DTO（5 个）

| 类 | 字段 | 用途 |
|---|---|---|
| `TokenResponse` | access_token, refresh_token, token_type, expires_in, user_info? | 登录响应 |
| `RefreshTokenResponse` | access_token, refresh_token, token_type, expires_in | 刷新响应 |
| `UserInfoResponse` | id, email, username, nickname?, avatar_url?, is_active, is_verified, roles, permissions, created_at?, last_login_at? | 用户详情 |
| `UserListResponse` | items, total, page, page_size, pages | 列表分页 |
| `RoleResponse` | id, code, name, description?, is_system, sort_order | 角色 |
| `PermissionResponse` | id, code, resource, action, name, description? | 权限 |

> 实际 6 个响应类。

### 2.4 关键代码骨架

```python
# backend/src/route/dto/auth.py
from datetime import datetime
from typing import Optional
from pydantic import BaseModel, Field, EmailStr


# ── 请求 ──
class RegisterRequest(BaseModel):
    email: EmailStr = Field(..., description="邮箱")
    username: str = Field(..., min_length=3, max_length=32)
    password: str = Field(..., min_length=8, max_length=64)
    nickname: Optional[str] = Field(None, max_length=64)


class LoginRequest(BaseModel):
    email: EmailStr = Field(..., description="邮箱")
    password: str = Field(..., description="密码")


class RefreshTokenRequest(BaseModel):
    refresh_token: str = Field(..., description="Refresh token")


class ChangePasswordRequest(BaseModel):
    old_password: str
    new_password: str = Field(..., min_length=8, max_length=64)


class UpdateUserStatusRequest(BaseModel):
    is_active: bool


class ResetPasswordRequest(BaseModel):
    new_password: str = Field(..., min_length=8, max_length=64)


class AssignRoleRequest(BaseModel):
    role_id: int


# ── 响应 ──
class TokenResponse(BaseModel):
    access_token: str
    refresh_token: str
    token_type: str = "Bearer"
    expires_in: int
    user_info: Optional[dict] = None


class RefreshTokenResponse(BaseModel):
    access_token: str
    refresh_token: str
    token_type: str = "Bearer"
    expires_in: int


class UserInfoResponse(BaseModel):
    id: int
    email: str
    username: str
    nickname: Optional[str] = None
    avatar_url: Optional[str] = None
    is_active: bool
    is_verified: bool
    roles: list[dict] = []
    permissions: list[str] = []
    created_at: Optional[datetime] = None
    last_login_at: Optional[datetime] = None


class UserListResponse(BaseModel):
    items: list[UserInfoResponse]
    total: int
    page: int = 1
    page_size: int = 20
    pages: int = 0


class RoleResponse(BaseModel):
    id: int
    code: str
    name: str
    description: Optional[str] = None
    is_system: bool
    sort_order: int


class PermissionResponse(BaseModel):
    id: int
    code: str
    resource: str
    action: str
    name: str
    description: Optional[str] = None
```

## 3. `route/api/v1/deps_auth.py` — FastAPI Depends 工厂

### 3.1 4 个依赖

| 依赖 | 作用 | 用法 |
|---|---|---|
| `get_current_user_id` | 解 token → 返 user_id (int) | 路由参数注入 |
| `get_current_user` | 解 token → 查 user → 返完整 dict | 路由参数注入 |
| `require_role(*roles)` | 工厂：要求用户拥有指定角色之一 | 路由 `dependencies=[Depends(require_role("admin"))]` |
| `require_permission(*perms)` | 工厂：要求用户拥有指定权限之一 | 路由 `dependencies=[Depends(require_permission("user:write"))]` |

### 3.2 关键代码骨架

```python
# backend/src/route/api/v1/deps_auth.py
from typing import Optional
from fastapi import Depends, HTTPException, Request, status
from fastapi.security import HTTPBearer, HTTPAuthorizationCredentials

from application.service.auth_app_service import AuthAppService
from infrastructure.persistence.connection import get_db
from sqlalchemy.ext.asyncio import AsyncSession

security = HTTPBearer(auto_error=False)


async def get_current_user_id(
    request: Request,
    credentials: Optional[HTTPAuthorizationCredentials] = Depends(security),
    session: AsyncSession = Depends(get_db),
) -> int:
    if not credentials or not credentials.credentials:
        raise HTTPException(
            status_code=401,
            detail={"code": 401, "message": "未登录", "data": None},
            headers={"WWW-Authenticate": "Bearer"},
        )
    service = AuthAppService(session)
    try:
        payload = await service.parse_access_token(credentials.credentials)
        user_id = int(payload["sub"])
        request.state.token_claims = payload  # 给后续依赖用
        return user_id
    except Exception as e:
        raise HTTPException(
            status_code=401,
            detail={"code": 401, "message": str(e), "data": None},
            headers={"WWW-Authenticate": "Bearer"},
        )


async def get_current_user(
    user_id: int = Depends(get_current_user_id),
    session: AsyncSession = Depends(get_db),
) -> dict:
    from infrastructure.persistence.repositories.auth_repository import UserRepoImpl
    from domain.auth.entity import UserNotFoundError
    repo = UserRepoImpl(session)
    user = await repo.find_by_id(user_id)
    if not user:
        raise HTTPException(401, {"code": 401, "message": "用户不存在", "data": None})
    service = AuthAppService(session)
    permissions = await service._get_user_permissions(user)
    return {
        "id": user.id, "email": user.email, "username": user.username,
        "nickname": user.nickname, "avatar_url": user.avatar_url,
        "is_active": user.is_active, "is_verified": user.is_verified,
        "roles": [{"id": r.id, "code": r.code, "name": r.name} for r in user.roles],
        "permissions": permissions,
        "created_at": user.created_at, "last_login_at": user.last_login_at,
    }


def require_role(*required_roles: str):
    async def dependency(
        request: Request,
        user_id: int = Depends(get_current_user_id),
        session: AsyncSession = Depends(get_db),
    ):
        from infrastructure.persistence.repositories.auth_repository import UserRepoImpl
        repo = UserRepoImpl(session)
        user = await repo.find_by_id(user_id)
        if not user:
            raise HTTPException(401, {"code": 401, "message": "用户不存在", "data": None})
        user_roles = [r.code for r in user.roles]
        if not any(role in user_roles for role in required_roles):
            raise HTTPException(
                403,
                {"code": 403, "message": f"需要角色: {', '.join(required_roles)}", "data": None},
            )
    return dependency


def require_permission(*required_permissions: str):
    async def dependency(
        request: Request,
        user_id: int = Depends(get_current_user_id),
        session: AsyncSession = Depends(get_db),
    ):
        from infrastructure.persistence.repositories.auth_repository import UserRepoImpl
        from application.service.auth_app_service import AuthAppService
        repo = UserRepoImpl(session)
        user = await repo.find_by_id(user_id)
        if not user:
            raise HTTPException(401, {"code": 401, "message": "用户不存在", "data": None})
        service = AuthAppService(session)
        user_perms = await service._get_user_permissions(user)
        if not any(perm in user_perms for perm in required_permissions):
            raise HTTPException(
                403,
                {"code": 403, "message": f"需要权限: {', '.join(required_permissions)}", "data": None},
            )
    return dependency
```

### 3.3 设计要点

| 项 | 选型 | Why |
|---|---|---|
| `HTTPBearer(auto_error=False)` | 手动处理 401 | 自动 error 抛出的格式与现有 `{code, message, data}` 不一致 |
| `require_*` 用工厂 | 返回 dependency 函数 | 灵活：`require_role("admin")`、`require_permission("user:write")` |
| 403 区分"未登录"vs"无权限" | 401 vs 403 | 401 → 跳登录；403 → 提示"权限不足" |
| 把 claims 存 `request.state` | 后续依赖可读 | 减少重复查 DB |

```python
# backend/src/route/api/v1/auth.py
from fastapi import APIRouter, Depends, Request, status
from sqlalchemy.ext.asyncio import AsyncSession

from application.service.auth_app_service import AuthAppService
from infrastructure.config.di import get_db
from infrastructure.persistence.repositories.auth_repository import UserRepoImpl
from infrastructure.security.key_manager import get_jwks
from route.dto.auth import (
    RegisterRequest, LoginRequest, RefreshTokenRequest,
    TokenResponse, RefreshTokenResponse,
)
from route.api import _response as R
from route.api.v1.deps_auth import get_current_user_id

router = APIRouter(prefix="/auth", tags=["认证"])


def get_auth_service(session: AsyncSession = Depends(get_db)) -> AuthAppService:
    return AuthAppService(session)


@router.post("/register", summary="注册新用户", status_code=201)
async def register(
    req: RegisterRequest,
    service: AuthAppService = Depends(get_auth_service),
):
    user = await service.register(
        email=req.email, username=req.username,
        password=req.password, nickname=req.nickname,
    )
    return R.created({
        "id": user.id, "email": user.email,
        "username": user.username, "nickname": user.nickname,
        "message": "注册成功，请登录",
    })


@router.post("/login", summary="登录", response_model=TokenResponse)
async def login(
    req: LoginRequest,
    request: Request,
    service: AuthAppService = Depends(get_auth_service),
):
    user_agent = request.headers.get("user-agent", "")
    ip = request.client.host if request.client else ""
    result = await service.login(req.email, req.password, user_agent, ip)
    return R.ok(result)


@router.post("/refresh", summary="刷新 token")
async def refresh(
    req: RefreshTokenRequest,
    service: AuthAppService = Depends(get_auth_service),
):
    result = await service.refresh_token(req.refresh_token)
    return R.ok(result)


@router.post("/logout", summary="登出")
async def logout(
    user_id: int = Depends(get_current_user_id),
    service: AuthAppService = Depends(get_auth_service),
):
    await service.logout(user_id)
    return R.no_content("已登出")


@router.get("/me", summary="当前用户信息")
async def get_me(
    user_id: int = Depends(get_current_user_id),
    session: AsyncSession = Depends(get_db),
):
    repo = UserRepoImpl(session)
    user = await repo.find_by_id(user_id)
    if not user:
        return R.bad_request("用户不存在")
    service = AuthAppService(session)
    permissions = await service._get_user_permissions(user)
    return R.ok({
        "id": user.id, "email": user.email, "username": user.username,
        "nickname": user.nickname, "avatar_url": user.avatar_url,
        "is_active": user.is_active, "is_verified": user.is_verified,
        "roles": [{"id": r.id, "code": r.code, "name": r.name} for r in user.roles],
        "permissions": permissions,
        "created_at": user.created_at.isoformat() if user.created_at else None,
        "last_login_at": user.last_login_at.isoformat() if user.last_login_at else None,
    })


@router.get("/.well-known/jwks.json", summary="JWKS 公钥")
async def get_jwks_endpoint():
    return get_jwks()
```

## 5. `route/api/v1/user.py` — 5 个 user 接口

| 方法 | 路径 | 鉴权 | 用途 |
|---|---|---|---|
| `GET` | `/users/` | `user:read` | 列表（分页 + 关键字 + 状态）|
| `PATCH` | `/users/{id}/status` | `user:write` | 启用/禁用 |
| `POST` | `/users/{id}/reset-password` | `user:write` | 重置密码（强制重登）|
| `POST` | `/users/{id}/roles` | `user:write` + `role:write` | 分配角色 |
| `DELETE` | `/users/{id}/roles/{rid}` | `user:write` + `role:write` | 移除角色 |

```python
# backend/src/route/api/v1/user.py
from fastapi import APIRouter, Depends, Query
from sqlalchemy.ext.asyncio import AsyncSession

from application.service.auth_app_service import AuthAppService
from infrastructure.persistence.connection import get_db
from route.dto.auth import (
    UpdateUserStatusRequest, ResetPasswordRequest, AssignRoleRequest,
    UserInfoResponse, UserListResponse,
)
from route.api import _response as R
from route.api.v1.deps_auth import require_permission

router = APIRouter(prefix="/users", tags=["用户管理"])


def get_auth_service(session: AsyncSession = Depends(get_db)) -> AuthAppService:
    return AuthAppService(session)


@router.get(
    "/",
    summary="用户列表",
    response_model=UserListResponse,
    dependencies=[Depends(require_permission("user:read"))],
)
async def list_users(
    page: int = Query(1, ge=1),
    page_size: int = Query(20, ge=1, le=200),
    keyword: str = Query(None),
    is_active: bool = Query(None),
    service: AuthAppService = Depends(get_auth_service),
):
    users, total = await service.list_users(page, page_size, keyword, is_active)
    items = []
    for u in users:
        perms = await service._get_user_permissions(u)
        items.append(UserInfoResponse(
            id=u.id, email=u.email, username=u.username,
            nickname=u.nickname, avatar_url=u.avatar_url,
            is_active=u.is_active, is_verified=u.is_verified,
            roles=[{"id": r.id, "code": r.code, "name": r.name} for r in u.roles],
            permissions=perms,
            created_at=u.created_at, last_login_at=u.last_login_at,
        ))
    pages = (total + page_size - 1) // page_size
    return R.ok(UserListResponse(
        items=items, total=total, page=page, page_size=page_size, pages=pages
    ).model_dump())


@router.patch(
    "/{user_id}/status",
    summary="启用/禁用用户",
    dependencies=[Depends(require_permission("user:write"))],
)
async def update_user_status(
    user_id: int,
    req: UpdateUserStatusRequest,
    service: AuthAppService = Depends(get_auth_service),
):
    user = await service.update_user_status(user_id, req.is_active)
    return R.ok({
        "id": user.id, "is_active": user.is_active,
        "message": "已启用" if user.is_active else "已禁用",
    })


@router.post(
    "/{user_id}/reset-password",
    summary="重置密码",
    dependencies=[Depends(require_permission("user:write"))],
)
async def reset_password(
    user_id: int,
    req: ResetPasswordRequest,
    service: AuthAppService = Depends(get_auth_service),
):
    await service.reset_user_password(user_id, req.new_password)
    return R.no_content("密码已重置，用户需重新登录")


@router.post(
    "/{user_id}/roles",
    summary="分配角色",
    dependencies=[Depends(require_permission("user:write"))],
)
async def assign_role(
    user_id: int,
    req: AssignRoleRequest,
    service: AuthAppService = Depends(get_auth_service),
):
    await service.assign_role(user_id, req.role_id)
    return R.no_content("角色已分配")


@router.delete(
    "/{user_id}/roles/{role_id}",
    summary="移除角色",
    dependencies=[Depends(require_permission("user:write"))],
)
async def remove_role(
    user_id: int,
    role_id: int,
    service: AuthAppService = Depends(get_auth_service),
):
    await service.remove_role(user_id, role_id)
    return R.no_content("角色已移除")
```

## 10. 响应格式统一（复用 `route/api/_response.py`）

| 场景 | 用法 | 响应 |
|---|---|---|
| 成功 | `R.ok({...})` | `{"code": 200, "message": "success", "data": {...}}` |
| 创建 | `R.created({...})` | `{"code": 201, "message": "创建成功", "data": {...}}` |
| 无内容 | `R.no_content("msg")` | `{"code": 200, "message": "msg", "data": null}` |
| 错误 | `R.bad_request("msg")` | 400 |
