## 3. 关键交互流程

### 3.1 登录

```
用户输入邮箱密码 → 点"登录"
  │
  ├─→ 前端校验（邮箱非空 + 密码 ≥ 8 位）
  ├─→ authStore.login(email, password)
  │     │
  │     ├─→ authApi.login → POST /api/v1/auth/login
  │     │     │
  │     │     ├─→ 200 OK → 存 localStorage + setUserInfo
  │     │     └─→ 401 INVALID_CREDENTIALS → ElMessage.error
  │     │
  │     └─→ ElMessage.success('登录成功')
  │
  └─→ router.push('/home/index')
```

### 3.2 注册

```
用户填写邮箱+用户名+密码+昵称 → 点"注册"
  │
  ├─→ 前端校验（所有字段合法）
  ├─→ authStore.register(data)
  │     │
  │     ├─→ authApi.register → POST /api/v1/auth/register
  │     │     │
  │     │     ├─→ 201 → ElMessage.success('注册成功，请登录')
  │     │     │           activeTab = 'login'
  │     │     │           loginForm.email = 注册的 email
  │     │     │           loginForm.password = 注册的 password
  │     │     │
  │     │     ├─→ 400 USER_EXISTS → ElMessage.error
  │     │     └─→ 400 password 校验失败 → ElMessage.error
  │     │
  │     └─→ 切换 Tab 让用户直接登录
```

<!-- frontend/src/views/auth/LoginPage.vue -->
<template>
  <div class="login-container">
    <div class="login-card">
      <div class="login-header">
        <h2>📈 智能金融归因分析平台</h2>

      </div>

      <el-tabs v-model="activeTab" class="login-tabs">

          >
            <el-form-item prop="email">

                v-model="registerForm.email"

                :prefix-icon="Message"

              />

              />

              <el-input
                v-model="registerForm.password"
                type="password"
                placeholder="密码（至少 8 位）"
                :prefix-icon="Lock"
                show-password
              />

            <el-form-item prop="nickname">
              <el-input
                v-model="registerForm.nickname"
                placeholder="昵称（可选）"
                :prefix-icon="Avatar"
              />
            </el-form-item>
            <el-form-item>
              <el-button
                type="success"
                :loading="loading"
                @click="onRegister"
                style="width: 100%"
              >
                注册
              </el-button>
            </el-form-item>
          </el-form>
        </el-tab-pane>
      </el-tabs>

      <div class="login-footer">
        <small>登录即表示同意《用户协议》和《隐私政策》</small>
      </div>

  </div>

<script setup lang="ts">
import { ref, reactive } from 'vue'
import { useRouter } from 'vue-router'
import { ElMessage, type FormInstance } from 'element-plus'
import { Message, Lock, User, Avatar } from '@element-plus/icons-vue'
import { useAuthStore } from '@/stores/auth'

const router = useRouter()
const authStore = useAuthStore()

const activeTab = ref('login')
const loading = ref(false)

const loginFormRef = ref<FormInstance>()
const loginForm = reactive({ email: '', password: '' })
const loginRules = {
  email: [
    { required: true, message: '请输入邮箱', trigger: 'blur' },
    { type: 'email', message: '邮箱格式错误', trigger: 'blur' },
  ],
  password: [
    { required: true, message: '请输入密码', trigger: 'blur' },
    { min: 8, message: '密码至少 8 位', trigger: 'blur' },
  ],
}

const registerFormRef = ref<FormInstance>()
const registerForm = reactive({
  email: '',
  username: '',
  password: '',
  nickname: '',
})
const registerRules = {
  email: [
    { required: true, message: '请输入邮箱', trigger: 'blur' },
    { type: 'email', message: '邮箱格式错误', trigger: 'blur' },
  ],
  username: [
    { required: true, message: '请输入用户名', trigger: 'blur' },
    { min: 3, max: 32, message: '用户名长度 3-32', trigger: 'blur' },
  ],
  password: [
    { required: true, message: '请输入密码', trigger: 'blur' },
    { min: 8, message: '至少 8 位', trigger: 'blur' },
  ],
}

async function onSubmit() {
  if (!loginFormRef.value) return
  try {
    await loginFormRef.value.validate()
    loading.value = true
    await authStore.login(loginForm.email, loginForm.password)
    ElMessage.success('登录成功')
    router.push('/home/index')
  } catch (e: any) {
    ElMessage.error(e.message || '登录失败')
  } finally {
    loading.value = false
  }
}

async function onRegister() {
  if (!registerFormRef.value) return
  try {
    await registerFormRef.value.validate()
    loading.value = true
    await authStore.register({
      email: registerForm.email,
      username: registerForm.username,
      password: registerForm.password,
      nickname: registerForm.nickname || undefined,
    })
    ElMessage.success('注册成功，请登录')
    activeTab.value = 'login'
    loginForm.email = registerForm.email
    loginForm.password = registerForm.password
  } catch (e: any) {
    ElMessage.error(e.message || '注册失败')
  } finally {
    loading.value = false
  }
}
</script>

## 5. 关键设计

| 项 | 选型 | Why |
|---|---|---|
| 登录 + 注册同页（不是两个路由）| `<el-tabs>` 切换 | 用户体验顺滑；注册完直接登录无需跳页 |
| 注册成功 → 切回登录 Tab | 同一页面 | 避免路由跳转；表单已预填 |
| 邮箱不限学生邮箱 | 不做校验 | 需求 |
| 头像首字母 | register 后端默认 `nickname || username` | 用户无需额外输入 |
| 渐变背景 | `#1e293b → #0f172a` | 与现有 TopBar 风格一致（深色）|
| Form 校验 | element-plus 内置 | 与现有 DTs 系统行为一致 |

| 测试 | 用例 |
|---|---|
| 登录成功 | mock authApi.login → 跳 /home/index |

| 注册成功 | mock authApi.register → 切回登录 Tab + 表单预填 |
