# 操作池 - 应用层设计

> 本文档定义"操作池"模块的应用层设计：应用服务（用例编排）、DTO。
>
> **设计原则**：应用层不包含业务逻辑，只负责业务流程的编排和协调。

---

## 1. 目录结构

```
application/
├── dto/
│   ├── __init__.py
│   ├── pool.py               ← 池 CRUD 相关 DTO
│   └── pool_operation.py     ← 池操作相关 DTO
├── pool_service.py           ← StockPoolAppService
├── pool_operation_service.py ← PoolOperationAppService
└── exceptions.py             ← 应用层异常（补充）
```

```python
from application.exceptions import ApplicationError


class PoolNotFoundError(ApplicationError):
    def __init__(self, pool_id: int):
        self.pool_id = pool_id
        super().__init__(f"操作池不存在: {pool_id}", code=404)


class PoolOperationNotFoundError(ApplicationError):
    def __init__(self, op_id: int):
        self.op_id = op_id
        super().__init__(f"操作记录不存在: {op_id}", code=404)


class CannotDeleteDefaultPoolError(ApplicationError):
    def __init__(self):
        super().__init__("无法删除默认池", code=400)


class DuplicatePoolMemberError(ApplicationError):
    def __init__(self, symbol: str):
        self.symbol = symbol
        super().__init__(f"股票 {symbol} 已在该池中", code=400)


class PoolMemberNotFoundError(ApplicationError):
    def __init__(self, symbol: str):
        self.symbol = symbol
        super().__init__(f"股票 {symbol} 不在该池中", code=404)


class StockNotFoundError(ApplicationError):
    def __init__(self, symbol: str):
        self.symbol = symbol
        super().__init__(f"股票 {symbol} 不存在", code=404)


class PoolOperationConflictError(ApplicationError):
    def __init__(self, pool_id: int):
        self.pool_id = pool_id
        super().__init__(
            f"池 {pool_id} 已有进行中的操作，请等待完成后重试",
            code=409,
        )
```

```python
from __future__ import annotations

from datetime import datetime
from typing import Optional

from pydantic import BaseModel, Field


# ════════════════════════════════════════════════════════════════
# 请求 DTO
# ════════════════════════════════════════════════════════════════

class PoolCreateRequest(BaseModel):
    """创建池请求"""

    name: str = Field(..., min_length=1, max_length=64, description="池名称")
    pool_type: str = Field("custom", description="池类型")
    description: Optional[str] = Field(None, max_length=255, description="描述")
    color: Optional[str] = Field(None, pattern=r"^#[0-9A-Fa-f]{6}$", description="颜色")
    icon: Optional[str] = Field(None, max_length=32, description="图标")

    model_config = {
        "json_schema_extra": {
            "example": {
                "name": "银行股组合",
                "pool_type": "industry",
                "description": "主要持仓的银行股",
                "color": "#52c41a",
                "icon": "🏦",
            }
        }
    }


class PoolUpdateRequest(BaseModel):
    """更新池请求（部分更新）"""

    name: Optional[str] = Field(None, min_length=1, max_length=64)
    description: Optional[str] = Field(None, max_length=255)
    color: Optional[str] = Field(None, pattern=r"^#[0-9A-Fa-f]{6}$")
    icon: Optional[str] = Field(None, max_length=32)
    sort_order: Optional[int] = Field(None, ge=0)


class PoolAddMembersRequest(BaseModel):
    """批量添加成员请求"""

    symbols: list[str] = Field(
        ...,
        min_length=1,
        max_length=500,
        description="股票代码列表（6位数字）",
    )
    validate_exists: bool = Field(
        True, description="是否校验股票是否存在"
    )


class PoolRemoveMembersRequest(BaseModel):
    """批量移除成员请求"""

    symbols: list[str] = Field(..., min_length=1, max_length=500)


class PoolUpdateMemberMemoRequest(BaseModel):
    """更新成员备注请求"""

    symbol: str = Field(..., description="股票代码")
    memo: str = Field("", max_length=255, description="备注")


# ════════════════════════════════════════════════════════════════
# 响应 DTO
# ════════════════════════════════════════════════════════════════

class PoolMemberVO(BaseModel):
    """池成员视图对象"""

    symbol: str
    memo: Optional[str] = None
    sort_order: int = 0
    added_at: Optional[datetime] = None
    name: Optional[str] = None
    industry: Optional[str] = None
    market: Optional[str] = None
    exchange: Optional[str] = None
    is_valid: bool = True  # 股票是否仍存在

    model_config = {"from_attributes": True}


class PoolVO(BaseModel):
    """池视图对象（不含成员）"""

    id: int
    name: str
    pool_type: str
    description: Optional[str] = None
    color: Optional[str] = None
    icon: Optional[str] = None
    sort_order: int = 0
    is_default: bool = False
    is_archived: bool = False
    member_count: int = 0
    created_at: Optional[datetime] = None
    updated_at: Optional[datetime] = None

    model_config = {"from_attributes": True}


class PoolDetailVO(PoolVO):
    """池详情视图对象（包含成员列表）"""

    members: list[PoolMemberVO] = Field(default_factory=list)


class PoolListResponse(BaseModel):
    """池列表响应"""

    total: int

class PoolMemberListResponse(BaseModel):

    pool_id: int

class PoolAddMembersResponse(BaseModel):

    skipped: list[str]  # 已存在或其他原因跳过
    total_added: int
    total_skipped: int


class PoolPoolsBySymbolResponse(BaseModel):
    """反向查询：某股票在哪些池"""

    pools: list[PoolVO]  # 简化版（不含成员）
    total: int
```

```python
class PoolKlineCollectRequest(BaseModel):

    pool_id: int = Field(..., description="池 ID")
    operation_type: Literal["kline_collect"] = "kline_collect"
    days: int = Field(365, ge=1, le=3650, description="回溯天数")
    source: Optional[str] = Field(None, description="数据源 tushare/akshare")


class PoolOperationListRequest(BaseModel):
    """查询操作历史请求"""

    pool_id: int
    limit: int = Field(20, ge=1, le=100)

# ════════════════════════════════════════════════════════════════
# 响应 DTO
# ════════════════════════════════════════════════════════════════

class PoolOperationProgressVO(BaseModel):
    """操作进度视图对象"""

    done: int = 0
    total: int = 0
    failed: int = 0

    @property
    def percent(self) -> int:
        if self.total == 0:
            return 0
        return int(self.done / self.total * 100)


class PoolOperationDetailVO(BaseModel):
    """操作记录明细（包含每只股票的状态）"""

    symbol: str
    status: str  # success / skipped / failed
    count: Optional[int] = None  # 保存条数
    message: Optional[str] = None  # 失败原因


class PoolOperationVO(BaseModel):
    """池操作记录视图对象"""

    id: int
    pool_id: Optional[int] = None
    operation_type: str
    status: str
    params: dict = Field(default_factory=dict)
    result_summary: Optional[dict] = None
    progress: PoolOperationProgressVO
    error_message: Optional[str] = None
    started_at: Optional[datetime] = None
    finished_at: Optional[datetime] = None
    created_at: Optional[datetime] = None

    model_config = {"from_attributes": True}


class PoolOperationListResponse(BaseModel):
    """操作列表响应"""

    pool_id: int
    total: int
    items: list[PoolOperationVO]


class PoolOperationCreateResponse(BaseModel):
    """发起操作响应"""

    operation_id: int
    pool_id: int
    operation_type: str
    status: str
    total: int
    message: str = "操作已派发"


class PoolOperationDetailResponse(BaseModel):
    """操作详情响应（包含明细）"""
```

### 4.1 `StockPoolAppService`

```python
# application/pool_service.py

"""操作池应用服务"""

from __future__ import annotations

import logging
from typing import Optional

from sqlalchemy.ext.asyncio import AsyncSession

from application.dto.pool import (
    PoolCreateRequest,
    PoolUpdateRequest,
    PoolAddMembersRequest,
    PoolRemoveMembersRequest,
    PoolUpdateMemberMemoRequest,
    PoolVO,
    PoolDetailVO,
    PoolMemberVO,
    PoolListResponse,
    PoolMemberListResponse,
    PoolAddMembersResponse,
    PoolPoolsBySymbolResponse,
)
from application.exceptions import (
    PoolNotFoundError,
    CannotDeleteDefaultPoolError,
    DuplicatePoolMemberError,
    PoolMemberNotFoundError,
    StockNotFoundError,
)
from domain.stock_pool.entity import StockPool
from domain.stock_pool.value_objects import PoolMember
from domain.stock_pool.repository import StockPoolRepository
from infrastructure.repositories.pool_repository import StockPoolRepoImpl
from infrastructure.repositories.stock_repository import StockRepoImpl

logger = logging.getLogger(__name__)


class StockPoolAppService:
    """操作池应用服务"""

    def __init__(self, session: AsyncSession):
        self._session = session
        self._repo: StockPoolRepository = StockPoolRepoImpl(session)
        self._stock_repo = StockRepoImpl(session)

    # ════════════════════════════════════════════════════════════════
    # 池 CRUD
    # ════════════════════════════════════════════════════════════════

    async def create_pool(self, request: PoolCreateRequest) -> PoolVO:
        """创建池"""
        pool = StockPool.create(
            name=request.name,
            pool_type=request.pool_type,
            description=request.description,
            color=request.color,
            icon=request.icon,
        )
        created = await self._repo.create(pool)
        return PoolVO.model_validate(created.to_dict())

    async def list_pools(
        self,
        include_archived: bool = False,
        limit: int = 100,
        offset: int = 0,
    ) -> PoolListResponse:
        """列出所有池"""
        pools = await self._repo.find_all(
            include_archived=include_archived,
            limit=limit,
            offset=offset,
        )
        items = [PoolVO.model_validate(p.to_dict()) for p in pools]
        return PoolListResponse(
            total=len(items),
            items=items,
        )

    async def get_pool(self, pool_id: int) -> PoolDetailVO:
        """获取池详情（含成员）"""
        pool = await self._repo.find_by_id(pool_id)
        if not pool:
            raise PoolNotFoundError(pool_id)

        # 加载成员信息（关联 stock_infos）
        members = await self._repo.list_members(pool_id, limit=1000)
        member_vos = []
        for m in members:
            # 查询关联的股票信息
            stock = await self._stock_repo.find_by_symbol(m.symbol)
            member_vos.append(
                PoolMemberVO(
                    symbol=m.symbol,
                    memo=m.memo,
                    sort_order=m.sort_order,
                    added_at=m.added_at,
                    name=stock.name if stock else None,
                    industry=stock.industry.name if stock and stock.industry else None,
                    market=stock.market.name if stock and stock.market else None,
                    exchange=stock.exchange,
                    is_valid=stock is not None,
                )
            )

        vo = PoolDetailVO.model_validate(pool.to_dict())
        vo.members = member_vos
        return vo

    async def update_pool(
        self, pool_id: int, request: PoolUpdateRequest
    ) -> PoolVO:
        """更新池"""
        pool = await self._repo.find_by_id(pool_id)
        if not pool:
            raise PoolNotFoundError(pool_id)

        if request.name is not None:
            pool.rename(request.name)
        if request.description is not None:
            pool.update_description(request.description)
        if request.color is not None:
            pool.update_color(request.color)
        if request.icon is not None:
            pool.update_icon(request.icon)
        if request.sort_order is not None:
            pool.sort_order = request.sort_order

        updated = await self._repo.update(pool)
        return PoolVO.model_validate(updated.to_dict())

    async def delete_pool(self, pool_id: int) -> None:
        """删除池（默认池不可删）"""
        pool = await self._repo.find_by_id(pool_id)
        if not pool:
            raise PoolNotFoundError(pool_id)
        if pool.is_default:
            raise CannotDeleteDefaultPoolError()

        await self._repo.delete(pool_id)

    async def archive_pool(self, pool_id: int) -> PoolVO:
        """归档池"""
        pool = await self._repo.find_by_id(pool_id)
        if not pool:
            raise PoolNotFoundError(pool_id)
        pool.archive()
        updated = await self._repo.update(pool)
        return PoolVO.model_validate(updated.to_dict())

    # ════════════════════════════════════════════════════════════════
    # 成员管理
    # ════════════════════════════════════════════════════════════════

    async def add_members(
        self, pool_id: int, request: PoolAddMembersRequest
    ) -> PoolAddMembersResponse:
        """批量添加成员"""
        pool = await self._repo.find_by_id(pool_id)
        if not pool:
            raise PoolNotFoundError(pool_id)

        # 验证股票是否存在（可选）
        if request.validate_exists:
            valid_symbols, invalid_symbols = [], []
            for symbol in request.symbols:
                stock = await self._stock_repo.find_by_symbol(symbol)
                if stock:
                    valid_symbols.append(symbol)
                else:
                    invalid_symbols.append(symbol)
        else:
            valid_symbols = request.symbols
            invalid_symbols = []

        added, skipped = await self._repo.add_members_batch(
            pool_id, valid_symbols
        )
        skipped.extend(invalid_symbols)

        return PoolAddMembersResponse(
            pool_id=pool_id,
            added=added,
            skipped=skipped,
            total_added=len(added),
            total_skipped=len(skipped),
        )

    async def remove_members(
        self, pool_id: int, request: PoolRemoveMembersRequest
    ) -> int:
        """批量移除成员"""
        pool = await self._repo.find_by_id(pool_id)
        if not pool:
            raise PoolNotFoundError(pool_id)

        count = await self._repo.remove_members_batch(
            pool_id, request.symbols
        )
        return count

    async def clear_members(self, pool_id: int) -> int:
        """清空池内所有成员"""
        pool = await self._repo.find_by_id(pool_id)
        if not pool:
            raise PoolNotFoundError(pool_id)

        return await self._repo.clear_members(pool_id)

    async def update_member_memo(
        self, pool_id: int, request: PoolUpdateMemberMemoRequest
    ) -> PoolMemberVO:
        """更新成员备注"""
        pool = await self._repo.find_by_id(pool_id)
        if not pool:
            raise PoolNotFoundError(pool_id)

        if not pool.has_member(request.symbol):
            raise PoolMemberNotFoundError(request.symbol)

        await self._repo.update_member_memo(
            pool_id, request.symbol, request.memo
        )

        stock = await self._stock_repo.find_by_symbol(request.symbol)
        return PoolMemberVO(
            symbol=request.symbol,
            memo=request.memo,
            name=stock.name if stock else None,
            is_valid=stock is not None,
        )

    async def list_members(
        self,
        pool_id: int,
        limit: int = 100,
        offset: int = 0,
    ) -> PoolMemberListResponse:
        """列出池成员"""
        pool = await self._repo.find_by_id(pool_id)
        if not pool:
            raise PoolNotFoundError(pool_id)

        members = await self._repo.list_members(pool_id, limit, offset)
        member_vos = []
        for m in members:
            stock = await self._stock_repo.find_by_symbol(m.symbol)
            member_vos.append(
                PoolMemberVO(
                    symbol=m.symbol,
                    memo=m.memo,
                    sort_order=m.sort_order,
                    added_at=m.added_at,
                    name=stock.name if stock else None,
                    industry=stock.industry.name if stock and stock.industry else None,
                    market=stock.market.name if stock and stock.market else None,
                    exchange=stock.exchange,
                    is_valid=stock is not None,
                )
            )

        total = await self._repo.count_members(pool_id)
        return PoolMemberListResponse(
            pool_id=pool_id,
            total=total,
            items=member_vos,
        )

    # ════════════════════════════════════════════════════════════════
    # 反向查询
    # ════════════════════════════════════════════════════════════════

    async def find_pools_by_symbol(self, symbol: str) -> PoolPoolsBySymbolResponse:
        """反向查询：某股票在哪些池里"""
        pools = await self._repo.find_pools_by_symbol(symbol)
        pool_vos = [PoolVO.model_validate(p.to_dict()) for p in pools]
        return PoolPoolsBySymbolResponse(
            symbol=symbol,
            pools=pool_vos,
            total=len(pool_vos),
        )

    # ════════════════════════════════════════════════════════════════
    # 辅助方法
    # ════════════════════════════════════════════════════════════════

    async def ensure_default_pool(self) -> PoolVO:
        """确保存在默认池（首次启动调用）"""
        existing = await self._repo.find_by_default()
        if existing:
            return PoolVO.model_validate(existing.to_dict())

        pool = StockPool.create(
            name="我的自选",
            pool_type="watchlist",
            is_default=True,
            icon="⭐",
            color="#FFB800",
        )
        created = await self._repo.create(pool)
        return PoolVO.model_validate(created.to_dict())
```

```python
# application/pool_operation_service.py

"""池操作（批量任务）应用服务"""

from __future__ import annotations

import logging
from typing import Optional

from sqlalchemy.ext.asyncio import AsyncSession

from application.dto.pool_operation import (
    PoolKlineCollectRequest,
    PoolOperationListRequest,
    PoolOperationVO,
    PoolOperationListResponse,
    PoolOperationCreateResponse,
    PoolOperationDetailResponse,
    PoolOperationProgressVO,
)
from application.exceptions import (
    PoolNotFoundError,
    PoolOperationNotFoundError,
    PoolOperationConflictError,
)
from domain.stock_pool.repository import StockPoolRepository, PoolOperationRepository
from infrastructure.repositories.pool_repository import (
    StockPoolRepoImpl,
    PoolOperationRepoImpl,
)
from infrastructure.tasks.operation_dispatcher import OperationDispatcher

logger = logging.getLogger(__name__)


class PoolOperationAppService:
    """池操作应用服务"""

    def __init__(self, session: AsyncSession):
        self._session = session
        self._pool_repo: StockPoolRepository = StockPoolRepoImpl(session)
        self._op_repo: PoolOperationRepository = PoolOperationRepoImpl(session)
        self._dispatcher = OperationDispatcher(session)

    async def create_kline_collect_operation(
        self, request: PoolKlineCollectRequest
    ) -> PoolOperationCreateResponse:
        """对池发起 K 线采集操作"""
        pool = await self._pool_repo.find_by_id(request.pool_id)
        if not pool:
            raise PoolNotFoundError(request.pool_id)

        # 检查是否有进行中的同类型操作
        existing = await self._op_repo.list_by_pool(request.pool_id, limit=1)
        for op in existing:
            if op["status"] in ("pending", "running"):
                raise PoolOperationConflictError(request.pool_id)

        # 获取成员列表
        members = await self._pool_repo.list_members(request.pool_id, limit=10000)
        symbols = [m.symbol for m in members]

        if not symbols:
            return PoolOperationCreateResponse(
                operation_id=0,
                pool_id=request.pool_id,
                operation_type=request.operation_type,
                status="skipped",
                total=0,
                message="池内没有成员",
            )

        # 创建操作记录
        params = {
            "days": request.days,
            "source": request.source,
        }
        op_id = await self._op_repo.create(
            pool_id=request.pool_id,
            operation_type=request.operation_type,
            params=params,
        )

        # 更新进度总数
        await self._op_repo.update_progress(op_id, done=0, total=len(symbols))

        # 派发后台任务
        await self._dispatcher.dispatch_kline_collect(
            op_id=op_id,
            pool_id=request.pool_id,
            symbols=symbols,
            days=request.days,
            source=request.source,
        )

        return PoolOperationCreateResponse(
            operation_id=op_id,
            pool_id=request.pool_id,
            operation_type=request.operation_type,
            status="pending",
            total=len(symbols),
            message=f"操作已派发，共 {len(symbols)} 只股票",
        )

    async def list_operations(
        self, request: PoolOperationListRequest
    ) -> PoolOperationListResponse:
        """列出池的操作历史"""
        pool = await self._pool_repo.find_by_id(request.pool_id)
        if not pool:
            raise PoolNotFoundError(request.pool_id)

        ops = await self._op_repo.list_by_pool(
            request.pool_id, limit=request.limit, offset=request.offset
        )

        items = []
        for op in ops:
            progress = PoolOperationProgressVO(
                done=op["progress"].get("done", 0),
                total=op["progress"].get("total", 0),
                failed=op["progress"].get("failed", 0),
            )
            items.append(
                PoolOperationVO(
                    id=op["id"],
                    pool_id=op["pool_id"],
                    operation_type=op["operation_type"],
                    status=op["status"],
                    params=op["params"],
                    result_summary=op["result_summary"],
                    progress=progress,
                    error_message=op["error_message"],
                    started_at=op["started_at"],
                    finished_at=op["finished_at"],
                    created_at=op["created_at"],
                )
            )

        return PoolOperationListResponse(
            pool_id=request.pool_id,
            total=len(items),
            items=items,
        )

    async def get_operation(self, op_id: int) -> PoolOperationVO:
        """获取操作详情"""
        op = await self._op_repo.find_by_id(op_id)
        if not op:
            raise PoolOperationNotFoundError(op_id)

        progress = PoolOperationProgressVO(
            done=op["progress"].get("done", 0),
            total=op["progress"].get("total", 0),
            failed=op["progress"].get("failed", 0),
        )
        return PoolOperationVO(
            id=op["id"],
            pool_id=op["pool_id"],
            operation_type=op["operation_type"],
            status=op["status"],
            params=op["params"],
            result_summary=op["result_summary"],
            progress=progress,
            error_message=op["error_message"],
            started_at=op["started_at"],
            finished_at=op["finished_at"],
            created_at=op["created_at"],
        )

    async def get_operation_progress(
        self, op_id: int
    ) -> PoolOperationProgressVO:
        """获取操作进度（轻量接口，供轮询）"""
        op = await self._op_repo.find_by_id(op_id)
        if not op:
            raise PoolOperationNotFoundError(op_id)

        return PoolOperationProgressVO(
            done=op["progress"].get("done", 0),
            total=op["progress"].get("total", 0),
            failed=op["progress"].get("failed", 0),
        )

    async def cancel_operation(self, op_id: int) -> PoolOperationVO:
        """取消操作（仅 pending 可取消）"""
        op = await self._op_repo.find_by_id(op_id)
        if not op:
            raise PoolOperationNotFoundError(op_id)

        if op["status"] not in ("pending", "running"):
            from application.exceptions import ApplicationError
            raise ApplicationError("该操作无法取消", code=400)

        await self._op_repo.update_status(
            op_id, status="cancelled", finished_at=op.get("started_at")
        )
        return await self.get_operation(op_id)
```

## 6. 关键设计决策

### 6.1 批量操作参数校验

```python
# 添加成员时校验股票存在性（可选）
async def add_members(self, pool_id: int, request: PoolAddMembersRequest):
    if request.validate_exists:
        for symbol in request.symbols:
            stock = await self._stock_repo.find_by_symbol(symbol)
            if not stock:
                raise StockNotFoundError(symbol)
```

**理由**：某些场景（如批量导入）可能不需要校验，降低开销。

### 6.2 操作冲突检测

```python
# 发起操作前检查进行中的同类型操作
existing = await self._op_repo.list_by_pool(request.pool_id, limit=1)
for op in existing:
    if op["status"] in ("pending", "running"):
        raise PoolOperationConflictError(request.pool_id)
```

**理由**：避免同一池同时跑两个采集任务，导致数据竞争。

### 6.3 成员列表关联查询

```python
# 加载成员时 JOIN stock_infos
stock = await self._stock_repo.find_by_symbol(m.symbol)
```

**理由**：池成员只存 symbol，需要 JOIN 才能显示名称 / 行业等。需要时可优化为批量查询（IN 查询）。
