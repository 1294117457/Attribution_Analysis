# 操作池 - 前端设计

> 本文档定义"操作池"模块的 Vue 3 前端设计：组件、Pinia Store、API 封装。
>
> **技术栈**：Vue 3 + TypeScript + Element Plus + Pinia + Vite
>
> **设计风格**：与现有 `DataCollect.vue` / `StockPanel.vue` 保持一致。

## 3. API 封装

```typescript
// src/api/pool.ts

import type {
  ApiResponse,
  Pool,
  PoolDetail,
  PoolMember,
  PoolOperation,
  PoolListResponse,
  PoolMemberListResponse,
  PoolAddMembersResponse,
  PoolOperationListResponse,
  PoolOperationCreateResponse,
  PoolCreateRequest,
  PoolUpdateRequest,
  PoolAddMembersRequest,
  PoolKlineCollectRequest,
} from '@/common/types/pool'

const BASE = '/api/v1'

async function request<T>(path: string, options?: RequestInit): Promise<T> {
  const res = await fetch(`${BASE}${path}`, {
    headers: { 'Content-Type': 'application/json' },
    ...options,
  })
  const json: ApiResponse<T> = await res.json()
  if (json.code >= 400) throw new Error(json.message)
  return json.data
}

// ── 池管理 ──────────────────────────────────────────────────────────────

export const poolApi = {
  /** 创建池 */
  create(data: PoolCreateRequest) {
    return request<Pool>('/pools', {
      method: 'POST',
      body: JSON.stringify(data),
    })
  },

  /** 池列表 */
  list(params?: { include_archived?: boolean; limit?: number; offset?: number }) {
    const q = new URLSearchParams(
      Object.entries(params || {}).filter(([, v]) => v !== undefined) as any
    ).toString()
    return request<PoolListResponse>(`/pools${q ? `?${q}` : ''}`)
  },

  /** 池详情 */
  get(poolId: number) {
    return request<PoolDetail>(`/pools/${poolId}`)
  },

  /** 更新池 */
  update(poolId: number, data: PoolUpdateRequest) {
    return request<Pool>(`/pools/${poolId}`, {
      method: 'PATCH',
      body: JSON.stringify(data),
    })
  },

  /** 删除池 */
  delete(poolId: number) {
    return request<void>(`/pools/${poolId}`, { method: 'DELETE' })
  },

  // ── 成员管理 ─────────────────────────────────────────────────────────

  /** 批量添加成员 */
  addMembers(poolId: number, data: PoolAddMembersRequest) {
    return request<PoolAddMembersResponse>(`/pools/${poolId}/members`, {
      method: 'POST',
      body: JSON.stringify(data),
    })
  },

  /** 批量删除成员 */
  removeMembers(poolId: number, symbols: string[]) {
    return request<{ removed_count: number }>(`/pools/${poolId}/members`, {
      method: 'DELETE',
      body: JSON.stringify({ symbols }),
    })
  },

  /** 成员列表 */
  listMembers(poolId: number, params?: { limit?: number; offset?: number }) {
    const q = new URLSearchParams(
      Object.entries(params || {}).filter(([, v]) => v !== undefined) as any
    ).toString()
    return request<PoolMemberListResponse>(
      `/pools/${poolId}/members${q ? `?${q}` : ''}`
    )
  },

  /** 更新成员备注 */
  updateMemberMemo(poolId: number, symbol: string, memo: string) {
    return request<PoolMember>(`/pools/${poolId}/members/${symbol}`, {
      method: 'PATCH',
      body: JSON.stringify({ memo }),
    })
  },

  /** 删除单个成员 */
  removeMember(poolId: number, symbol: string) {
    return request<void>(`/pools/${poolId}/members/${symbol}`, {
      method: 'DELETE',
    })
  },

  // ── 反向查询 ─────────────────────────────────────────────────────────

  /** 某股票在哪些池 */
  findBySymbol(symbol: string) {
    return request<{
      symbol: string
      pools: Pool[]
      total: number
    }>(`/pools/by-symbol/${symbol}`)
  },

  // ── 池级操作 ─────────────────────────────────────────────────────────

  /** 发起操作 */
  createOperation(poolId: number, data: PoolKlineCollectRequest) {
    return request<PoolOperationCreateResponse>(
      `/pools/${poolId}/operations`,
      { method: 'POST', body: JSON.stringify({ pool_id: poolId, ...data }) }
    )
  },

  /** 操作历史 */
  listOperations(poolId: number, params?: { limit?: number; offset?: number }) {
    const q = new URLSearchParams(
      Object.entries(params || {}).filter(([, v]) => v !== undefined) as any
    ).toString()
    return request<PoolOperationListResponse>(
      `/pools/${poolId}/operations${q ? `?${q}` : ''}`
    )
  },

  /** 操作详情 */
  getOperation(opId: number) {
    return request<PoolOperation>(`/operations/${opId}`)
  },

  /** 操作进度 */
  getProgress(opId: number) {
    return request<{ done: number; total: number; failed: number }>(
      `/operations/${opId}/progress`
    )
  },

  /** 取消操作 */
  cancelOperation(opId: number) {
    return request<{ id: number; status: string }>(`/operations/${opId}/cancel`, {
      method: 'POST',
    })
  },
}
```

## 4. Pinia Store

```typescript
// src/stores/pool.ts

import { defineStore } from 'pinia'
import { ref, computed } from 'vue'
import type {
  Pool,
  PoolDetail,
  PoolOperation,
  PoolMember,
} from '@/common/types/pool'
import { poolApi } from '@/api/pool'

export const usePoolStore = defineStore('pool', () => {
  // ── State ─────────────────────────────────────────────────────────────

  const pools = ref<Pool[]>([])
  const currentPool = ref<PoolDetail | null>(null)
  const operations = ref<PoolOperation[]>([])
  const loading = ref(false)
  const currentOperation = ref<PoolOperation | null>(null)

  // ── Getters ──────────────────────────────────────────────────────────

  const defaultPool = computed(() => pools.value.find((p) => p.is_default))
  const sortedPools = computed(() =>
    [...pools.value].sort((a, b) => {
      if (a.is_default !== b.is_default) return b.is_default ? 1 : -1
      return a.sort_order - b.sort_order
    })
  )

  // ── Actions ───────────────────────────────────────────────────────────

  async function fetchPools() {
    loading.value = true
    try {
      const res = await poolApi.list({ limit: 100 })
      pools.value = res.items
    } finally {
      loading.value = false
    }
  }

  async function fetchPoolDetail(poolId: number) {
    loading.value = true
    try {
      currentPool.value = await poolApi.get(poolId)
    } finally {
      loading.value = false
    }
  }

  async function createPool(data: Parameters<typeof poolApi.create>[0]) {
    const pool = await poolApi.create(data)
    pools.value.push(pool)
    return pool
  }

  async function updatePool(
    poolId: number,
    data: Parameters<typeof poolApi.update>[1]
  ) {
    const updated = await poolApi.update(poolId, data)
    const idx = pools.value.findIndex((p) => p.id === poolId)
    if (idx >= 0) pools.value[idx] = { ...pools.value[idx], ...updated }
    if (currentPool.value?.id === poolId) {
      currentPool.value = { ...currentPool.value, ...updated }
    }
    return updated
  }

  async function deletePool(poolId: number) {
    await poolApi.delete(poolId)
    pools.value = pools.value.filter((p) => p.id !== poolId)
    if (currentPool.value?.id === poolId) currentPool.value = null
  }

  async function addMembers(
    poolId: number,
    symbols: string[],
    validateExists = true
  ) {
    const result = await poolApi.addMembers(poolId, {
      symbols,
      validate_exists: validateExists,
    })
    // 刷新池详情
    await fetchPoolDetail(poolId)
    return result
  }

  async function removeMembers(poolId: number, symbols: string[]) {
    await poolApi.removeMembers(poolId, symbols)
    if (currentPool.value?.id === poolId) {
      currentPool.value.members = currentPool.value.members.filter(
        (m) => !symbols.includes(m.symbol)
      )
      if (currentPool.value) currentPool.value.member_count -= symbols.length
    }
  }

  async function fetchOperations(poolId: number, limit = 20) {
    const res = await poolApi.listOperations(poolId, { limit })
    operations.value = res.items
  }

  async function startKlineCollect(
    poolId: number,
    days = 365,
    source?: string
  ) {
    const result = await poolApi.createOperation(poolId, {
      operation_type: 'kline_collect',
      days,
      source,
    })
    // 添加到操作列表
    operations.value.unshift({
      id: result.operation_id,
      pool_id: poolId,
      operation_type: 'kline_collect',
      status: 'pending',
      params: { days, source },
      progress: { done: 0, total: result.total, failed: 0 },
      created_at: new Date().toISOString(),
    } as PoolOperation)
    currentOperation.value = operations.value[0]
    return result
  }

  async function pollProgress(opId: number) {
    const progress = await poolApi.getProgress(opId)
    const op = operations.value.find((o) => o.id === opId)
    if (op) {
      op.progress = progress
      if (progress.done === progress.total && progress.total > 0) {
        op.status = progress.failed > 0 ? 'partial' : 'success'
      } else if (progress.done > 0) {
        op.status = 'running'
      }
    }
    return progress
  }

  async function refreshOperation(opId: number) {
    const op = await poolApi.getOperation(opId)
    const idx = operations.value.findIndex((o) => o.id === opId)
    if (idx >= 0) operations.value[idx] = op
    return op
  }

  return {
    // State
    pools,
    currentPool,
    operations,
    loading,
    currentOperation,
    // Getters
    defaultPool,
    sortedPools,
    // Actions
    fetchPools,
    fetchPoolDetail,
    createPool,
    updatePool,
    deletePool,
    addMembers,
    removeMembers,
    fetchOperations,
    startKlineCollect,
    pollProgress,
    refreshOperation,
  }
})
```

## 7. 进度追踪方案

### 7.1 轮询方案（简单，推荐 P0）

```typescript
// PoolOperations.vue 中
const pollTimer = ref<ReturnType<typeof setInterval> | null>(null)

async function startPolling(opId: number) {
  stopPolling()
  pollTimer.value = setInterval(async () => {
    const progress = await poolStore.pollProgress(opId)
    if (progress.done === progress.total) {
      stopPolling()
      await poolStore.refreshOperation(opId)
    }
  }, 2000) // 每 2 秒轮询
}

function stopPolling() {
  if (pollTimer.value) {
    clearInterval(pollTimer.value)
    pollTimer.value = null
  }
}
```

### 7.2 SSE 方案（推荐 P1）

```typescript
// 建立 SSE 连接
const eventSource = new EventSource(`/api/v1/operations/${opId}/stream`)

eventSource.addEventListener('progress', (e) => {
  const data = JSON.parse(e.data)
  poolStore.operations.find((o) => o.id === opId)!.progress = data
})

eventSource.addEventListener('finished', (e) => {
  const data = JSON.parse(e.data)
  poolStore.refreshOperation(opId)
  eventSource.close()
})
```

## 10. 关键 UI 决策

| 问题 | 决策 |
|------|------|
| 池列表布局 | 卡片式（`el-card`），方便一眼看到多个池 |
| 成员表格 | 类似 StockPanel 的 `el-table`，方便批量操作 |
| 操作按钮 | 主要 CTA（采集K线）放在右上角 |
| 操作进度 | 内联进度条（`el-progress`）+ 实时数字 |
| 添加成员 | 复用 StockPanel 的股票选择逻辑（已有） |
| 空状态 | "从左侧选股" 引导（与 DataCollect 一致） |
| 错误提示 | `ElMessage.error`（与全局一致） |
