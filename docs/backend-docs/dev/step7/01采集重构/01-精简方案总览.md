# 01 · 精简方案总览

> 目标：把 26 个 `BaseCollectTask` 子类、5 装饰器链、23 个 `XxxPort`、`DataGateway` 全部拆掉，**用 22 个轻量 `CollectService` + `ServiceRegistry`（内存）+ 2 张表（`collect_schedule` / `collect_run_log`）+ `ScheduleRunner` 取代**。
>
> 本文给出 **Service 抽象**、**ServiceRegistry**、**ScheduleRunner**、**2 张表的 DDL/ORM**、**第一个完整 Service 实现**（DailyKlineService）。

## 1. CollectService 抽象

### 1.1 Protocol 定义

```python
# contexts/collect/application/service/base.py
from __future__ import annotations

from dataclasses import dataclass
from typing import ClassVar, Protocol


@dataclass(frozen=True)
class CollectResult:
    """采集一次的结果——调度日志和 HTTP 返回都用这个"""
    total: int = 0          # 单元总数（如 5000 只股票）
    success: int = 0        # 成功的单元数
    fail: int = 0           # 失败的单元数
    skip: int = 0           # 跳过的单元数（无数据 / 已覆盖）
    saved: int = 0          # 写入 DB 的行数
    message: str = ""       # 一行可读的总结（"日K采集 共 5000 只, 成功 4995, 失败 0, 跳过 5"）

    def to_dict(self) -> dict:
        return {
            "total": self.total,
            "success": self.success,
            "fail": self.fail,
            "skip": self.skip,
            "saved": self.saved,
            "message": self.message,
        }


class CollectService(Protocol):
    """采集 service 抽象接口——所有 service 都要实现

    约束：
      ① name 必须全局唯一（用于 collect_schedule.service_name 关联）
      ② facet 必须是 tech / capital / fundamental / news / sentiment 之一
      ③ collect(params) 是统一入口：调度和 HTTP 都调这个
      ④ DEFAULT_PARAMS 用于 collect_schedule.params 为空时的兜底
      ⑤ RATE_LIMIT 用于 service 内部的限流（RedisCache.rate_limit 滑动窗口）
      ⑥ CONCURRENCY 用于 service 内部的并发（asyncio.Semaphore）
    """

    name: ClassVar[str]                          # "daily_kline"
    facet: ClassVar[str]                         # "tech"
    sub_facet: ClassVar[str] = ""                # "kline"
    label: ClassVar[str] = ""                    # "日 K 线"
    description: ClassVar[str] = ""              # 描述

    DEFAULT_PARAMS: ClassVar[dict] = {}          # {"days": 7, "source": "tushare"}
    RATE_LIMIT: ClassVar[tuple[str, int, int] | None] = None  # ("tushare", 200, 60)
    CONCURRENCY: ClassVar[int] = 1               # 默认 5

    async def collect(self, params: dict) -> CollectResult:
        """统一入口：调度触发和 HTTP 触发都调这个

        params 来源（合并顺序，后者覆盖前者）：
          ① DEFAULT_PARAMS（类变量）
          ② collect_schedule.params（数据库）
          ③ 用户 HTTP 请求的 body（仅 HTTP 触发时）
        """
        ...
```

### 1.2 与 DDD 的契合度

- **不算"贫血"**：service 持有 fetcher / repo / cache / universe，业务逻辑（限流、并发、入库、过滤、汇总）都在 service
- **不算"充血"**：service 不持有状态（所有依赖通过构造器注入）；collect() 是无状态操作
- **不引入"task" 概念**：service = task 的全部职责 + 更简单（不抽象成基类）

## 2. ServiceRegistry（内存）

```python
# contexts/collect/application/service/registry.py
from __future__ import annotations

import logging
from typing import Optional

from contexts.collect.application.service.base import CollectService

logger = logging.getLogger(__name__)


class ServiceRegistry:
    """service_name → CollectService 实例的内存注册表

    替代旧 CollectTaskRegistry：
      - 启动期 setup_services() 装配 22 个 service
      - 调度和 HTTP 都通过 name 查
      - O(1) dict 查；不持久化
    """

    def __init__(self) -> None:
        self._services: dict[str, CollectService] = {}

    def register(self, service: CollectService) -> None:
        if not service.name:
            raise ValueError(f"{type(service).__name__}.name 不能为空")
        if service.name in self._services:
            raise ValueError(
                f"service {service.name!r} 重复注册：{type(self._services[service.name]).__name__} / {type(service).__name__}"
            )
        self._services[service.name] = service
        logger.info("注册 service: %s (%s)", service.name, type(service).__name__)

    def get(self, name: str) -> Optional[CollectService]:
        return self._services.get(name)

    def require(self, name: str) -> CollectService:
        svc = self._services.get(name)
        if svc is None:
            raise KeyError(f"service {name!r} 未注册；已注册：{list(self._services)}")
        return svc

    def all(self) -> list[CollectService]:
        return list(self._services.values())

    def names(self) -> list[str]:
        return list(self._services.keys())


_registry: Optional[ServiceRegistry] = None


def get_service_registry() -> ServiceRegistry:
    """全局单例（启动期 setup_services 注入，运行时只读）"""
    global _registry
    if _registry is None:
        _registry = ServiceRegistry()
    return _registry


def set_service_registry(registry: ServiceRegistry) -> None:
    """测试期可替换"""
    global _registry
    _registry = registry


__all__ = ["ServiceRegistry", "get_service_registry", "set_service_registry"]
```

**特点**：
- 不查 DB
- 不做"目录"（不返回 facet 分组——`collect_schedule` 表已经有 facet 字段）
- 不做"健康检查"（service 自己初始化时挂掉就挂掉）
- **O(1) dict 查**

## 3. 2 张表的 DDL

### 3.1 `collect_schedule`（调度计划）

**用途**：记录"哪个 service 什么时候跑、跑什么参数、并发多少、是否启用"。

**DDL**：

```sql
-- ============================================================
-- collect_schedule：调度计划表
-- 替代 collect_jobs 的"调度字段"（mode/times/interval_seconds/schedule_type/concurrency/enabled/params）
-- 运行时状态移到 collect_run_log
-- ============================================================
CREATE TABLE collect_schedule (
    id              SERIAL PRIMARY KEY,

    -- 哪个 service
    service_name    VARCHAR(64) NOT NULL,            -- "daily_kline" / "stock_basic" / ...

    -- 调度配置
    enabled         BOOLEAN NOT NULL DEFAULT TRUE,   -- false 时不触发（保留配置）
    trigger_type    VARCHAR(16) NOT NULL,            -- 'time' / 'interval'（不允许 once——once 用 HTTP 触发即可）
    -- 当 trigger_type='time' 时
    times           TEXT[] NOT NULL DEFAULT '{}',    -- ['09:31', '15:31']，HH:MM，分钟精度，去重升序
    -- 当 trigger_type='interval' 时
    interval_seconds INT,                            -- >= 30（避免打爆源）

    -- 业务参数（透传给 service.collect()）
    params          JSONB NOT NULL DEFAULT '{}',     -- {"days": 7, "source": "tushare"}

    -- 执行控制
    concurrency     INT NOT NULL DEFAULT 5,          -- service 内部并发（asyncio.Semaphore 容量）

    -- 描述
    display_name    VARCHAR(128),
    description     TEXT,

    created_at      TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at      TIMESTAMPTZ NOT NULL DEFAULT NOW(),

    -- 校验
    CONSTRAINT ck_collect_schedule_trigger CHECK (
        (trigger_type = 'time' AND times <> '{}' AND interval_seconds IS NULL) OR
        (trigger_type = 'interval' AND interval_seconds IS NOT NULL AND interval_seconds >= 30)
    )
);

CREATE INDEX ix_collect_schedule_enabled
    ON collect_schedule(enabled) WHERE enabled = TRUE;

COMMENT ON TABLE collect_schedule IS
    '采集调度计划：定义某个 service 在何时被自动触发；运行时日志在 collect_run_log';
COMMENT ON COLUMN collect_schedule.service_name IS
    'service 名称（与 ServiceRegistry 注册名一致）';
COMMENT ON COLUMN collect_schedule.trigger_type IS
    'time=每日定时（用 times 字段） / interval=固定频率（用 interval_seconds 字段）';
COMMENT ON COLUMN collect_schedule.times IS
    '每日触发时刻（HH:MM）；仅 trigger_type=time 时使用；启动期去重升序校验';
COMMENT ON COLUMN collect_schedule.interval_seconds IS
    '触发间隔（秒，>= 30）；仅 trigger_type=interval 时使用';
COMMENT ON COLUMN collect_schedule.concurrency IS
    'service 内部并发（覆盖 service.CONCURRENCY 默认值）';
```

**约束**：
- `trigger_type='time'` 必须有 `times`，`interval_seconds` 为 NULL
- `trigger_type='interval'` 必须有 `interval_seconds >= 30`，`times` 为空
- 1 行 = 1 个调度计划（不像旧 `collect_jobs` 一行兼有运行时状态）

### 3.2 `collect_run_log`（运行日志）

**用途**：记录每次 service.collect() 执行的开始/结束/结果。**只追加，不更新**（除了 `finished_at` / `status` / `duration_ms`）。

**DDL**：

```sql
-- ============================================================
-- collect_run_log：采集运行日志
-- 替代 collect_jobs 的"运行时字段"（status/total_count/success_count/fail_count/skip_count/saved_count/started_at/finished_at/duration_ms/message/trigger_type）
-- 替代原 collect_logs（业务事件日志；现统一在本表 status=success/failed + message 字段）
-- ============================================================
CREATE TABLE collect_run_log (
    id              BIGSERIAL PRIMARY KEY,

    -- 来源
    schedule_id     INT,                                -- 关联 collect_schedule.id；手动触发为 NULL
    service_name    VARCHAR(64) NOT NULL,                -- 冗余存（service 删/改后仍可追溯）
    trigger_type    VARCHAR(16) NOT NULL,                -- 'schedule' / 'manual' / 'api'

    -- 参数（执行时合并后的最终值，便于事后复盘）
    params          JSONB NOT NULL DEFAULT '{}',

    -- 时间
    started_at      TIMESTAMPTZ NOT NULL,
    finished_at     TIMESTAMPTZ,
    duration_ms     INT,

    -- 结果
    status          VARCHAR(16) NOT NULL DEFAULT 'running',   -- 'running' / 'success' / 'failed' / 'cancelled'
    total_count     INT NOT NULL DEFAULT 0,
    success_count   INT NOT NULL DEFAULT 0,
    fail_count      INT NOT NULL DEFAULT 0,
    skip_count      INT NOT NULL DEFAULT 0,
    saved_count     INT NOT NULL DEFAULT 0,

    message         TEXT,                                -- 一行可读总结
    error           TEXT,                                -- 异常 stack（status=failed 时填）

    created_at      TIMESTAMPTZ NOT NULL DEFAULT NOW(),

    -- 索引：按 service + 时间查（前端"采集历史"页）
    CONSTRAINT ck_collect_run_log_status CHECK (status IN ('running', 'success', 'failed', 'cancelled'))
);

CREATE INDEX ix_collect_run_log_service_time
    ON collect_run_log(service_name, started_at DESC);
CREATE INDEX ix_collect_run_log_status
    ON collect_run_log(status);
CREATE INDEX ix_collect_run_log_schedule
    ON collect_run_log(schedule_id);

COMMENT ON TABLE collect_run_log IS
    '采集运行日志：每次 service.collect() 执行一行；只追加';
COMMENT ON COLUMN collect_run_log.trigger_type IS
    'schedule=APScheduler 触发 / manual=用户从采集管理页触发 / api=HTTP /collect/run 触发';
COMMENT ON COLUMN collect_run_log.params IS
    '执行时合并后的最终参数（DEFAULT_PARAMS + schedule.params + HTTP body）';
COMMENT ON COLUMN collect_run_log.status IS
    'running=执行中 / success=全部成功 / failed=有失败 / cancelled=被取消';
```

**特点**：
- 1 行 = 1 次执行
- `schedule_id` 可空（手动触发时）
- `params` 是执行时的最终参数（不是 schedule 表的 params）——便于回放
- `error` 字段只在 `status=failed` 时填

### 3.3 迁移 SQL（从旧 `collect_jobs` 拆出来）

```sql
-- 1) 新建 collect_schedule
CREATE TABLE collect_schedule ( ... );  -- 见 3.1

-- 2) 新建 collect_run_log
CREATE TABLE collect_run_log ( ... );   -- 见 3.2

-- 3) 从 collect_jobs 迁移历史数据
INSERT INTO collect_schedule (
    service_name, enabled, trigger_type, times, interval_seconds,
    params, concurrency, display_name, description, created_at, updated_at

SELECT
    task_type,
    enabled,
    COALESCE(schedule_type, 'time'),  -- 旧 collect_jobs.schedule_type 可能是 NULL
    COALESCE(times, '{}'),

    COALESCE(params, '{}'),
    concurrency,
    display_name,
    NULL,
    created_at,
    updated_at
FROM collect_jobs
WHERE mode IN ('time', 'interval');    -- 只迁有调度的；mode=once 是手动

INSERT INTO collect_run_log (
    schedule_id, service_name, trigger_type, params,
    started_at, finished_at, duration_ms, status,
    total_count, success_count, fail_count, skip_count, saved_count,
    message, created_at
)
SELECT
    NULL,                                -- 旧 collect_jobs 没有 schedule_id 概念，全置 NULL

    trigger_type,
    COALESCE(params, '{}'),
    started_at,
    finished_at,
    duration_ms,
    status,
    total_count,
    success_count,
    fail_count,
    skip_count,
    0,                                   -- 旧表没有 saved_count 字段

    created_at
FROM collect_jobs
WHERE status IN ('success', 'failed', 'cancelled');

DROP TABLE IF EXISTS collect_data_ports;
```

## 4. ScheduleRunner（调度执行器）

```python
# contexts/collect/infrastructure/adapter/runner/schedule_runner.py
from __future__ import annotations

import asyncio
import logging
import time
from datetime import datetime
from typing import Optional

from shared.infrastructure.cache.redis_cache import RedisCache
from contexts.collect.application.service.base import CollectResult, CollectService
from contexts.collect.application.service.registry import ServiceRegistry
from contexts.collect.infrastructure.persistence.repo.collect_run_log_repo import (
    CollectRunLogRepo,
)

logger = logging.getLogger(__name__)


class ScheduleRunner:
    """调度执行器：APScheduler 到点 → 查 service → 跑 → 写日志

    替代旧 CollectRunner.execute（state machine）：
      - 不再维护"running/success/failed" 状态机的 5 步流程
      - 不再写 collect_jobs 表的 status/started_at/finished_at
      - 只写 collect_run_log；service 内部异常被捕获、记录到 log.error
    """

    def __init__(
        self,
        services: ServiceRegistry,
        run_log_repo: CollectRunLogRepo,
    ) -> None:
        self._services = services
        self._repo = run_log_repo

    # ── 调度触发（APScheduler 回调）───────────────────────────────────

    async def fire_by_schedule_id(self, schedule_id: int) -> int:
        """APScheduler 到点调这个；返回 run_log_id"""
        from contexts.collect.infrastructure.persistence.repo.collect_schedule_repo import (
            CollectScheduleRepo,
        )
        sched_repo = CollectScheduleRepo()
        sched = await sched_repo.find_enabled(schedule_id)
        if sched is None:
            logger.warning("schedule_id=%s 不存在或已禁用，跳过", schedule_id)
            return -1
        return await self._run(sched, trigger_type="schedule")

    # ── HTTP 手动触发 ────────────────────────────────────────────────

    async def run_manual(
        self, service_name: str, params: dict, *,
        trigger_type: str = "api", schedule_id: Optional[int] = None,
    ) -> CollectResult:
        """HTTP 入口用：同步跑一次，返回 CollectResult + 写日志"""
        from contexts.collect.infrastructure.persistence.repo.collect_schedule_repo import (
            CollectScheduleEntity,
        )
        # 合成一个"伪 schedule"用于参数合并
        pseudo = CollectScheduleEntity(
            id=schedule_id or 0,
            service_name=service_name,
            params=params,
            concurrency=0,  # 0 表示"用 service 默认"
        )
        run_log_id = await self._run(pseudo, trigger_type=trigger_type, sync=True)
        # 同步模式下 result 已经在 run_log 里；再读一次
        log = await self._repo.find(run_log_id)
        return CollectResult(
            total=log.total_count,
            success=log.success_count,
            fail=log.fail_count,
            skip=log.skip_count,
            saved=log.saved_count,
            message=log.message or "",
        )

    # ── 内部：跑一次 + 写日志 ────────────────────────────────────────

    async def _run(
        self, sched, trigger_type: str, *, sync: bool = False,
    ) -> int:
        """执行：写启动日志 → 调 service → 写收尾日志"""
        service = self._services.require(sched.service_name)

        # 合并参数：DEFAULT_PARAMS < schedule.params < (HTTP body)
        merged_params = {**service.DEFAULT_PARAMS, **(sched.params or {})}
        concurrency = sched.concurrency or service.CONCURRENCY
        if concurrency != service.CONCURRENCY:
            merged_params["concurrency"] = concurrency

        started_at = datetime.now()
        log_id = await self._repo.create(
            schedule_id=sched.id or None,
            service_name=service.name,
            trigger_type=trigger_type,
            params=merged_params,
            started_at=started_at,
        )

        started_perf = time.perf_counter()
        try:
            logger.info(
                "schedule_runner: service=%s trigger=%s params=%s",
                service.name, trigger_type, merged_params,
            )
            result = await service.collect(merged_params)
            duration_ms = int((time.perf_counter() - started_perf) * 1000)
            await self._repo.finish_success(
                log_id,
                total=result.total,
                success=result.success,
                fail=result.fail,
                skip=result.skip,
                saved=result.saved,
                message=result.message,
                duration_ms=duration_ms,
            )
            logger.info(
                "schedule_runner: service=%s success total=%d fail=%d skip=%d saved=%d in %dms",
                service.name, result.total, result.fail, result.skip, result.saved, duration_ms,
            )
            return log_id
        except Exception as e:
            duration_ms = int((time.perf_counter() - started_perf) * 1000)
            logger.exception("schedule_runner: service=%s failed", service.name)
            await self._repo.finish_failed(
                log_id,
                error=repr(e),
                duration_ms=duration_ms,
            )
            if not sync:
                # 调度模式：仅记日志，不重抛（APScheduler 已记录异常）
                return log_id
            raise  # 同步模式：让 HTTP 入口看到异常


__all__ = ["ScheduleRunner"]
```

**关键点**：
- 调度模式（`sync=False`）：异常被吞掉，**只记日志**——APScheduler 会自己打 stack
- 同步模式（`sync=True`）：异常向上抛，**HTTP 入口能返回 500**
- **写日志和跑 service 是串行的**——不会出现"日志写了一半 service 跑完"的乱序

## 5. 第一个完整 Service 实现：DailyKlineService

```python
# contexts/collect/application/service/daily_kline_service.py
from __future__ import annotations

import asyncio
import logging
from datetime import date, timedelta
from typing import Optional

from shared.infrastructure.cache.redis_cache import RedisCache
from contexts.collect.application.service.base import CollectResult, CollectService
from contexts.collect.application.port.data.base import RateLimited
from contexts.collect.infrastructure.fetcher.tech.DailyKlineTushareFetcher import (
    DailyKlineTushareFetcher,
)
from contexts.collect.infrastructure.persistence.repo.kline_repo import KlineRepo
from contexts.collect.domain.entitys.stock_universe import StockUniverse

logger = logging.getLogger(__name__)


class DailyKlineService:
    """日 K 线业务 service

    触发场景：
      ① 盘后补齐：collect_schedule trigger_type=time times=['15:31']，每天 15:31 跑
      ② 用户手动：POST /collect/run/daily_kline body={"days": 7}
      ③ 实时查询：GET /klines/daily?symbol=000001.SZ → 调 get()，不走 collect()
    """

    name = "daily_kline"
    facet = "tech"
    sub_facet = "kline"
    label = "日 K 线"
    description = "5000+ 只股 OHLCV（不复权）"

    DEFAULT_PARAMS = {"days": 7, "source": "tushare"}
    RATE_LIMIT = ("tushare", 200, 60)         # 滑动窗口：200 次/60s
    CONCURRENCY = 5

    def __init__(
        self,
        fetcher: DailyKlineTushareFetcher,
        kline_repo: KlineRepo,
        cache: RedisCache,
        stock_universe: StockUniverse,
    ) -> None:
        self._fetcher = fetcher
        self._repo = kline_repo
        self._cache = cache
        self._universe = stock_universe

    # ── 采集入口（调度 + HTTP 触发）───────────────────────────────

    async def collect(self, params: dict) -> CollectResult:
        """统一入口

        params:
          - days: int (默认 7)
          - start_date / end_date: 'YYYY-MM-DD' / 'YYYYMMDD'（同时给定时优先）
          - source: str (默认 tushare)
          - mode: 'full' / 'condition' / None（None = 全量）
          - symbols: list[str] / None（None = 全 A）
          - concurrency: int (默认 self.CONCURRENCY)
        """
        params = {**self.DEFAULT_PARAMS, **params}
        start, end = self._parse_window(params)
        symbols = params.get("symbols") or await self._universe.list_symbols()
        symbols = [str(s).zfill(6) for s in symbols]

        if params.get("mode"):
            symbols = await self._filter_missing(symbols, start, end, params["mode"])

        return await self._collect_batch(symbols, start, end, params)

    # ── 实时查询入口（HTTP GET）────────────────────────────────────

    async def get(self, symbol: str, start: date, end: date) -> list[dict]:
        """GET /klines/daily → 走 repo（历史）+ fetcher+缓存（当日）

        返回 dict 列表（route 层 model_dump）
        """
        if end < date.today():
            rows = await self._repo.find(symbol, start, end)
            return [r.model_dump(mode="json") for r in rows]

        # 当日：fetcher + redis 缓存
        key = f"kline:daily:{symbol}:{start.isoformat()}:{end.isoformat()}"
        ttl = self._ttl_for_query(60)  # 交易时段 60s
        rows = await self._cache.get_or_set(
            key,
            lambda: self._fetcher.fetch(symbol=symbol, start=start, end=end),
            ttl,
        )
        return [r.model_dump(mode="json") for r in rows]

    # ── 内部：并发 + 限流 + 入库 ──────────────────────────────────

    async def _collect_batch(
        self, symbols: list[str], start: date, end: date, params: dict,
    ) -> CollectResult:
        if not symbols:
            return CollectResult(message="无单元")

        concurrency = int(params.get("concurrency", self.CONCURRENCY))
        sem = asyncio.Semaphore(concurrency)

        async def one(symbol: str) -> CollectResult:
            async with sem:
                return await self._collect_one(symbol, start, end, params)

        outcomes = await asyncio.gather(
            *[one(s) for s in symbols], return_exceptions=True,
        )
        return self._aggregate(symbols, outcomes)

    async def _collect_one(
        self, symbol: str, start: date, end: date, params: dict,
    ) -> CollectResult:
        # 1) 限流
        if self.RATE_LIMIT:
            allowed, _ = await self._cache.rate_limit(*self.RATE_LIMIT)
            if not allowed:
                raise RateLimited(retry_after=float(self.RATE_LIMIT[2]))

        # 2) 调 fetcher（带简单重试）
        klines = await self._fetch_with_retry(symbol, start, end, retries=2)

        # 3) 入库 / skip
        if not klines:
            return CollectResult(skip=1, message="empty")

        saved = await self._repo.save_batch(klines)
        return CollectResult(success=1, saved=saved, message=f"saved {saved}")

    async def _fetch_with_retry(
        self, symbol: str, start: date, end: date, retries: int,
    ):
        last_err: Optional[Exception] = None
        for attempt in range(retries + 1):
            try:
                return await self._fetcher.fetch(symbol=symbol, start=start, end=end)
            except RateLimited as e:
                last_err = e
                if attempt < retries:
                    await asyncio.sleep(e.retry_after)
                    continue
                raise
        raise last_err  # pragma: no cover

    # ── 内部：辅助 ────────────────────────────────────────────────

    def _parse_window(self, params: dict) -> tuple[date, date]:
        if params.get("start_date") and params.get("end_date"):
            return self._to_date(params["start_date"]), self._to_date(params["end_date"])
        days = int(params.get("days", 7))
        end = date.today()
        return end - timedelta(days=days - 1), end

    @staticmethod
    def _to_date(v) -> date:
        if isinstance(v, date):
            return v
        s = str(v).replace("-", "")
        return date(int(s[:4]), int(s[4:6]), int(s[6:8]))

    async def _filter_missing(
        self, symbols: list[str], start: date, end: date, mode: str,
    ) -> list[str]:
        """按库中已有数据过滤：full 模式从已覆盖的末日+1 开始；condition 模式已覆盖的跳过"""
        if mode == "full":
            req_start = date(2015, 1, 1)
            req_end = end
        else:
            req_start = start
            req_end = end

        coverage = await self._repo.get_coverage(symbols)
        list_dates = await self._universe.list_dates(symbols)
        pending = []
        for s in symbols:
            s_start = max(req_start, list_dates.get(s) or req_start)
            cov = coverage.get(s)
            if cov:
                cov_min, cov_max = cov
                if mode == "full" or cov_min <= s_start <= cov_max:
                    s_start = max(s_start, self._next_day(cov_max))
            if s_start <= req_end:
                pending.append(s)
        logger.info("daily_kline mode=%s: 共 %d, 待采 %d, 跳过 %d",
                    mode, len(symbols), len(pending), len(symbols) - len(pending))
        return pending

    @staticmethod
    def _next_day(d: date) -> date:
        return d + timedelta(days=1)

    def _ttl_for_query(self, trading_seconds: int) -> int:
        """复用 market_session.ttl_for（交易时段 short，盘后 long）"""
        from contexts.market.domain.services.market_session import ttl_for
        return ttl_for(trading_seconds)

    def _aggregate(
        self, symbols: list[str], outcomes: list,
    ) -> CollectResult:
        total = len(symbols)
        success = fail = skip = saved = 0
        for o in outcomes:
            if isinstance(o, BaseException):
                fail += 1
                logger.warning("日K 单元失败: %s", o)
            else:
                success += o.success
                fail += o.fail
                skip += o.skip
                saved += o.saved
        return CollectResult(
            total=total,
            success=success,
            fail=fail,
            skip=skip,
            saved=saved,
            message=f"日K: 共 {total} 只, 成功 {success} (无数据 {skip}), 失败 {fail}; 写入 {saved} 条",
        )


__all__ = ["DailyKlineService"]
```

```python
# contexts/collect/infrastructure/adapter/scheduler/apscheduler_adapter.py
"""SchedulerPort 的 APScheduler 实现（单进程，单 worker）

启动期：
  ① 读 collect_schedule 表所有 enabled=True 的行
  ② 调 schedule_runner 注入的 scheduler.upsert(key, schedule, payload)
  ③ payload = {"schedule_id": id}

触发时（on_fire）：
  ① 调 schedule_runner.fire_by_schedule_id(schedule_id)
"""
from __future__ import annotations

import logging
from datetime import datetime
from typing import Optional
from zoneinfo import ZoneInfo

from apscheduler.schedulers.asyncio import AsyncIOScheduler
from apscheduler.triggers.cron import CronTrigger
from apscheduler.triggers.interval import IntervalTrigger

from contexts.collect.domain.vo_collect_schedule import Schedule

logger = logging.getLogger(__name__)

TIME_MISFIRE_GRACE = 300
INTERVAL_MISFIRE_GRACE = 60


class ApschedulerAdapter:
    def __init__(self, timezone: str, on_fire) -> None:
        """
        on_fire: async def fire(payload: dict) -> None
                 payload = {"schedule_id": int}
        """
        self._tz = ZoneInfo(timezone)
        self._scheduler = AsyncIOScheduler(timezone=self._tz)
        self._on_fire = on_fire

    def start(self) -> None:
        self._scheduler.start()
        logger.info("调度器已启动: %d 个 job", len(self._scheduler.get_jobs()))

    def shutdown(self) -> None:
        if self._scheduler.running:
            self._scheduler.shutdown(wait=False)

    # ── 对外接口（供 ScheduleRunner 同步调度配置时调）────────

    def upsert(self, key: str, schedule: Schedule, payload: dict) -> None:
        self.remove(key)
        common = dict(args=[payload], replace_existing=True, max_instances=1, coalesce=True)
        if schedule.type == "time":
            for hhmm in schedule.times:
                hour, minute = (int(x) for x in hhmm.split(":"))
                self._scheduler.add_job(
                    self._fire,
                    CronTrigger(hour=hour, minute=minute, timezone=self._tz),
                    id=f"{key}@{hhmm}",
                    misfire_grace_time=TIME_MISFIRE_GRACE,
                    **common,
                )
        else:  # interval
            self._scheduler.add_job(
                self._fire,
                IntervalTrigger(seconds=schedule.interval_seconds),
                id=key,
                misfire_grace_time=INTERVAL_MISFIRE_GRACE,
                **common,
            )
        logger.info("注册调度: key=%s payload=%s", key, payload)

    def remove(self, key: str) -> None:
        for job in self._jobs(key):
            self._scheduler.remove_job(job.id)

    def _jobs(self, key: str):
        return [
            j for j in self._scheduler.get_jobs()
            if j.id == key or j.id.startswith(key + "@")
        ]

    async def _fire(self, payload: dict) -> None:
        """到点回调：转给 ScheduleRunner"""
        try:
            await self._on_fire(payload)
        except Exception:
            logger.exception("调度执行失败: payload=%s", payload)
```

## 10. 与原设计的对比

| 维度 | 旧（step6 收尾） | 新（step7） |
|:--|:--|:--|
| 抽象层 | 6 层（route / manage / runner / task / gateway+port+装饰器 / fetcher） | **3 层**（route / service / fetcher） |
| 调度入口 | `CollectManageService.submit` + `CollectJobService` + `CollectRunner` | **`ScheduleRunner.fire_by_schedule_id`** |
| 调度配置 | `collect_jobs`（一张表兼任配置+日志） | **`collect_schedule`**（配置）+ **`collect_run_log`**（日志） |
| 任务抽象 | `BaseCollectTask`（26 个子类） | **`CollectService` Protocol**（22 个实现） |
| 任务注册 | `CollectTaskRegistry`（内存 name → task） | **`ServiceRegistry`**（内存 name → service） |
| 限流 | `ThrottleDecorator`（跨 group 共享进程内 Semaphore） | **`RedisCache.rate_limit`**（service 内 1 行） |
| 重试 | `RetryDecorator`（指数退避 `UNAVAILABLE_BACKOFF`） | **`for attempt in retries: try/except → sleep → retry`**（service 内 6 行） |
| 统计 | `MetricsDecorator` + `read_stats` | **logger.info**——不写 Redis |
| 缓存 | `CacheDecorator`（实时类 Port 走 redis） | **`RedisCache.get_or_set`**（service 内 1 行） |
| 多源 | `DataGateway._select` 按 priority 循环 | **不存在**——`fetcher` 1:1 对应 `source` |
| 选源 | `source` 字段（缺省 = 优先级循环） | `source` 字段（**1:1 映射 fetcher**） |
| 抽象 | `XxxPort`（23 个 Generic）+ `XxxAdapter(XxxPort, AdapterBase)` | **`BaseFetcher`（ABC）**——子类直接 `async def fetch(**params) → list[BO]` |
| 错误处理 | `InvalidParams` / `RateLimited` / `SourceUnavailable` / `NotSupported` | **`RateLimited`**（其它错误直接抛） |
| 业务仓储调用 | task 调 service（间接） | service 调 fetcher + repo（直接） |
| HTTP 入口 | `POST /collect/jobs/{id}/run`（异步） | `POST /collect/run/{service_name}`（同步） |
| 进度跟踪 | `on_unit_done` 回调 + Redis 进度键 | **不存在**——只记最终结果 |
| 多 worker 部署 | `SELECT FOR UPDATE SKIP LOCKED` 抢 job | scheduler 独立进程 + HTTP worker N 个 |
