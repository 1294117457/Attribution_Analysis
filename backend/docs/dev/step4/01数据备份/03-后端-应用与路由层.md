```python
async def list_tables_with_stats(self) -> list[TableInfo]:
    """返回所有可备份表的元信息（行数 + 大小估算）"""
    exclude = BACKUP_EXCLUDE_TABLES
    result = []
    async with self.engine.connect() as conn:
        for table_name, table in Base.metadata.tables.items():
            if table_name in exclude:
                continue
            try:
                count = (await conn.execute(text(f'SELECT count(*) FROM "{table_name}"'))).scalar()
            except Exception:
                count = 0
            # 估算大小
            try:
                size_mb = (await conn.execute(text(
                    "SELECT pg_total_relation_size(:t) / 1024.0 / 1024.0"
                ), {"t": table_name})).scalar() or 0
            except Exception:
                size_mb = 0
            result.append(TableInfo(
                name=table_name,
                row_count=count,
                size_mb=round(size_mb, 2),
                category=_categorize_table(table_name),  # 业务/系统/采集
            ))
    return sorted(result, key=lambda t: t.category + t.name)
```

#### `create_backup()` — 创建任务

```python
async def create_backup(self, req: BackupRequest, user_id: int) -> int:
    """同步创建记录 + 提交后台任务；返回 record_id"""
    req.validate()

    # 校验路径（不实际写盘）
    _ = self.path_resolver.resolve_write(req.output_dir)

    # 决定实际表列表（scope=all 时展开）
    tables = req.tables
    tables_count: int
    if req.scope == "all":
        all_tables = [t for t in Base.metadata.tables.keys() if t not in BACKUP_EXCLUDE_TABLES]
        tables_snapshot = all_tables
        tables_count = len(all_tables)
    else:
        tables_snapshot = list(req.tables)
        tables_count = len(req.tables)

    output_dir = str(self.path_resolver.resolve_write(req.output_dir))

    # 插入记录
    async with self.engine.begin() as conn:
        record_id = (await conn.execute(text(
            """INSERT INTO sys_backup_records
               (name, backup_type, scope, tables, tables_count, output_dir,
                created_by, status)
               VALUES (:name, :bt, :scope, CAST(:tables AS jsonb), :tc, :od,
                       :uid, 'pending')
               RETURNING id"""
        ), {
            "name": req.name or "",
            "bt": req.backup_type,
            "scope": req.scope,
            "tables": json.dumps(tables_snapshot, ensure_ascii=False),
            "tc": tables_count,
            "od": output_dir,
            "uid": user_id,
        })).scalar()

    # 提交后台任务
    await self.task_manager.submit_backup(record_id, req)
    return record_id
```

#### `list_backups() / get_backup() / delete_backup()`

```python
async def list_backups(self, page: int, page_size: int, status: str | None) -> BackupListResult:
    offset = (page - 1) * page_size
    where = "WHERE status = :s" if status else ""
    params = {"limit": page_size, "offset": offset, **({"s": status} if status else {})}
    async with self.engine.connect() as conn:
        total = (await conn.execute(text(f"SELECT count(*) FROM sys_backup_records {where}"), params)).scalar()
        rows = (await conn.execute(text(
            f"""SELECT id, name, backup_type, scope, tables_count, output_dir, file_path,
                       file_size, row_count, progress, progress_msg, status, error_message,
                       started_at, finished_at, created_by, created_at
                FROM sys_backup_records {where}
                ORDER BY id DESC
                LIMIT :limit OFFSET :offset"""
        ), params)).mappings().all()
    return BackupListResult(total=total, items=[_to_dto(r) for r in rows])

async def get_backup(self, record_id: int) -> BackupDetailDTO:
    async with self.engine.connect() as conn:
        row = (await conn.execute(text(
            "SELECT * FROM sys_backup_records WHERE id = :id"
        ), {"id": record_id})).mappings().first()
    if not row:
        raise BackupNotFoundError(f"备份记录 #{record_id} 不存在")
    return _to_dto(row)

async def delete_backup(self, record_id: int) -> None:
    """删记录 + 删文件"""
    async with self.engine.begin() as conn:
        row = (await conn.execute(text(
            "SELECT file_path, status FROM sys_backup_records WHERE id = :id FOR UPDATE"
        ), {"id": record_id})).mappings().first()
        if not row:
            raise BackupNotFoundError(...)
        # 文件存在则删
        if row["file_path"]:
            p = Path(row["file_path"])
            if p.exists():
                p.unlink()
        await conn.execute(text("DELETE FROM sys_backup_records WHERE id = :id"), {"id": record_id})
```

```python
@router.post("/upload", response_model=dict)
async def upload_backup_file(
    file: UploadFile = File(...),
    _: int = Depends(require_permission("system:backup")),
    svc: BackupAppService = Depends(get_backup_service),
):
    """上传到 tmp 目录，返回 upload_id；前端再用 upload_id 调 /restore"""
    if not file.filename.endswith(".sql"):
        raise InvalidFileError("只接受 .sql 文件")
    # 校验大小（流式）
    settings = get_settings()
    upload_id = secrets.token_urlsafe(16)
    tmp_path = svc.path_resolver.resolve_write(None) / "tmp" / f"{upload_id}.sql"
    tmp_path.parent.mkdir(parents=True, exist_ok=True)
    size = 0
    async with aiofiles.open(tmp_path, "wb") as f:
        while chunk := await file.read(1024 * 1024):
            size += len(chunk)
            if size > settings.BACKUP_MAX_FILE_SIZE:
                tmp_path.unlink(missing_ok=True)
                raise FileTooLargeError(...)
            await f.write(chunk)
    # 头部校验
    sql_text = tmp_path.read_text(encoding="utf-8")
    header = parse_header(sql_text)
    _validate_header(header)
    return {
        "upload_id": upload_id,
        "tmp_path": str(tmp_path),
        "size": size,
        "header": header,
    }


# ── 8. 提交恢复 ──
@router.post("/restore", response_model=RestoreResponse, status_code=202)
async def restore(
    req: RestoreRequest,
    user_id: int = Depends(require_permission("system:backup")),
    svc: BackupAppService = Depends(get_backup_service),
):
    source_path = req.source_path
    if req.source_type == "upload":
        if not req.upload_id:
            raise ValueError("upload 模式必须带 upload_id")
        source_path = str(svc.path_resolver.resolve_write(None) / "tmp" / f"{req.upload_id}.sql")
    elif req.source_type == "history":
        if not req.source_backup_id:
            raise ValueError("history 模式必须带 source_backup_id")
        record = await svc.get_backup(req.source_backup_id)
        if not record.file_path or not Path(record.file_path).exists():
            raise FileNotFoundError("历史备份文件已不存在")
        source_path = record.file_path

    restore_id = await svc.restore(
        RestoreRequest(
            source_type=req.source_type,
            source_path=source_path,
            source_backup_id=req.source_backup_id,
            restore_mode=req.restore_mode,
            confirm=req.confirm,
            auto_pre_backup=req.auto_pre_backup,
        ),
        user_id=user_id,
    )
    return RestoreResponse(restore_id=restore_id, pre_backup_id=None)
```

## 5. 临时上传文件管理

```python
# 上传到 backend/backups/tmp/{upload_id}.sql
# 恢复完成后（或失败后）自动清理
# 防止 tmp 目录膨胀：启动时清理 > 24h 的临时文件

# task_manager 内
async def cleanup_tmp_files(self, ttl_hours: int = 24):
    tmp_dir = self.path_resolver.resolve_write(None) / "tmp"
    if not tmp_dir.exists():
        return
    now = time.time()
    for f in tmp_dir.glob("*.sql"):
        if now - f.stat().st_mtime > ttl_hours * 3600:
            f.unlink(missing_ok=True)
```

| `ConfirmRequiredError` | 400 | 必须 confirm=True |
| `TaskAlreadyRunningError` | 409 | 任务已在执行 |
| `BackupQueueFullError` | 429 | 同时只允许 1 个备份任务 |
| `FileTooLargeError` | 413 | 文件超过 BACKUP_MAX_FILE_SIZE |

---

## 7. 不做项

| 不做 | 原因 |
|---|---|
| 不做 OSS / S3 远程备份 | 业务不需要；v2 再做 |
| 不做增量备份 | 复杂度高；v2 再做 |
| 不做压缩 / 加密 | 交给 OS |
| 不做定时备份 | 业务可手动 + 操作系统 cron 兜底；v2 接入 APScheduler |
| 不做备份校验（备份后再导入测试）| 性能开销大；恢复时单事务会自然校验 schema |
| 不做"备份与采集并发"锁 | PG MVCC 保证读一致；恢复时用户主动触发，不并发 |
| 不做"备份文件在线预览" | 50MB+ 文件不适合网页预览 |
| 不做备份加密 | 业务不需要 |
| 不做备份传输到其他机器 | 业务不需要 |
| 不做备份对比 / diff | 复杂度高 |
