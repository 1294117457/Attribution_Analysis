async def _migrate_auth_system(conn):
    """执行 002_auth_system.sql（首次启动）"""
    from pathlib import Path
    sql_path = Path(__file__).parent.parent.parent / "migrations" / "002_auth_system.sql"
    if not sql_path.exists():
        logging.warning("002_auth_system.sql 不存在：%s", sql_path)
        return
    sql = sql_path.read_text(encoding="utf-8")
    for stmt in sql.split(";"):
        stmt = stmt.strip()
        if stmt:
            try:
                await conn.execute(text(stmt))
            except Exception as e:
                logging.warning("auth 迁移跳过: %s", e)

async def _ensure_initial_admin(conn):
    """如 sys_users 为空，创建 admin"""
    import bcrypt
    count = await conn.scalar(text("SELECT COUNT(*) FROM sys_users"))
    if count == 0:
        admin_pw_hash = bcrypt.hashpw(b"admin123", bcrypt.gensalt(12)).decode("utf-8")
        admin_id = await conn.scalar(text("""
            INSERT INTO sys_users
              (email, username, password_hash, nickname, is_active, is_verified)
            VALUES
              ('admin@local', 'admin', :hash, '系统管理员', TRUE, TRUE)
            RETURNING id
            """), {"hash": admin_pw_hash})
        await conn.execute(
            text("""
            INSERT INTO sys_user_roles (user_id, role_id)
            VALUES (:u, (SELECT id FROM sys_roles WHERE code='admin'))
            """),
            {"u": admin_id},
        )
        logging.warning("=" * 60)
        logging.warning("初始 admin 已创建")
        logging.warning("  email:    admin@local")
        logging.warning("  password: admin123")
        logging.warning("  请尽快登录并重置默认密码！")
        logging.warning("=" * 60)

# 在 lifespan 里加：
async with async_engine.begin() as conn:

## 3. 现有路由加鉴权（保守做法）

**当前阶段**：只把 `/auth/*`、`/users/*`、`/roles/*`、`/permissions/*` 加 `dependencies=[Depends(require_permission(...))]`。**业务路由（stock / pool / kline / collect / concept）保持公开**。

理由：
- 现有前端未登录态已经在用这些业务接口；
- 收紧前需要先让所有用户登录后才能用 → 业务体验断裂；
- 业务路由的鉴权收紧按需逐步推进（采集 → 操作池 → K 线 → 股票）。

> 每加一个 `dependencies=`，前端对应接口要么传 token（已自动），要么面对 403。**前端无需改业务代码**（axios 拦截器已自动带 token；403 由 ElMessage 提示）。

## 4. 验证清单（端到端）

### 4.1 后端 API 验证

```
□ POST /api/v1/auth/register
    body: {"email":"test1@xx.com","username":"test1","password":"test12345678"}
    expect: 201 + {"id":1, "email":"test1@xx.com", "username":"test1", ...}

□ POST /api/v1/auth/register  重复注册
    expect: 400 + USER_EXISTS

□ POST /api/v1/auth/login  admin
    body: {"email":"admin@local","password":"admin123"}
    expect: 200 + {access_token, refresh_token, expires_in:900, user_info:{roles:["admin"]}}

□ GET  /api/v1/auth/me
    Header: Authorization: Bearer <access_token>
    expect: 200 + user_info dict

□ POST /api/v1/auth/refresh
    body: {"refresh_token": "<refresh>"}
    expect: 200 + 新 access_token + 新 refresh_token（rotation）

□ POST /api/v1/auth/refresh  二次刷新同一旧 token
    expect: 401 + TOKEN_REUSE_DETECTED

□ POST /api/v1/auth/logout
    Header: Authorization: Bearer <access_token>
    expect: 200

□ GET  /api/v1/users/  用 admin
    expect: 200 + 列表

□ GET  /api/v1/users/  用 member
    expect: 403 + PERMISSION_DENIED

□ PATCH /api/v1/users/2/status  {"is_active": false}  用 admin
    expect: 200 + {is_active: false}
    # 验证：member 用户的所有 refresh_token 都被撤销（数据库查 sys_refresh_tokens）

□ POST /api/v1/users/2/reset-password  {"new_password": "newpass12345"}  用 admin
    expect: 200 + 该用户下次需重新登录

□ GET  /api/v1/roles/
    expect: 200 + 3 行（admin/member/viewer）

□ GET  /api/v1/permissions/
    expect: 200 + 14+ 行（按 resource 分组）
```

### 4.3 边界场景

```
□ 同 email 注册 → 400 USER_EXISTS

□ 同 username 注册 → 400 USER_EXISTS

□ 密码 7 位 → 400 AUTH_ERROR

□ 登录密码错 → 401 INVALID_CREDENTIALS

□ 登录被禁用户 → 403 USER_INACTIVE

□ 过期 refresh_token → 401 REFRESH_TOKEN_EXPIRED

□ 伪造 token（任意字符串） → 401 TOKEN_INVALID

```

## 7. 常见问题 & 注意事项

### 7.1 常见坑

| 坑 | 原因 | 解决 |
| --- | --- | --- |
| `RuntimeError: Form data requires "python-multipart"` | FastAPI 接收 form 表单需 multipart | 当前用 JSON 请求，无需装 |
| `cryptography` 装不上 | Windows 编译错误 | 用预编译 wheel：`pip install cryptography`（自动选）|
| `bcrypt` 装不上 | 同上 | 同上 |
| 私钥文件被提交 | 没加 `.gitignore` | 加 `backend/.keys/` |
| 401 refresh 死循环 | refresh_token 也过期 → 拦截器再调 refresh 失败 | 拦截器 catch 到 refresh 错误 → 清 token + 跳登录 |
| 前端 dev proxy 失败 | vite.config.ts 没配 `/api` | 配 `server.proxy['/api']` → 后端 8000 |

### 7.2 安全建议

| 建议 | Why |
|---|---|
| **生产前必须改初始 admin 密码** | admin123 是默认密码 |
| HTTPS（生产） | 防 token 中间人 |
| CORS 收紧 | 开发允许 `localhost:*`，生产收紧到实际域名 |
| 登录限流（当前不实现） | 防止爆破 |
| Token rotation 依赖数据库 | 性能可接受 |

### 7.3 性能影响

| 场景 | 性能开销 |
|---|---|
| 每次 API 请求解析 token（公钥验签）| RSA 2048 验签 ≈ 0.5ms，可接受 |
| `_get_user_permissions` 每次 refresh 都查 DB | 可接受 |
| bcrypt 登录验证 | cost=12 ≈ 250ms，登录接口可接受 |
| access_token 内嵌 permissions | 每次请求无需查 DB；token 略大（< 2KB）|
