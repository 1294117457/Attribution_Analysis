### 2.2 改为

```typescript
// 改动后
import { createRouter, createWebHistory } from 'vue-router'
import type { RouteRecordRaw } from 'vue-router'
import { useAuthStore } from '@/stores/auth'

import homeRoutes from './home'
import authRoutes from './auth-routes'

const routes: RouteRecordRaw[] = [
  { path: '/', redirect: '/home/index' },
  authRoutes,           // ← 新增（顺序：在 home 前，404 在最后）
  homeRoutes,
  { path: '/:pathMatch(.*)*', component: () => import('@/views/NotFound.vue') },
]

const router = createRouter({
  history: createWebHistory(),
  routes,
})

// ── 路由守卫：登录拦截 ──
router.beforeEach(async (to, from, next) => {
  const authStore = useAuthStore()

  // 已登录但 userInfo 为空 → 拉一次（页面刷新后）
  if (authStore.isLoggedIn && !authStore.userInfo) {
    try {
      await authStore.fetchMe()
    } catch {
      authStore.clear()
    }
  }

  const isPublic = to.path === '/login'

  if (!authStore.isLoggedIn && !isPublic) {
    // 未登录访问受保护路由 → 跳登录
    next('/login')
  } else if (authStore.isLoggedIn && isPublic) {
    // 已登录访问 /login → 跳首页
    next('/home/index')
  } else {
    next()
  }
})

export default router
```

### 2.3 设计要点

| 项 | Why |
|---|---|
| 路由守卫 `beforeEach` | 全局拦截 |
| 已登录但 userInfo 为空 → 自动 fetchMe | 页面刷新后状态恢复 |
| 401 已在 axios 拦截器处理 → 这里不重复 | 单点责任 |
| `isPublic` 仅 `'/login'` | 注册功能在 LoginPage 内部 |
| 跳首页写死 `/home/index` | 与现有 redirect 一致 |

需要：

```typescript
import { computed } from 'vue'
import { useAuthStore } from '@/stores/auth'
import { UserFilled } from '@element-plus/icons-vue'

const authStore = useAuthStore()

const menuItems = computed<MenuItem[]>(() => {
  const items: MenuItem[] = [
    { path: '/home/index',         title: '数据大盘',   icon: Odometer },
    { path: '/home/stock-panel',   title: '股票信息',   icon: DataLine },

    items.push({ path: '/home/account', title: '账户管理', icon: UserFilled })
  }
  return items
})
```

### 5.2 改为（动态 + 加登出）

```vue
<!-- 改动后 -->
<template>
  <header class="topbar">
    <div class="topbar-brand">
      <span class="brand-icon">📈</span>
      <span class="brand-text">智能金融归因分析平台</span>
    </div>
    <div class="topbar-right">
      <el-dropdown trigger="click" @command="handleCommand">
        <div class="user-info">
          <div class="user-avatar">{{ avatarText }}</div>
          <span class="user-name">{{ authStore.userInfo?.nickname || authStore.userInfo?.username || '未登录' }}</span>
          <el-icon class="user-arrow"><ArrowDown /></el-icon>
        </div>
        <template #dropdown>
          <el-dropdown-menu>
            <el-dropdown-item command="profile">个人设置</el-dropdown-item>
            <el-dropdown-item command="logout" divided>退出登录</el-dropdown-item>
          </el-dropdown-menu>
        </template>
      </el-dropdown>
    </div>
  </header>
</template>

<script setup lang="ts">
import { computed } from 'vue'
import { useRouter } from 'vue-router'
import { ArrowDown } from '@element-plus/icons-vue'
import { ElMessage } from 'element-plus'
import { useAuthStore } from '@/stores/auth'

const router = useRouter()
const authStore = useAuthStore()

const avatarText = computed(() => {
  const nick = authStore.userInfo?.nickname || authStore.userInfo?.username
  return nick ? nick.charAt(0).toUpperCase() : '?'
})

async function handleCommand(cmd: string) {
  if (cmd === 'logout') {
    await authStore.logout()
    ElMessage.success('已退出登录')
    router.push('/login')
  } else if (cmd === 'profile') {
    router.push('/home/account')
  }
}
</script>
```

### 5.3 关键点

| 项 | Why |
|---|---|
| 用户名从 store 读 | 不再硬编码"管理员" |
| 头像首字母（`.toUpperCase()`） | "admin" → "A"、"小" → "小" |
| "个人设置" 跳 `/home/account` | 简化：暂未做单独 profile 页 |
| "退出登录" 调 `authStore.logout()` | 自动清 token + 跳 /login |
| `@command` 统一处理 | 菜单命令分发 |
