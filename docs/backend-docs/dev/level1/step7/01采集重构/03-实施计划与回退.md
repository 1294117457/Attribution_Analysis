# 03 · 实施计划与回退方案

> 本文给出**4 周分阶段实施计划**，每阶段独立可验证可回退。同时给出**回退方案**（每阶段独立 commit + 数据迁移可逆 + 新表幂等创建）。
>
> 实施原则：
> 1. **新代码先行**——建新表 / 新 service / 新 runner / 新路由，**不动旧代码**
> 2. **双跑期**——新旧并行跑 1-2 天，验证新 service 输出与旧 task 一致
> 3. **切流**——验证通过后切流量（HTTP 入口指向新 service；scheduler 指向新 runner）
> 4. **删旧**——观察 1 周无问题后删旧代码 + 旧表

### 3.2 `StockBasicService` 模板（参考 `DailyKlineService`）

```python
class StockBasicService:
    name = "stock_basic"
    facet = "fundamental"
    sub_facet = "stock"
    label = "股票基本信息"
    description = "全 A 股基本信息"

    DEFAULT_PARAMS = {"list_status": "L", "source": "tushare"}
    RATE_LIMIT = ("tushare", 200, 60)
    CONCURRENCY = 1   # 全 A 一次拉完，不需要并发

    def __init__(self, fetcher, stock_repo, cache):
        self._fetcher = fetcher
        self._repo = stock_repo
        self._cache = cache

    async def collect(self, params: dict) -> CollectResult:
        params = {**self.DEFAULT_PARAMS, **params}
        list_status = params.get("list_status", "L")

        # 限流
        allowed, _ = await self._cache.rate_limit(*self.RATE_LIMIT)
        if not allowed:
            raise RateLimited(retry_after=60.0)

        # 拉数据
        try:
            stocks = await self._fetcher.fetch(list_status=list_status)
        except RateLimited as e:
            await asyncio.sleep(e.retry_after)
            stocks = await self._fetcher.fetch(list_status=list_status)

        if not stocks:
            return CollectResult(skip=1, message="empty")

        # 入库
        saved = await self._repo.save_batch(stocks)
        return CollectResult(success=1, saved=saved, message=f"saved {saved}")

    async def get(self, ts_code: str) -> list[dict]:
        # 走 repo
        rows = await self._repo.find(ts_code)
        return [r.model_dump(mode="json") for r in rows]
```

# main.py lifespan（切流后）
@asynccontextmanager
async def lifespan(app: FastAPI):
    await init_redis()
    await init_db()

    # ── 新流程（启用） ──
    services = setup_services()
    set_service_registry(services)
    run_log_repo = CollectRunLogRepo()
    schedule_repo = CollectScheduleRepo()
    runner = ScheduleRunner(services, run_log_repo)

    async def on_fire(payload):
        await runner.fire_by_schedule_id(payload["schedule_id"])

    scheduler = ApschedulerAdapter(timezone="Asia/Shanghai", on_fire=on_fire)
    for sched in await schedule_repo.list_enabled():
        sched_vo = build_schedule(
            mode=sched.trigger_type, schedule_type=sched.trigger_type,
            times=sched.times, interval_seconds=sched.interval_seconds,
        )
        if sched_vo is not None:
            scheduler.upsert(f"sched:{sched.id}", sched_vo, {"schedule_id": sched.id})
    scheduler.start()

    app.state.services = services
    app.state.runner = runner
    app.state.scheduler = scheduler

    # ── 旧流程（仍跑 1 天用于对比） ──
    # 注意：旧 scheduler 仍会写 collect_jobs 表
    setup_default_registry()
    await load_port_catalog()
    ctx = build_collect_context()
    setup_collect(ctx, all_collect_tasks(ctx))
    await _sync_fetcher_catalog()
    old_scheduler = ApschedulerAdapter(settings.COLLECT_SCHEDULER_TIMEZONE)
    set_scheduler(old_scheduler)
    old_scheduler.on_fire(lambda p: get_collect_manage_service().handle_job(p))
    await get_collect_manage_service().restore_schedules()
    old_scheduler.start()
    app.state.old_scheduler = old_scheduler
    app.state.old_collect_manage = get_collect_manage_service()

    yield

    old_scheduler.shutdown()
    scheduler.shutdown()
```

# scripts/migrate_step7_data.py
"""从 collect_jobs 迁移历史数据到 collect_schedule + collect_run_log"""
import asyncio
from sqlalchemy import text
from shared.infrastructure.persistence.connection import async_engine


async def main():
    async with async_engine.begin() as conn:
        # 1) 迁移调度配置
        await conn.execute(text("""
            INSERT INTO collect_schedule (
                service_name, enabled, trigger_type, times, interval_seconds,
                params, concurrency, display_name, created_at, updated_at
            )
            SELECT
                task_type,
                enabled,
                COALESCE(schedule_type, 'time'),
                COALESCE(times, '{}'),
                interval_seconds,
                COALESCE(params, '{}'),
                concurrency,
                display_name,
                created_at,
                updated_at
            FROM collect_jobs
            WHERE mode IN ('time', 'interval')
        """))

        # 2) 迁移运行历史
        await conn.execute(text("""
            INSERT INTO collect_run_log (
                schedule_id, service_name, trigger_type, params,
                started_at, finished_at, duration_ms, status,
                total_count, success_count, fail_count, skip_count, saved_count,
                message, created_at
            )
            SELECT
                NULL,
                task_type,
                trigger_type,
                COALESCE(params, '{}'),
                started_at,
                finished_at,
                duration_ms,
                status,
                total_count,
                success_count,
                fail_count,
                skip_count,
                0,
                message,
                created_at
            FROM collect_jobs
            WHERE status IN ('success', 'failed', 'cancelled')
        """))

        sched_count = (await conn.execute(text("SELECT COUNT(*) FROM collect_schedule"))).scalar()
        log_count = (await conn.execute(text("SELECT COUNT(*) FROM collect_run_log"))).scalar()
        print(f"Migrated: {sched_count} schedules, {log_count} run logs")


if __name__ == "__main__":
    asyncio.run(main())
```
