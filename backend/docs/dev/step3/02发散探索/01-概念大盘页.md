## 1. 业务需求

| 维度 | 说明 |
|---|---|
| 场景 | 用户进首页 / 主动访问 `/home/concepts` 快速浏览「当前 A 股哪些概念在涨/跌」 |
| 核心功能 | ① 涨幅榜 / 跌幅榜（左红右绿双栏 Top 20） ② 概念卡片网格（按类型分组 + 涨幅色块） ③ 点击概念 → 抽屉显示成分股列表（含最新价/涨跌幅/换手） |

| 排序 | 涨幅榜按 `pct_change DESC`、跌幅榜按 `pct_change ASC`、卡片网格按类型内涨幅降序 |
| 过滤 | 概念类型 `concept_type`（industry / theme / style / region / event / other）单选或全选 |

### 3.5 仓储实现（**只在文件末尾追加**）

```python
# backend/src/infrastructure/persistence/repositories/concept_repository.py
# 末尾追加 1 个 dataclass + 2 个方法

# v2 新增：私有 dataclass（放实现层，不进 domain）
from dataclasses import dataclass

@dataclass
class _ConceptBoardRow:
    """概念大盘行（仓储私有 VO — 不进 domain/vo/，因带实时字段）"""
    concept_id: int
    index_code: str
    name: str
    source: str
    concept_type: str
    description: Optional[str]
    stock_count: int
    is_active: bool


async def list_board_rows(
    self,
    type_filter: Optional[str] = None,
    sort_by: str = "pct_change",
    order: str = "desc",
    page: int = 1,
    page_size: int = 50,
) -> tuple[list[ConceptBoardRow], int]:
    """按 type 筛选 + 排序 + 分页
    注：sort_by="pct_change" 时只按 stock_count DESC 占位（实时字段由 service 补）
    """
    # 1) 构造基础 query
    stmt = select(
        Concept.id, Concept.index_code, Concept.name, Concept.source,
        Concept.concept_type, Concept.description, Concept.stock_count,
        Concept.is_active,
    ).where(Concept.is_active == True)
    if type_filter and type_filter != "all":
        stmt = stmt.where(Concept.concept_type == type_filter)

    # 2) 排序（pct_change 用 stock_count 兜底）
    sort_col_map = {
        "pct_change": Concept.stock_count,
        "stock_count": Concept.stock_count,
        "name": Concept.name,
    }
    sort_col = sort_col_map.get(sort_by, Concept.stock_count)
    stmt = stmt.order_by(sort_col.desc() if order == "desc" else sort_col.asc())

    # 3) 分页
    total = await self._session.scalar(
        select(func.count()).select_from(stmt.subquery())
    )
    stmt = stmt.offset((page - 1) * page_size).limit(page_size)
    result = await self._session.execute(stmt)
    rows = result.all()

    # 4) 映射（v2: 用 _ConceptBoardRow 私有 dataclass）
    return [
        _ConceptBoardRow(
            concept_id=r.id, index_code=r.index_code, name=r.name,
            source=r.source.value if hasattr(r.source, "value") else r.source,
            concept_type=r.concept_type.value if hasattr(r.concept_type, "value") else r.concept_type,
            description=r.description, stock_count=r.stock_count,
            is_active=r.is_active,
        )
        for r in rows
    ], (total or 0)


async def list_members_by_concept(
    self,
    concept_id: int,
    sort_by: str = "pct_change",
    order: str = "desc",
    page: int = 1,
    page_size: int = 50,
) -> tuple[list[dict], int]:
    """单概念成分股：stock_concept_members ⨝ stock_infos ⨝ fin_daily_basics(最新一行)
    返回 dict 列表（路由层组装为 ConceptMemberItemVO）
    """
    # 1) 子查询：每只股票 fin_daily_basics 最新一行
    latest_fdb = (
        select(
            FinDailyBasic.symbol,
            func.max(FinDailyBasic.trade_date).label("max_date"),
        )
        .group_by(FinDailyBasic.symbol)
        .subquery()
    )

    # 2) 主查询
    stmt = (
        select(
            StockInfo.symbol, StockInfo.name, StockInfo.industry,
            StockInfo.market, StockInfo.exchange,
            FinDailyBasic.close.label("latest_close"),
            FinDailyBasic.total_mv, FinDailyBasic.pe_ttm,
        )
        .select_from(StockConceptMember)
        .join(StockInfo, StockConceptMember.symbol == StockInfo.symbol)
        .outerjoin(
            latest_fdb,
            latest_fdb.c.symbol == StockInfo.symbol,
        )
        .outerjoin(
            FinDailyBasic,
            and_(
                FinDailyBasic.symbol == latest_fdb.c.symbol,
                FinDailyBasic.trade_date == latest_fdb.c.max_date,
            ),
        )
        .where(StockConceptMember.concept_id == concept_id)
    )

    # 3) 排序
    sort_col_map = {
        "pct_change": FinDailyBasic.close,    # 兜底：按收盘价
        "latest_close": FinDailyBasic.close,
        "total_mv": FinDailyBasic.total_mv,
        "name": StockInfo.name,
    }
    sort_col = sort_col_map.get(sort_by, FinDailyBasic.close)
    stmt = stmt.order_by(sort_col.desc(nulls_last=True) if order == "desc"
                         else sort_col.asc(nulls_first=True))

    # 4) 分页 + 返回 dict
    total = await self._session.scalar(
        select(func.count()).select_from(stmt.subquery())
    )
    stmt = stmt.offset((page - 1) * page_size).limit(page_size)
    result = await self._session.execute(stmt)
    rows = result.mappings().all()
    return [dict(r) for r in rows], (total or 0)
```

# backend/src/application/service/concept_app_service.py
# 末尾追加 2 个方法

class ConceptAppService:
    # ... 现有方法不动 ...

    # ═══ 概念大盘（新增） ═══

    async def query_board(
        self, req: ConceptBoardQueryRequest
    ) -> ConceptBoardListVO:
        """概念大盘：分页 + 实时行情补齐 + 排序
        性能：1 DB 查询（分页）+ 1 实时接口批量调用（page_size≤500 时一次性 query_many）
        """
        rows, total = await self._repo.list_board_rows(
            type_filter=req.type_filter,
            sort_by=req.sort_by,
            order=req.order,
            page=req.page,
            page_size=req.page_size,
        )

        # 实时行情批量补齐
        index_codes = [r.index_code for r in rows]
        names = {r.index_code: r.name for r in rows}
        if self._realtime and index_codes:
            quotes = await self.get_quotes(index_codes, names=names)
        else:
            quotes = {}

        # 组装 VO（按 req.sort_by + order 重排，因为有 pct_change 之后才能真排序）
        items_dicts = []
        for r in rows:
            q = quotes.get(r.index_code, {})
            items_dicts.append({
                "concept_id": r.concept_id,
                "index_code": r.index_code,
                "name": r.name,
                "source": r.source,
                "concept_type": r.concept_type,
                "concept_type_label": CONCEPT_BOARD_TYPE_LABELS.get(r.concept_type, r.concept_type),
                # v2: 私有 dataclass 用属性访问；如仍 dict 则 r["stock_count"]
                "stock_count": getattr(r, "stock_count", r.get("stock_count", 0)),
                "description": getattr(r, "description", r.get("description")),
                "price": q.get("price"),
                "prev_close": q.get("prev_close"),
                "change": q.get("change"),
                "pct_change": q.get("pct_change"),
                "color": q.get("color", "flat"),
                "trade_time": q.get("trade_time"),
                "captured_at": q.get("captured_at"),
                "stale": q.get("stale", False),
            })

        # 若 sort_by="pct_change"，用内存排序（实时数据到位后）
        if req.sort_by == "pct_change":
            items_dicts.sort(
                key=lambda x: (x["pct_change"] is None,
                              -(x["pct_change"] or 0) if req.order == "desc"
                              else (x["pct_change"] or 0))
            )

        items = [ConceptBoardItemVO(**d) for d in items_dicts]
        return ConceptBoardListVO.from_list(items, total, req.page, req.page_size)

    async def query_board_members(
        self, req: ConceptMembersQueryRequest
    ) -> ConceptMemberListVO:
        """概念成分股：分页 + 池信息 + 实时涨幅（仅前 50 只调实时）
        性能：1 DB 查询（分页 + 池反查） + 至多 50 次实时查询
        """
        rows, total = await self._repo.list_members_by_concept(
            concept_id=req.concept_id,
            sort_by=req.sort_by,
            order=req.order,
            page=req.page,
            page_size=req.page_size,
        )

        # 池信息（避免 N+1）
        symbols = [r["symbol"] for r in rows]
        pool_map: dict[str, list[PoolMembershipVO]] = {}
        if req.with_pools and symbols:
            pool_map = await self._repo.list_membership_by_symbols(symbols)

        # 实时涨幅（仅前 50 只 — 避免 500 次 pytdx 调用拖死源站）
        # 注：使用 stock_minute_kline 实时接口，从返回 data.last_close / pct_change 解析
        realtime_map: dict[str, dict] = {}
        if self._realtime and symbols and len(symbols) <= 50:
            results = await self._realtime.query_many(
                "stock_minute_kline",
                [{"symbol": s, "interval": "1d"} for s in symbols],
            )
            for s, res in zip(symbols, results):
                if res and res.data and not res.stale:
                    d = res.data
                    realtime_map[s] = {
                        "pct_change": d.get("pct_change"),
                        "color": "up" if (d.get("pct_change") or 0) > 0
                                 else "down" if (d.get("pct_change") or 0) < 0
                                 else "flat",
                    }

        # 组装
        items = [
            ConceptMemberItemVO(
                symbol=r["symbol"],
                name=r.get("name"),
                industry=r.get("industry"),
                market=r.get("market"),
                latest_close=r.get("latest_close"),
                total_mv=r.get("total_mv"),
                pe_ttm=r.get("pe_ttm"),
                pct_change=realtime_map.get(r["symbol"], {}).get("pct_change"),
                color=realtime_map.get(r["symbol"], {}).get("color", "flat"),
                pools=pool_map.get(r["symbol"], []),
            )
            for r in rows
        ]

        # 若按 pct_change 排序，内存重排
        if req.sort_by == "pct_change":
            items.sort(
                key=lambda x: (x.pct_change is None,
                              -(x.pct_change or 0) if req.order == "desc"
                              else (x.pct_change or 0))
            )

        return ConceptMemberListVO.from_list(items, total, req.page, req.page_size)
```

      with_pools: true, page: page.value, page_size: pageSize.value,
    })
    rows.value = res.items
    total.value = res.total
  } finally { loading.value = false }
}

watch(() => props.concept.concept_id, () => { page.value = 1; load() }, { immediate: true })
watch([page, pageSize], load)
</script>
```

## 5. 性能与边界

| 维度 | 数值 | 备注 |
|---|---|---|
| 主页首次加载 | ~500ms | 1 DB 分页 + 1 实时接口 `query_many`（500 个） |
| 实时刷新（15s） | ~200ms | 命中 Redis 缓存，极少穿透 |
| 成分股首次 | ~300ms | 1 DB + ≤50 次实时 |
| 成分股实时 | 不轮询 | 抽屉打开时才查，不主动刷新 |
| 最大数据集 | 500 概念 | 受 `page_size=500` 限制 |
| 弱网降级 | `stale=true` | 概念行显示 `stale=true` + 灰色 "行情源暂不可用" |

## 8. 已知风险与边界

| 风险 | 应对 |
|---|---|
| 概念数 > 500 时 `page_size=500` 截断 | 前端明确 `page_size ≤ 500`；超过时分页 |
| 同花顺实时接口不可用 | `stale=true` 兜底用日 K 收盘 |
| 成分股 50 只以外没有实时涨幅 | 文档明示；后续可考虑"前 50 优先" |
| 菜单组「实验性」位置 | 显式挂在最末尾；用户随时可反馈"移上去/改组名" |
| `_ConceptBoardRow` 是否需要进 domain | **不进**（v2 矫正）；仅在仓储实现层 |
| `import CONCEPT_TYPE_LABELS from concept` 循环依赖 | 实测无循环（`concept.py` 不 import `concept_board.py`） |
| 路由路径 `/board` vs `/concept-board` | 改用 **`/concept-board`**（因 `/{name}` 通配符会吞掉 `/board`），实测可避让 |
| vite proxy 指向 | v2 实施期间，**vite.config.ts 临时改指 `localhost:8001`**（8000 旧后端无法 kill） |
