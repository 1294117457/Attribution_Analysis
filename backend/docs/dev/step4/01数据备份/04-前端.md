# 04 数据备份与导入 — 前端

> 范围：侧边栏菜单、API 封装（Pinia store）、备份向导页、恢复向导页、路径配置
>
> **本文档严格遵循 [`frontend/docs/page-development-guide.md`](../../../../frontend/docs/page-development-guide.md) 规范**：
> - 所有页面使用 `<PageWrapper :style="densityStyle" class="bkup-page">` 包裹
> - 4 个密度锚点（page / top / middle / bottom）+ 4 个滑块 ⚙️ 面板
> - 表格列宽用 `colW(px)` 函数
> - 子组件 class 加 `bkup-` 前缀避免污染
> - 命名：备份模块 = `bkup-`（bkup = backup 缩写）

## 3. 路由配置

### 3.1 路由结构（嵌套 + 独立子组件）

```
/backup/                 →  备份主页（父布局：标题 + 工具栏 + 切换子页签）
   ├── /backup/create/   →  创建备份子页（向导）
```

```typescript
import request from "@/utils/request"

export interface TableInfo {
  name: string
  row_count: number
  size_mb: number
  category: "业务" | "系统" | "采集"
}

export interface BackupRecord {
  id: number
  name: string
  backup_type: "full" | "schema" | "data"
  scope: "all" | "partial"
  tables: string[]
  tables_count: number
  output_dir: string
  file_path: string | null
  file_size: number | null
  row_count: number
  progress: number
  progress_msg: string
  status: "pending" | "running" | "success" | "failed" | "interrupted"
  error_message: string | null
  started_at: string | null
  finished_at: string | null
  duration_sec: number | null
  created_at: string
}

export interface BackupConfig {
  default_output_dir: string
  allowed_roots: string[]
  max_file_size: number
}

// 1. 列出所有表
export const listTables = () =>
  request.get<TableInfo[]>("/api/v1/backups/tables")

// 2. 创建备份
export const createBackup = (payload: {
  backup_type: "full" | "schema" | "data"
  scope: "all" | "partial"
  tables: string[]
  output_dir: string | null
  name: string
}) => request.post<BackupRecord>("/api/v1/backups/", payload)

// 3. 列表
export const listBackups = (params: {
  page: number
  page_size: number
  status?: string
}) => request.get<{ total: number; items: BackupRecord[] }>(
  "/api/v1/backups/",
  { params }
)

// 4. 详情
export const getBackup = (id: number) =>
  request.get<BackupRecord>(`/api/v1/backups/${id}`)

// 5. 下载（注意：返回 URL，不发起请求）
export const downloadBackupUrl = (id: number) =>
  `${import.meta.env.VITE_API_BASE}/api/v1/backups/${id}/download`

// 6. 删除
export const deleteBackup = (id: number) =>
  request.delete(`/api/v1/backups/${id}`)

// 7. 上传
export const uploadBackupFile = (file: File) => {
  const form = new FormData()
  form.append("file", file)
  return request.post<{
    upload_id: string
    tmp_path: string
    size: number
    header: Record<string, string>
  }>("/api/v1/backups/upload", form, {
    headers: { "Content-Type": "multipart/form-data" },
  })
}

// 8. 恢复
export const restore = (payload: {
  source_type: "file" | "upload" | "history"
  source_path?: string
  source_backup_id?: number
  upload_id?: string
  restore_mode: "cover" | "upsert"
  confirm: boolean
}) => request.post<{ restore_id: number; pre_backup_id: number | null }>(
  "/api/v1/backups/restore",
  payload
)

// 9. 恢复历史
export const listRestores = (params: { page: number; page_size: number }) =>
  request.get("/api/v1/backups/restore", { params })

// 10. 读配置
export const getBackupConfig = () =>
  request.get<BackupConfig>("/api/v1/backups/config")

// 11. 改配置
export const updateBackupConfig = (output_dir: string) =>
  request.put("/api/v1/backups/config", null, { params: { output_dir } })
```

## 5. Pinia Store

### 5.1 `stores/backup.ts`

```typescript
// stores/backup.ts
import { defineStore } from "pinia"
import { ref, computed } from "vue"
import {
  listBackups,
  getBackup,
  type BackupRecord,
} from "@/api/backup"

export const useBackupStore = defineStore("backup", () => {
  // ── 路径配置（持久化到 localStorage）─────────────────
  const pathConfig = ref({
    outputDir: localStorage.getItem("backup.outputDir") || "",
    inputDir: localStorage.getItem("backup.inputDir") || "",
  })

  function setOutputDir(dir: string) {
    pathConfig.value.outputDir = dir
    localStorage.setItem("backup.outputDir", dir)
  }
  function setInputDir(dir: string) {
    pathConfig.value.inputDir = dir
    localStorage.setItem("backup.inputDir", dir)
  }

  // ── 当前任务进度轮询 ────────────────────────────────
  const pollingTasks = ref<Map<number, number>>() // record_id → interval_id

  async function startPolling(id: number, onUpdate: (r: BackupRecord) => void) {
    stopPolling(id)
    const interval = setInterval(async () => {
      try {
        const record = await getBackup(id)
        onUpdate(record)
        if (["success", "failed", "interrupted"].includes(record.status)) {
          stopPolling(id)
        }
      } catch (e) {
        stopPolling(id)
      }
    }, 2000) as unknown as number
    pollingTasks.value?.set(id, interval)
  }

  function stopPolling(id: number) {
    const interval = pollingTasks.value?.get(id)
    if (interval) {
      clearInterval(interval)
      pollingTasks.value?.delete(id)
    }
  }

  return {
    pathConfig,
    setOutputDir,
    setInputDir,
    startPolling,
    stopPolling,
  }
})
```

### 6.2 `composables/useBackupDensity.ts` — 复用密度逻辑

```typescript
// composables/useBackupDensity.ts
import { ref, computed, watch } from "vue"

type DensityShape = { page: number; top: number; middle: number; bottom: number }

const DENSITY_KEY = "bkup-density"  // 备份模块专用 key

function clamp(v: number) {
  const n = Math.max(0.6, Math.min(1.5, Number.isFinite(v) ? v : 1))
  return Math.round(n * 100) / 100
}

function loadDensity(): DensityShape {
  try {
    const raw = localStorage.getItem(DENSITY_KEY)
    if (!raw) return { page: 1, top: 1, middle: 1, bottom: 1 }
    const p = JSON.parse(raw) as Partial<DensityShape>
    return {
      page:   clamp(Number(p.page   ?? 1)),
      top:    clamp(Number(p.top    ?? 1)),
      middle: clamp(Number(p.middle ?? 1)),
      bottom: clamp(Number(p.bottom ?? 1)),
    }
  } catch { return { page: 1, top: 1, middle: 1, bottom: 1 } }
}

export function useBackupDensity() {
  const init = loadDensity()
  const pageDensity   = ref<number>(init.page)
  const topDensity    = ref<number | null>(init.top    === 1 ? null : init.top)
  const middleDensity = ref<number | null>(init.middle === 1 ? null : init.middle)
  const bottomDensity = ref<number | null>(init.bottom === 1 ? null : init.bottom)

  const densityStyle = computed<Record<string, string>>(() => {
    const s: Record<string, string> = { "--bkup-page-density": String(pageDensity.value) }
    if (topDensity.value    != null) s["--bkup-top-density"]    = String(topDensity.value)
    if (middleDensity.value != null) s["--bkup-middle-density"] = String(middleDensity.value)
    if (bottomDensity.value != null) s["--bkup-bottom-density"] = String(bottomDensity.value)
    return s
  })

  const topDensityProxy = computed<number>({
    get: () => topDensity.value ?? pageDensity.value,
    set: (v) => { topDensity.value = v },
  })
  const middleDensityProxy = computed<number>({
    get: () => middleDensity.value ?? pageDensity.value,
    set: (v) => { middleDensity.value = v },
  })
  const bottomDensityProxy = computed<number>({
    get: () => bottomDensity.value ?? pageDensity.value,
    set: (v) => { bottomDensity.value = v },
  })

  const allSynced = computed(() =>
    topDensity.value === null && middleDensity.value === null && bottomDensity.value === null,
  )

  function syncAreasToPage() {
    topDensity.value = null; middleDensity.value = null; bottomDensity.value = null
  }
  function resetDensities() {
    pageDensity.value = 1
    topDensity.value = null; middleDensity.value = null; bottomDensity.value = null
  }

  watch([pageDensity, topDensity, middleDensity, bottomDensity], ([p, t, m, b]) => {
    try {
      localStorage.setItem(DENSITY_KEY, JSON.stringify({
        page: p, top: t ?? 1, middle: m ?? 1, bottom: b ?? 1,
      }))
    } catch {}
  }, { deep: false })

  function colW(px: number) { return Math.round(px * middleDensityProxy.value) }

  return {
    pageDensity, topDensity, middleDensity, bottomDensity,
    topDensityProxy, middleDensityProxy, bottomDensityProxy,
    densityStyle, allSynced, syncAreasToPage, resetDensities, colW,
  }
}
```

```vue
    const record = await createBackup({

      scope: form.value.scope,
      tables: form.value.tables,
      output_dir: form.value.output_dir || null,
      name: form.value.name,
    })
    runningRecord.value = record
    backupStore.setOutputDir(form.value.output_dir)
```

```vue
  if (form.value.source_type === "upload") return !!form.value.upload_id
  if (form.value.source_type === "file") return !!form.value.source_path

  const msg = form.value.restore_mode === "cover"

      confirmButtonText: "确认",
      cancelButtonText: "取消",

      source_type: form.value.source_type,

      upload_id: form.value.upload_id ?? undefined,
      source_path: form.value.source_path || undefined,
      restore_mode: form.value.restore_mode,
      confirm: true,
    })

    backupStore.startPolling(res.restore_id, (r: any) => {

      } else if (r.status === "failed") {
        ElMessage.error("恢复失败: " + r.error_message)
        runningRestore.value = null
      }
    })
  } catch (e: any) {
    ElMessage.error(e?.message || "恢复失败")
  }
}
```

## 10. 页面状态约定

| 状态 | 显示 |
|---|---|
| 备份任务 `pending` | 等待轮询 |
| 备份任务 `running` | 进度条 |
| 备份任务 `success` | 绿色提示 + "下载"按钮 |
| 备份任务 `failed` | 红色提示 + 错误信息 |
| 备份任务 `interrupted` | 橙色提示 + "重新备份"按钮 |
| 网络错误 | ElMessage.error |
| 401 | axios 拦截器处理（自动 refresh / 跳转登录）|
